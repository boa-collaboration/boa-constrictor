45def root_to_https(root_url: str) -> str:
46 """Convert an xrootd URL to an HTTPS URL for eospublic.cern.ch.
47
48 Example:
49 root://eospublic.cern.ch//eos/opendata/... -> https://eospublic.cern.ch/eos/opendata/...
50 """
51 prefix = "root://eospublic.cern.ch/"
52 if not root_url.startswith(prefix):
53 return root_url # fallback: return as-is
54 path = root_url[len(prefix) :]
55 # Ensure single leading slash after host in https
56 if not path.startswith("/"):
57 path = "/" + path
58 return "https://eospublic.cern.ch" + path

69def download_file(url: str, dest: Path, force: bool = False) -> Path:
70 """Download a file from HTTPS or via xrdcp fallback into dest path.
71
72 Returns the path to the downloaded file.
73 """
74 dest.parent.mkdir(parents=True, exist_ok=True)
75 if dest.exists() and not force:
76 print(f"File already exists: {dest} (use --force to re-download)")
77 return dest
78
79 https_url = root_to_https(url)
80 print(f"Downloading from: {https_url}")
81
82 try:
83 with requests.get(https_url, stream=True, timeout=60) as r:
84 r.raise_for_status()
85 total = int(r.headers.get("Content-Length", 0))
86 with open(dest, "wb") as f, tqdm(
87 total=total if total > 0 else None,
88 unit="B",
89 unit_scale=True,
90 desc=f"Downloading {dest.name}",
91 ) as pbar:
92 for chunk in r.iter_content(chunk_size=1024 * 1024): # 1 MiB chunks
93 if chunk:
94 f.write(chunk)
95 if total > 0:
96 pbar.update(len(chunk))
97 return dest
98 except Exception as https_err:
99 print(f"HTTPS download failed: {https_err}")
100
101 # Fallback to xrdcp if available
102 if has_xrdcp():
103 print("Falling back to xrdcp…")
104 cmd = [
105 "xrdcp",
106 "-f",
107 url,
108 str(dest),
109 ]
110 try:
111 subprocess.run(cmd, check=True)
112 return dest
113 except subprocess.CalledProcessError as e:
114 raise RuntimeError(f"xrdcp failed with exit code {e.returncode}") from e
115 else:
116 raise RuntimeError("Both HTTPS and xrdcp download attempts failed.")

123 def is_within_directory(directory: Path, target: Path) -> bool:
124 try:
125 directory = directory.resolve(strict=False)
126 target = target.resolve(strict=False)
127 except Exception:
128 # resolve(strict=False) can still raise in some edge cases; fallback to simple check
129 pass
130 return os.path.commonpath([str(directory), str(target)]) == str(directory)
131
132 with tarfile.open(tar_path, mode="r:*") as tf:
133 members = tf.getmembers()
134 for m in members:
135 target_path = extract_dir / m.name
136 if not is_within_directory(extract_dir, target_path):
137 raise RuntimeError(f"Unsafe path in tar archive: {m.name}")
138 tf.extractall(path=extract_dir)

141def find_hepmc_file(search_dir: Path) -> Path:
142 """Find a HEPMC payload file after extraction.
143
144 Preference order:
145 1) First file with .hepmc (case-insensitive)
146 2) First file with .hepmc.gz (will be gunzipped on-the-fly)
147 3) If none found, choose the largest regular file
148 """
149 hepmc_candidates = []
150 gz_candidates = []
151 other_files = []
152 for root, _dirs, files in os.walk(search_dir):
153 for fname in files:
154 p = Path(root) / fname
155 if p.is_file():
156 lname = fname.lower()
157 if lname.endswith(".hepmc"):
158 hepmc_candidates.append(p)
159 elif lname.endswith(".hepmc.gz"):
160 gz_candidates.append(p)
161 else:
162 other_files.append(p)
163
164 if hepmc_candidates:
165 return hepmc_candidates[0]
166 if gz_candidates:
167 # On-the-fly gunzip into a temp file
168 import gzip
169
170 gz_path = gz_candidates[0]
171 tmp_out = gz_path.with_suffix("") # drop .gz
172 with gzip.open(gz_path, "rb") as fin, open(tmp_out, "wb") as fout:
173 shutil.copyfileobj(fin, fout)
174 return tmp_out
175
176 if other_files:
177 # Choose the largest file as a fallback heuristic
178 return max(other_files, key=lambda p: p.stat().st_size)
179
180 raise FileNotFoundError("No files found in the extracted archive")

183def write_truncated_copy(src: Path, dest: Path, limit_bytes: int) -> None:
184 """Write the first limit_bytes from src into dest."""
185 with open(src, "rb") as fin, open(dest, "wb") as fout:
186 remaining = limit_bytes
187 bufsize = 4 * 1024 * 1024 # 4 MiB chunks
188 total_written = 0
189 while remaining > 0:
190 to_read = min(bufsize, remaining)
191 chunk = fin.read(to_read)
192 if not chunk:
193 break
194 fout.write(chunk)
195 total_written += len(chunk)
196 remaining -= len(chunk)
197 size = dest.stat().st_size if dest.exists() else 0
198 if size < limit_bytes:
199 print(
200 f"Warning: source smaller than {limit_bytes} bytes. Wrote only {size} bytes to {dest.name}."
201 )

204def main(argv: Optional[list[str]] = None) -> int:
205 parser = argparse.ArgumentParser(description="Download and prepare HEPMC files")
206 parser.add_argument("--force", action="store_true", help="Re-download and overwrite outputs if present")
207 parser.add_argument(
208 "--url",
209 default=ROOT_URL,
210 help="Source URL (root:// or https://). Default is the ATLAS Open Data sample.",
211 )
212 args = parser.parse_args(argv)
213
214 workdir = Path(__file__).parent.resolve()
215 tar_dest = workdir / Path(args.url.split("/")[-1]) # keep original filename
216 full_out = workdir / "hepmc.hepmc"
217 trunc_out = workdir / "hepmc_200m.hepmc"
218
219 # Step 1: Download
220 downloaded = download_file(args.url, tar_dest, force=args.force)
221 print(f"Downloaded to: {downloaded}")
222
223 # Step 2: Extract safely to temp dir
224 with tempfile.TemporaryDirectory(prefix="hepmc_extract_") as tmpdir:
225 tmpdir_p = Path(tmpdir)
226 print(f"Extracting archive to: {tmpdir_p}")
227 safe_extract_tar(downloaded, tmpdir_p)
228
229 # Step 3: Locate HEPMC payload
230 hepmc_payload = find_hepmc_file(tmpdir_p)
231 print(f"Found HEPMC payload: {hepmc_payload}")
232
233 # Step 4: Copy full payload to hepmc.hepmc
234 if full_out.exists() and not args.force:
235 print(f"Skipping write; already exists: {full_out} (use --force to overwrite)")
236 else:
237 shutil.copyfile(hepmc_payload, full_out)
238 print(f"Wrote full payload to: {full_out}")
239
240 # Step 5: Write truncated 200 MiB copy
241 if trunc_out.exists() and not args.force:
242 print(f"Skipping write; already exists: {trunc_out} (use --force to overwrite)")
243 else:
244 write_truncated_copy(full_out, trunc_out, BYTES_200M)
245 print(f"Wrote 200 MiB truncated copy to: {trunc_out}")
246
247 return 0

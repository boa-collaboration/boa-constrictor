25def compress_hydra(
26 model,
27 data_bytes: bytes,
28 K: int = 4,
29 device: str = "cpu",
30 progress: bool = True,
31) -> tuple:
32 """Compress *data_bytes* with HydraBOA.
33
34 Returns
35 -------
36 compressed : np.ndarray[uint32] — range-coded bitstream
37 metadata : dict — needed for decompression
38 """
39 import constriction
40
41 model.eval().to(device)
42 data = np.frombuffer(data_bytes, dtype=np.uint8).copy()
43 orig_len = len(data)
44
45 # Pad to a multiple of K
46 remainder = len(data) % K
47 if remainder:
48 pad = K - remainder
49 data = np.concatenate([data, np.zeros(pad, dtype=np.uint8)])
50 else:
51 pad = 0
52
53 L = len(data)
54 T = L // K
55
56 enc = constriction.stream.queue.RangeEncoder()
57 fam = constriction.stream.model.Categorical(perfect=False)
58
59 cache = model.init_stream(max_chunks=T, batch_size=1, device=device)
60 prev_chunk: torch.Tensor | None = None
61
62 pbar = tqdm(total=T, disable=not progress, desc="Compress (HydraBOA)",
63 unit="chunk", mininterval=0.3)
64
65 for t in range(T):
66 # ── heavy step: backbone ──
67 H_t = model.step_backbone(prev_chunk, cache) # [1, D]
68
69 chunk_bytes = data[t * K : (t + 1) * K]
70 decoded_in_chunk: list[int] = []
71
72 # ── light steps: chained heads ──
73 for k in range(K):
74 if k == 0:
75 prev_bytes = None
76 else:
77 prev_bytes = torch.tensor(
78 [decoded_in_chunk], dtype=torch.long, device=device
79 )
80
81 logits = model.predict_head(k, H_t, prev_bytes) # [1, V]
82 probs = torch.softmax(logits, dim=-1).cpu().numpy().astype(np.float32)
83
84 sym = int(chunk_bytes[k])
85 enc.encode(np.array([sym], dtype=np.int32), fam, probs)
86 decoded_in_chunk.append(sym)
87
88 # Prepare the chunk we just encoded as input for the next backbone step
89 prev_chunk = torch.tensor(
90 [chunk_bytes.tolist()], dtype=torch.long, device=device
91 )
92 pbar.update(1)
93
94 pbar.close()
95
96 compressed = enc.get_compressed()
97 metadata = {
98 "orig_len": orig_len,
99 "padded_len": L,
100 "pad": pad,
101 "K": K,
102 "T": T,
103 }
104 return compressed, metadata

107def decompress_hydra(
108 model,
109 compressed: np.ndarray,
110 metadata: dict,
111 device: str = "cpu",
112 progress: bool = True,
113) -> bytes:
114 """Decompress a HydraBOA bitstream back to the original bytes."""
115 import constriction
116
117 model.eval().to(device)
118
119 K = metadata["K"]
120 T = metadata["T"]
121 orig_len = metadata["orig_len"]
122
123 dec = constriction.stream.queue.RangeDecoder(compressed)
124 fam = constriction.stream.model.Categorical(perfect=False)
125
126 cache = model.init_stream(max_chunks=T, batch_size=1, device=device)
127 output = np.empty(T * K, dtype=np.uint8)
128 prev_chunk: torch.Tensor | None = None
129
130 pbar = tqdm(total=T, disable=not progress, desc="Decompress (HydraBOA)",
131 unit="chunk", mininterval=0.3)
132
133 for t in range(T):
134 # ── heavy step ──
135 H_t = model.step_backbone(prev_chunk, cache)
136
137 decoded_in_chunk: list[int] = []
138
139 # ── light steps ──
140 for k in range(K):
141 if k == 0:
142 prev_bytes = None
143 else:
144 prev_bytes = torch.tensor(
145 [decoded_in_chunk], dtype=torch.long, device=device
146 )
147
148 logits = model.predict_head(k, H_t, prev_bytes)
149 probs = torch.softmax(logits, dim=-1).cpu().numpy().astype(np.float32)
150
151 sym = int(dec.decode(fam, probs)[0])
152 decoded_in_chunk.append(sym)
153 output[t * K + k] = sym
154
155 prev_chunk = torch.tensor(
156 [decoded_in_chunk], dtype=torch.long, device=device
157 )
158 pbar.update(1)
159
160 pbar.close()
161 return bytes(output[:orig_len])

168@torch.inference_mode()
169def compress_hydra_gpu(
170 model,
171 data_bytes: bytes,
172 K: int = 4,
173 device: str = "cuda",
174 progress: bool = True,
175 batch_streams: int = 5000,
176) -> tuple:
177 """Compress *data_bytes* with HydraBOA using the GPU range coder.
178
179 Returns
180 -------
181 compressed_list : list[np.ndarray[uint32]]
182 metadata : dict
183 """
184 if not _HAS_GPU_RC:
185 raise ImportError("gpu_range_coder is required for GPU compression")
186
187 model.eval().to(device)
188 data = np.frombuffer(data_bytes, dtype=np.uint8).copy()
189 orig_len = len(data)
190
191 remainder = len(data) % K
192 pad = (K - remainder) if remainder else 0
193 if pad:
194 data = np.concatenate([data, np.zeros(pad, dtype=np.uint8)])
195
196 L_total = len(data)
197 T_total = L_total // K
198 vocab_size = model.vocab_size
199
200 # Split into N chunks of equal size for batched GPU processing.
201 # Each "chunk" is one file-segment whose backbone timesteps are
202 # processed in lockstep across all N streams.
203 #
204 # We pick chunk_size so that N ≈ batch_streams, but the backbone
205 # sequence length stays reasonable (≤ ~4096 steps).
206 max_backbone_steps = 2500 # T per stream
207 chunk_bytes = max_backbone_steps * K # bytes per stream
208 N = max(1, (L_total + chunk_bytes - 1) // chunk_bytes)
209 N = min(N, batch_streams)
210 chunk_bytes = ((L_total + N - 1) // N)
211 # Align to K
212 chunk_bytes = ((chunk_bytes + K - 1) // K) * K
213 # Recalculate N after alignment
214 N = (L_total + chunk_bytes - 1) // chunk_bytes
215
216 T_per_chunk = chunk_bytes // K
217
218 # Pad data to N * chunk_bytes
219 padded_len = N * chunk_bytes
220 if len(data) < padded_len:
221 data = np.concatenate([data, np.zeros(padded_len - len(data), dtype=np.uint8)])
222
223 # Reshape into [N, T_per_chunk, K] and pack onto GPU
224 chunks_np = data[:padded_len].reshape(N, T_per_chunk, K)
225 chunks_gpu = torch.from_numpy(chunks_np.astype(np.int64)).to(device) # [N, T, K]
226
227 # Total symbols to encode = N * T_per_chunk * K
228 total_syms = N * T_per_chunk * K
229 # GPU range coder: N streams, vocab_size alphabet, total_syms max length
230 batch_rc = _gr.gpu.queue.RangeCoderBatch(N, vocab_size, T_per_chunk * K)
231
232 # Initialize backbone streaming state
233 cache = model.init_stream(max_chunks=T_per_chunk, batch_size=N, device=device)
234 prev_chunks = None # will be [N, K] for subsequent steps
235
236 pbar = tqdm(total=T_per_chunk * K, disable=not progress,
237 desc=f"Compress (HydraBOA GPU x{N})", unit="sym", mininterval=0.3)
238
239 for t in range(T_per_chunk):
240 # ── heavy step: backbone across all N streams ──
241 H_t = model.step_backbone(prev_chunks, cache, batch_size=N) # [N, D]
242
243 current_chunk = chunks_gpu[:, t, :] # [N, K]
244
245 # ── light steps: K chained heads ──
246 for k in range(K):
247 if k == 0:
248 prev_bytes = None
249 else:
250 prev_bytes = current_chunk[:, :k] # [N, k] — teacher-forced (known)
251
252 logits = model.predict_head(k, H_t, prev_bytes) # [N, V]
253 probs = torch.softmax(logits, dim=-1).to(torch.float32)
254
255 syms = current_chunk[:, k].to(torch.int32) # [N]
256 batch_rc.encode_step(syms, probs)
257
258 pbar.update(N)
259
260 prev_chunks = current_chunk # [N, K]
261
262 pbar.close()
263 batch_rc.finalize()
264 compressed_list = batch_rc.get_compressed_list()
265
266 metadata = {
267 "orig_len": orig_len,
268 "padded_len": padded_len,
269 "pad": padded_len - orig_len,
270 "K": K,
271 "T_per_chunk": T_per_chunk,
272 "N": N,
273 "chunk_bytes": chunk_bytes,
274 }
275 return compressed_list, metadata

278@torch.inference_mode()
279def decompress_hydra_gpu(
280 model,
281 compressed_list: list,
282 metadata: dict,
283 device: str = "cuda",
284 progress: bool = True,
285) -> bytes:
286 """Decompress HydraBOA bitstreams using the GPU range coder."""
287 if not _HAS_GPU_RC:
288 raise ImportError("gpu_range_coder is required for GPU decompression")
289
290 model.eval().to(device)
291
292 K = metadata["K"]
293 T_per_chunk = metadata["T_per_chunk"]
294 N = metadata["N"]
295 orig_len = metadata["orig_len"]
296 chunk_bytes = metadata["chunk_bytes"]
297 vocab_size = model.vocab_size
298
299 batch_rc = _gr.gpu.queue.RangeCoderBatch(N, vocab_size, T_per_chunk * K)
300 batch_rc.load_compressed_list(compressed_list)
301 batch_rc.init_decoder()
302
303 # Output buffer on GPU
304 output = torch.empty((N, T_per_chunk, K), dtype=torch.uint8, device=device)
305
306 cache = model.init_stream(max_chunks=T_per_chunk, batch_size=N, device=device)
307 prev_chunks = None
308 out_syms = torch.empty((N,), dtype=torch.int32, device=device)
309
310 pbar = tqdm(total=T_per_chunk * K, disable=not progress,
311 desc=f"Decompress (HydraBOA GPU x{N})", unit="sym", mininterval=0.3)
312
313 for t in range(T_per_chunk):
314 H_t = model.step_backbone(prev_chunks, cache, batch_size=N)
315
316 decoded_k = [] # accumulate decoded bytes for this chunk-step
317
318 for k in range(K):
319 if k == 0:
320 prev_bytes = None
321 else:
322 prev_bytes = torch.stack(decoded_k, dim=1) # [N, k]
323
324 logits = model.predict_head(k, H_t, prev_bytes)
325 probs = torch.softmax(logits, dim=-1).to(torch.float32)
326
327 batch_rc.decode_step(probs, out_syms)
328 output[:, t, k] = out_syms.to(torch.uint8)
329 decoded_k.append(out_syms.to(torch.long))
330
331 pbar.update(N)
332
333 prev_chunks = torch.stack(decoded_k, dim=1) # [N, K]
334
335 pbar.close()
336
337 # Flatten and trim
338 flat = output.reshape(-1).cpu().numpy()
339 return bytes(flat[:orig_len])

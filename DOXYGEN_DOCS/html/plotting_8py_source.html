43def reshape_to_records(arr: np.ndarray, ncols: int):
44 if arr.size % ncols != 0:
45 raise ValueError(f"Array length ({arr.size}) not divisible by ncols ({ncols})")

49def plot_columns(orig_rec, decomp_rec, names, out_dir: Path, nrows_to_plot=200,
50 alpha_orig=0.6, alpha_decomp=0.9,
51 create_hist=True, bins=20, hist_log=False):
52 out_dir.mkdir(parents=True, exist_ok=True)
53 ncols = orig_rec.shape[1]
54 # limit rows
55 rows = min(nrows_to_plot, orig_rec.shape[0])
56
57 for i in range(ncols):
58 col_name = names[i] if i < len(names) else f"col_{i}"
59 y_orig = orig_rec[:rows, i]
60 y_decomp = decomp_rec[:rows, i]
61
62 # filter NaNs (if any)
63 y1 = y_orig[~np.isnan(y_orig)]
64 y2 = y_decomp[~np.isnan(y_decomp)]
65 residual = y_orig - y_decomp
66
67 if create_hist:
68 # Histogram plot (counts vs feature value) + residuals panel
69 fig = plt.figure(figsize=(4.5, 6))
70 gs = fig.add_gridspec(2, 1, height_ratios=[3, 1], hspace=0.25)
71 ax_h = fig.add_subplot(gs[0])
72 ax_res = fig.add_subplot(gs[1])
73
74 # use same bin edges for both so counts are comparable
75 try:
76 _, bin_edges = np.histogram(y1, bins=bins)
77 ax_h.hist(y1, bins=bin_edges, alpha=alpha_orig, label='original')
78 ax_h.hist(y2, bins=bin_edges, alpha=alpha_decomp, label='decompressed')
79 except ValueError:
80 # fallback: let matplotlib choose bins
81 ax_h.hist(y1, bins=bins, alpha=alpha_orig, label='original')
82 ax_h.hist(y2, bins=bins, alpha=alpha_decomp, label='decompressed')
83
84 ax_h.set_xlabel(col_name, fontsize=9)
85 ax_h.set_ylabel('count', fontsize=9)
86 if hist_log:
87 ax_h.set_yscale('log')
88 ax_h.legend(fontsize='small')
89
90 ax_res.plot(range(rows), residual, color='black', lw=0.5, alpha=0.8)
91 ax_res.set_xlabel('index', fontsize=8)
92 ax_res.set_ylabel('$\\Delta_{residual}$', fontsize=8)
93 ax_res.tick_params(axis='both', labelsize=8)
94
95 out_file = out_dir / f"col_{i:02d}_{sanitize_filename(col_name)}_hist.png"
96 fig.savefig(out_file, dpi=150, bbox_inches='tight')
97 plt.close(fig)
98 else:
99 # Only residual plot
100 fig, ax = plt.subplots(figsize=(7, 3.0))
101 ax.plot(range(rows), residual, color='black', lw=0.6, alpha=0.9)
102 ax.set_title(col_name, fontsize=10)
103 ax.set_xlabel('row index (first N)', fontsize=9)
104 ax.set_ylabel('$\\Delta$ = original - decompressed', fontsize=9)
105 ax.tick_params(axis='both', labelsize=9)
106

108 fig.savefig(out_file, dpi=150, bbox_inches='tight')

112def apply_hep_style(style: str):
113 """Apply a HEP plotting style.
114
115 style: one of 'none' (do nothing), 'atlas', 'hep', 'cms', or 'mplhep'.
116 Tries to use mplhep if installed; otherwise falls back to a compact matplotlib rcParams
117 that approximates common HEP plotting aesthetics (sans logos).
118 """
119 style = (style or "").lower()
120 if style in ("", "none", "no"):
121 return
122
123 try:
124 import mplhep as hep
125 # mplhep supports a style.use API that accepts common experiment strings
126 try:
127 # prefer explicit names
128 if style in ("atlas", "mplhep"):
129 hep.style.use("ATLAS")
130 elif style == "cms":
131 hep.style.use("CMS")
132 else:
133 # generic 'hep'
134 hep.style.use("hep")
135 except Exception:
136 # fallback to generic style application
137 try:
138 hep.style.use(style.upper())
139 except Exception:
140 plt.style.use("seaborn-whitegrid")
141 return
142 except Exception:
143 warnings.warn("mplhep not available; using fallback Matplotlib settings for HEP-like style")
144
145 # Fallback rcParams setting (small, clean, grid, ticks outward)
146 plt.rcParams.update({
147 "figure.figsize": (7, 4.5),
148 "font.size": 10,
149 "axes.titlesize": 11,
150 "axes.labelsize": 10,
151 "legend.fontsize": 9,
152 "xtick.labelsize": 9,
153 "ytick.labelsize": 9,
154 "axes.grid": True,
155 "grid.linestyle": "--",
156 "grid.alpha": 0.4,
157 "axes.linewidth": 0.8,
158 "xtick.direction": "out",
159 "ytick.direction": "out",
160 "legend.frameon": False,
161 })

163def sanitize_filename(s: str) -> str:
164 # produce a short safe filename segment from the column name
165 import re
166
167 s2 = re.sub(r"[^0-9A-Za-z._-]", "_", s)
168 # condense multiple underscores
169 s2 = re.sub(r"_+", "_", s2)
170 # limit length
171 return s2[:64]

178 p.add_argument("--dtype", default="float32", help="NumPy dtype used when interpreting bytes (default: float32)")
179 p.add_argument("--nrows", type=int, default=200, help="Number of rows to plot (default: 200)")
180 p.add_argument("--out-dir", type=Path, default=Path(__file__).parent / "plots", help="Output directory for plots")

182 p.add_argument("--alpha-decompressed", type=float, default=0.9, help="Alpha/transparency for decompressed histogram (default: 0.9)")
183 # histogram options
184 p.add_argument("--bins", type=int, default=50, help="Number of bins for histograms (default: 100)")
185 p.add_argument("--no-hist", dest='hist', action='store_false', help="Do not create histogram plots")
186 p.add_argument("--hist-log", dest='hist_log', action='store_true', help="Use log scale for histogram counts")
187 p.add_argument("--style", choices=["none", "atlas", "hep", "cms", "mplhep"], default="hep",
188 help="Plotting style to apply. Uses mplhep when available (choices: none, atlas, hep, cms, mplhep).")
189 p.set_defaults(hist=True)
190 args = p.parse_args()
191
192 dtype = np.dtype(args.dtype)
193 ncols = len(COLUMN_NAMES)
194
195 orig_arr = load_flat_float_array(args.original, dtype=dtype)
196 decomp_arr = load_flat_float_array(args.decompressed, dtype=dtype)
197
198 orig_rec = reshape_to_records(orig_arr, ncols)
199 decomp_rec = reshape_to_records(decomp_arr, ncols)

203
204 # Apply chosen plotting style (mplhep preferred, otherwise use fallback)
205 apply_hep_style(args.style)
206
207 plot_columns(orig_rec, decomp_rec, COLUMN_NAMES, args.out_dir, nrows_to_plot=args.nrows,
208 alpha_orig=args.alpha_original, alpha_decomp=args.alpha_decompressed,
209 create_hist=args.hist, bins=args.bins, hist_log=args.hist_log)
210 print(f"Plots written to {args.out_dir}")

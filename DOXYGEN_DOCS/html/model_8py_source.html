126# ── minGRU (Feng et al., 2024 — "Were RNNs All We Needed?") ─────
127#
128# z_t = σ(W_z · x_t + b_z) gate (input-only)
129# h̃_t = W_h · x_t + b_h candidate (input-only)
130# h_t = (1 − z_t) ⊙ h_{t−1} + z_t ⊙ h̃_t
131#
132# No hidden-to-hidden weights ⇒ the linear recurrence is amenable to
133# a parallel prefix (associative) scan during training. The sequential
134# fallback here is correct and simple; swap in a CUDA scan kernel for
135# O(L / P) wall-clock on very long sequences.
136# ──────────────────────────────────────────────────────────────────
137

145class _MinGRUCell(nn.Module):
146 """Minimal GRU cell — log-space parallel scan (Feng et al., 2024).
147
148 Training uses the numerically stable log-space formulation with
149 Heinsen's associative scan (``cumsum`` + ``logcumsumexp`` — two CUDA
150 kernels instead of L sequential Python iterations).
151
152 Hidden states are constrained to be positive via ``g()`` (Appendix
153 B.3 of the paper). The single-step path (``step``) uses the same
154 ``g()`` so training and inference are consistent.
155 """
156
157 def __init__(self, d_model: int):
158 super().__init__()
159 self.linear_z = nn.Linear(d_model, d_model)
160 self.linear_h = nn.Linear(d_model, d_model)
161 self.d_model = d_model
162
163 def forward(self, x, h_prev=None):
164 """x: [B, L, D] → [B, L, D] (parallel log-space scan)"""
165 B, L, D = x.shape
166 gate_logits = self.linear_z(x) # [B, L, D]
167 h_candidate = self.linear_h(x) # [B, L, D]
168
169 # Log-space parallel scan (Appendix B.3)
170 log_coeffs = -F.softplus(gate_logits) # log(1 − σ(z))
171 log_z = -F.softplus(-gate_logits) # log(σ(z))
172 log_values = log_z + _mingru_log_g(h_candidate) # log(z · g(h̃))
173
174 if h_prev is not None:
175 log_values = torch.cat([h_prev.clamp(min=1e-8).log().unsqueeze(1),
176 log_values], dim=1)
177 log_coeffs = F.pad(log_coeffs, (0, 0, 1, 0))
178
179 h = heinsen_associative_scan_log(log_coeffs, log_values)
180 return h[:, -L:]
181
182 def step(self, x, h_prev):
183 """Single-step recurrence. x, h_prev: [B, D]"""
184 z = torch.sigmoid(self.linear_z(x))
185 h_tilde = _mingru_g(self.linear_h(x))
186 h = (1 - z) * h_prev + z * h_tilde
187 return h, h

220# ── Transformer (RoPE + GQA + SwiGLU + Attention-Sink Sliding-Window)
221#
222# Three targeted fixes over the vanilla version:
223#
224# 1. **No train/test mismatch** – windowed causal attention is used
225# during training too (when the sequence is longer than the window),
226# so the model never learns to rely on context it won't have during
227# streaming. The first ``n_sink`` positions are always visible to
228# every query (attention sinks).
229#
230# 2. **SwiGLU FFN** – gated feed-forward (Shazeer 2020, used by LLaMA /
231# Mistral / DeepSeek) gives better parameter efficiency than the
232# plain GELU FFN.
233#
234# 3. **Grouped-Query Attention (GQA)** – shares KV heads across
235# multiple query heads (Ainslie et al. 2023), shrinking the KV
236# cache by ``n_heads / n_kv_heads`` and freeing capacity.
237#
238# Streaming: KV cache keeps the first ``n_sink`` tokens plus a sliding
239# window of the most recent ``window_size`` tokens, bounding memory to
240# O(n_sink + window_size) per layer — same as before but now matched to
241# training.
242#
243# References
244# Xiao et al., "Efficient Streaming Language Models with Attention
245# Sinks", 2023.
246# Ainslie et al., "GQA: Training Generalized Multi-Query Transformer
247# Models from Multi-Head Checkpoints", 2023.
248# Shazeer, "GLU Variants Improve Transformer", 2020.
249# ──────────────────────────────────────────────────────────────────

251class _RMSNorm(nn.Module):
252 """Root-Mean-Square Layer Normalization (Zhang & Sennrich 2019)."""
253
254 def __init__(self, d_model: int, eps: float = 1e-6):
255 super().__init__()
256 self.weight = nn.Parameter(torch.ones(d_model))
257 self.eps = eps
258
259 def forward(self, x):
260 rms = x.float().pow(2).mean(-1, keepdim=True).add(self.eps).rsqrt()
261 return (x.float() * rms).to(x.dtype) * self.weight

264class _SwiGLU(nn.Module):
265 """SwiGLU feed-forward: gate(x) * V(x) mapped back to d_model.
266
267 Uses 8/3·d_model hidden dim (≈ 2.67x), similar param count to 4x GELU
268 FFN but with gating.
269 """
270
271 def __init__(self, d_model: int):
272 super().__init__()
273 hidden = int(8 * d_model / 3)
274 # Round to nearest multiple of 8 for tensor-core efficiency
275 hidden = ((hidden + 7) // 8) * 8
276 self.w_gate = nn.Linear(d_model, hidden, bias=False)
277 self.w_up = nn.Linear(d_model, hidden, bias=False)
278 self.w_down = nn.Linear(hidden, d_model, bias=False)
279
280 def forward(self, x):
281 return self.w_down(F.silu(self.w_gate(x)) * self.w_up(x))

284def _make_sliding_window_mask(L: int, window_size: int, n_sink: int,
285 device: torch.device) -> torch.Tensor:
286 """Build a [L, L] bool attention mask for windowed + sink attention.
287
288 Entry ``mask[i, j] = True`` means query-i **cannot** attend to key-j.
289 Compatible with ``F.scaled_dot_product_attention(attn_mask=...)`` which
290 treats True as "masked out" (−inf) when the mask dtype is bool.
291 """
292 # Start with everything masked
293 mask = torch.ones(L, L, dtype=torch.bool, device=device)
294 row = torch.arange(L, device=device).unsqueeze(1)
295 col = torch.arange(L, device=device).unsqueeze(0)
296
297 # (a) Causal: can only attend to past+self
298 causal = col <= row
299 # (b) Within sliding window
300 in_window = (row - col) < window_size
301 # (c) Sink positions (first n_sink tokens are always visible)
302 is_sink = col < n_sink
303
304 visible = causal & (in_window | is_sink)
305 mask = ~visible # True = masked-out for SDPA
306 return mask

309class TransformerBlock(nn.Module):
310 """Transformer block: RoPE + GQA + SwiGLU + sliding-window attention.
311
312 Parameters
313 ----------
314 d_model : int – hidden dimension.
315 n_heads : int – number of **query** heads (default 4).
316 n_kv_heads : int – number of KV heads for GQA. Must divide n_heads
317 evenly. Default = n_heads (= standard MHA).
318 Set to 1 for Multi-Query Attention.
319 window_size : int – sliding-window width (default 4096).
320 n_sink : int – attention-sink positions (default 4).
321 """
322
323 def __init__(self, d_model: int, n_heads: int = 4, n_kv_heads: int = 0,
324 window_size: int = 4096, n_sink: int = 4):
325 super().__init__()
326 if n_kv_heads <= 0:
327 n_kv_heads = n_heads # default: standard MHA
328 assert d_model % n_heads == 0, \
329 f"d_model ({d_model}) must be divisible by n_heads ({n_heads})"
330 assert n_heads % n_kv_heads == 0, \
331 f"n_heads ({n_heads}) must be divisible by n_kv_heads ({n_kv_heads})"
332
333 self.n_heads = n_heads
334 self.n_kv_heads = n_kv_heads
335 self.n_rep = n_heads // n_kv_heads # GQA repeat factor
336 self.d_head = d_model // n_heads
337 self.d_model = d_model
338 self.window_size = window_size
339 self.n_sink = n_sink
340
341 self.ln1 = _RMSNorm(d_model)
342 # Separate Q and KV projections for GQA
343 self.q_proj = nn.Linear(d_model, n_heads * self.d_head, bias=False)
344 self.kv_proj = nn.Linear(d_model, 2 * n_kv_heads * self.d_head, bias=False)
345 self.out_proj = nn.Linear(d_model, d_model, bias=False)
346 self.ln2 = _RMSNorm(d_model)
347 self.ff = _SwiGLU(d_model)
348
349 # RoPE inverse frequencies (not a trained parameter)
350 inv_freq = 1.0 / (
351 10000.0 ** (torch.arange(0, self.d_head, 2).float() / self.d_head)
352 )
353 self.register_buffer("rope_inv_freq", inv_freq, persistent=False)
354
355 # ---- RoPE helpers ------------------------------------------------
356
357 def _apply_rope(self, x, positions):
358 """Apply rotary position embeddings.
359
360 x : [B, n_heads, L, d_head]
361 positions : LongTensor [L]
362 """
363 freqs = torch.outer(positions.float(), self.rope_inv_freq) # [L, dh/2]
364 cos = freqs.cos().unsqueeze(0).unsqueeze(0) # [1,1,L,dh/2]
365 sin = freqs.sin().unsqueeze(0).unsqueeze(0)
366 x1 = x[..., : self.d_head // 2]
367 x2 = x[..., self.d_head // 2 :]
368 return torch.cat([x1 * cos - x2 * sin,
369 x2 * cos + x1 * sin], dim=-1)
370
371 def _expand_kv(self, kv):
372 """Repeat KV heads to match the number of query heads (GQA)."""
373 if self.n_rep == 1:
374 return kv
375 B, n_kv, L, D = kv.shape
376 return kv[:, :, None, :, :].expand(B, n_kv, self.n_rep, L, D) \
377 .reshape(B, self.n_heads, L, D)
378
379 # ---- full-sequence forward (training) ----------------------------
380 # Uses the SAME sliding-window + sink mask as inference so there is
381 # no distribution shift.
382
383 def forward(self, x, inference_params=None):
384 B, L, D = x.shape
385 y = self.ln1(x)
386
387 q = self.q_proj(y).reshape(B, L, self.n_heads, self.d_head).transpose(1, 2)
388 kv = self.kv_proj(y).reshape(B, L, 2, self.n_kv_heads, self.d_head)
389 k, v = kv.unbind(dim=2)
390 k = k.transpose(1, 2) # [B, n_kv, L, dh]
391 v = v.transpose(1, 2)
392
393 positions = torch.arange(L, device=x.device)
394 q = self._apply_rope(q, positions)
395 k = self._apply_rope(k, positions)
396
397 # Expand KV heads for GQA
398 k = self._expand_kv(k)
399 v = self._expand_kv(v)
400
401 # Build sliding-window + sink mask (matches inference behaviour)
402 if L <= self.window_size + self.n_sink:
403 # Short sequence — plain causal is equivalent & faster
404 y = F.scaled_dot_product_attention(q, k, v, is_causal=True)
405 else:
406 mask = _make_sliding_window_mask(L, self.window_size, self.n_sink,
407 device=x.device)
408 y = F.scaled_dot_product_attention(q, k, v, attn_mask=mask)
409
410 y = y.transpose(1, 2).reshape(B, L, D)
411 y = self.out_proj(y)
412 x = x + y
413
414 y = self.ln2(x)
415 y = self.ff(y)
416 return x + y

418 # ---- streaming helpers -------------------------------------------
419
420 def init_cache(self, batch_size: int, device):
421 """Return empty KV cache: (k, v, step_count)."""
422 k = torch.zeros(batch_size, self.n_kv_heads, 0, self.d_head, device=device)
423 v = torch.zeros(batch_size, self.n_kv_heads, 0, self.d_head, device=device)
424 return (k, v, 0)
425
426 def step(self, x, cache):
427 """Single-token step with attention-sink + sliding-window eviction.
428
429 x : [B, D]
430 cache : (k_cache, v_cache, step_count)
431 """
432 k_cache, v_cache, step_count = cache
433 B = x.shape[0]
434
435 y = self.ln1(x).unsqueeze(1) # [B, 1, D]
436 q = self.q_proj(y).reshape(B, 1, self.n_heads, self.d_head).transpose(1, 2)
437 kv = self.kv_proj(y).reshape(B, 1, 2, self.n_kv_heads, self.d_head)
438 k_new, v_new = kv.unbind(dim=2)
439 k_new = k_new.transpose(1, 2) # [B, n_kv, 1, dh]
440 v_new = v_new.transpose(1, 2)
441
442 # RoPE at the current absolute position
443 pos_t = torch.tensor([step_count], device=x.device)
444 q = self._apply_rope(q, pos_t)
445 k_new = self._apply_rope(k_new, pos_t)
446
447 # Append to cache
448 k_cache = torch.cat([k_cache, k_new], dim=2)
449 v_cache = torch.cat([v_cache, v_new], dim=2)
450 step_count += 1
451
452 # Evict: keep first n_sink entries + last window_size entries
453 max_cache = self.n_sink + self.window_size
454 if k_cache.shape[2] > max_cache:
455 k_cache = torch.cat([k_cache[:, :, :self.n_sink],
456 k_cache[:, :, -self.window_size:]], dim=2)
457 v_cache = torch.cat([v_cache[:, :, :self.n_sink],
458 v_cache[:, :, -self.window_size:]], dim=2)
459
460 # Expand KV for GQA and attend
461 k_exp = self._expand_kv(k_cache)
462 v_exp = self._expand_kv(v_cache)
463
464 # Expand Q to match — already has n_heads
465 scale = self.d_head ** -0.5
466 attn = (q @ k_exp.transpose(-2, -1)) * scale
467 attn = F.softmax(attn, dim=-1)
468 y = (attn @ v_exp) # [B,nh,1,dh]
469 y = y.transpose(1, 2).reshape(B, 1, self.d_model)
470 y = self.out_proj(y).squeeze(1) # [B, D]
471
472 x = x + y
473 y = self.ln2(x)
474 y = self.ff(y)
475 return x + y, (k_cache, v_cache, step_count)

478# ── RWKV-6 (Peng et al., 2024 — "Eagle and Finch") ──────────────
479#
480# Linear-complexity RNN with data-dependent decay and matrix-valued
481# hidden state. Parallelisable via a prefix-sum scan; sequential
482# fallback here for portability.
483#
484# Time-mixing:
485# Token-shift interpolation → R, K, V, G projections
486# Data-dependent decay: w_t = base_w + W_w(x_t)
487# Matrix state update: S_t = diag(exp(w_t)) · S_{t-1} + k_t v_t^T
488# Output: o_t = (S_t @ r_t) ⊙ gate_t
489# ──────────────────────────────────────────────────────────────────

496 parallel implementation (O(C²) matmuls per chunk, O(L/C) sequential
497 chunk iterations). Both paths are numerically equivalent.
498 """

507

544 def _chunk_wkv(self, r, k, v, w, initial_state):
545 """Chunk-wise parallel WKV (pure PyTorch, works on CPU & GPU).
546
547 Args:
548 r, k, v: [B, L, H, dh]
549 w: [B, L, H, dh] (log-decay, negative)
550 initial_state: [B, H, dh, dh] or None
551 Returns:
552 out: [B, L, H, dh], state: [B, H, dh, dh]
553 """
554 B, L, H, dh = r.shape
555 C = self.chunk_size
556
557 # Pad to multiple of C
558 pad = (C - L % C) % C
559 if pad > 0:
560 r = F.pad(r, (0, 0, 0, 0, 0, pad))
561 k = F.pad(k, (0, 0, 0, 0, 0, pad))
562 v = F.pad(v, (0, 0, 0, 0, 0, pad))
563 w = F.pad(w, (0, 0, 0, 0, 0, pad))
564 Lp = L + pad
565 nc = Lp // C
566
567 r = r.view(B, nc, C, H, dh)
568 k = k.view(B, nc, C, H, dh)
569 v = v.view(B, nc, C, H, dh)
570 w = w.view(B, nc, C, H, dh)
571
572 w_cumsum = w.cumsum(dim=2)
573
574 # Intra-chunk: causal attention with decay
575 rel_log = w_cumsum.unsqueeze(3) - w_cumsum.unsqueeze(2)
576 causal = torch.tril(torch.ones(C, C, device=r.device, dtype=torch.bool))
577 causal = causal.view(1, 1, C, C, 1, 1)
578
579 attn = (r.unsqueeze(3) * k.unsqueeze(2) * torch.exp(rel_log))
580 attn = attn.masked_fill(~causal, 0.0).sum(dim=-1) # [B, nc, Ct, Cs, H]
581 o_intra = torch.einsum('bncsh,bnshd->bnchd',
582 attn.permute(0, 1, 2, 3, 4), v)
583
584 # Inter-chunk: propagate recurrent state
585 state = initial_state if initial_state is not None else r.new_zeros(B, H, dh, dh)
586 inter_list = []
587 for c_idx in range(nc):
588 r_c = r[:, c_idx]
589 decay_c = torch.exp(w_cumsum[:, c_idx])
590 o_inter_c = torch.einsum('bhde,bche->bchd', state, r_c * decay_c)
591 inter_list.append(o_inter_c)
592
593 total_decay = torch.exp(w_cumsum[:, c_idx, -1])
594 state = state * total_decay.unsqueeze(-1)
595 kv_decay = torch.exp(w_cumsum[:, c_idx, -1:] - w_cumsum[:, c_idx])
596 state = state + torch.einsum('bchd,bche->bhde',
597 k[:, c_idx] * kv_decay, v[:, c_idx])
598
599 o_inter = torch.stack(inter_list, dim=1)
600 out = (o_intra + o_inter).reshape(B, Lp, H, dh)
601 if pad > 0:
602 out = out[:, :L]
603 return out, state
604
605 # ── Forward (selects FLA kernels or PyTorch fallback) ─────────────
606
607 def forward(self, x, last_x=None, state=None):
608 """x: [B, L, D] → ([B, L, D], last_x_new, state_new)."""
609 B, L, D = x.shape
610 H, dh = self.n_heads, self.d_head
611 x_prev = self._token_shift(x, last_x)
612
613 # Interpolated inputs
614 r = self.W_r(x * self.mix_r + x_prev * (1 - self.mix_r)).view(B, L, H, dh)
615 k = self.W_k(x * self.mix_k + x_prev * (1 - self.mix_k)).view(B, L, H, dh)
616 v = self.W_v(x * self.mix_v + x_prev * (1 - self.mix_v)).view(B, L, H, dh)
617 g = torch.sigmoid(self.W_g(x * self.mix_g + x_prev * (1 - self.mix_g)))
618
619 # L2-normalize k per head (prevents state blow-up)
620 k = F.normalize(k, p=2, dim=-1)
621
622 w = self._decay(self.base_decay, self.W_decay(x).view(B, L, H, dh))
623
624 # Dispatch to FLA Triton kernels when on CUDA, else PyTorch fallback
625 if _HAS_FLA_V6 and x.is_cuda:
626 fn = _fla_fused_recurrent_rwkv6 if L <= 64 else _fla_chunk_rwkv6
627 out, s = fn(
628 r=r, k=k, v=v, w=w,
629 u=self.bonus,
630 scale=1.0,
631 initial_state=state,
632 output_final_state=True,
633 )
634 else:
635 out, s = self._chunk_wkv(r, k, v, w, state)
636
637 out = out.reshape(B, L, D)
638 out = self.ln_out(out.transpose(1, 2)).transpose(1, 2)
639 out = out * g
640 return self.W_o(out), x[:, -1], s

642 def step(self, x, last_x, state):
643 """Single-token step. x, last_x: [B, D], state: [B, H, dh, dh]."""
644 B = x.shape[0]
645 H, dh = self.n_heads, self.d_head
646
647 r = self.W_r(x * self.mix_r + last_x * (1 - self.mix_r)).view(B, H, dh)
648 k = self.W_k(x * self.mix_k + last_x * (1 - self.mix_k)).view(B, H, dh)
649 v = self.W_v(x * self.mix_v + last_x * (1 - self.mix_v)).view(B, H, dh)
650 g = torch.sigmoid(self.W_g(x * self.mix_g + last_x * (1 - self.mix_g)))
651
652 k = F.normalize(k, p=2, dim=-1)
653
654 w = self._decay(self.base_decay, self.W_decay(x).view(B, H, dh))
655 decay = torch.exp(w)
656
657 state = state * decay.unsqueeze(-1) + torch.einsum('bhd,bhe->bhde', k, v)
658 o = torch.einsum('bhde,bhd->bhe', state, r).reshape(B, self.d_model)
659 o = self.ln_out(o.unsqueeze(-1)).squeeze(-1)
660 o = o * g
661 return self.W_o(o), x, state

699# ── RWKV-7 "Goose" (Peng et al., 2025) ─────────────────────────
700#
701# Extends RWKV-6 with a *non-diagonal* state transition matrix,
702# enabling cross-dimension state interactions (beyond TC⁰).
703#
704# State update (per head):
705# S_t = (diag(w_t) + a_t b_t^T) · S_{t-1} + k_t v_t^T
706# o_t = S_t @ r_t
707#
708# The rank-1 term a_t b_t^T lets the model perform "copy"
709# state transitions and recognize all regular languages.
710#
711# Uses FLA Triton kernels on CUDA, sequential fallback on CPU.
712# ──────────────────────────────────────────────────────────────────

714class _RWKV7TimeMix(nn.Module):
715 """RWKV-7 time-mixing with non-diagonal state transition."""
716
717 def __init__(self, d_model: int, n_heads: int = 4, chunk_size: int = 32):
718 super().__init__()
719 assert d_model % n_heads == 0
720 self.n_heads = n_heads
721 self.d_head = d_model // n_heads
722 self.d_model = d_model
723 self.chunk_size = chunk_size
724
725 # Token-shift interpolation
726 self.mix_r = nn.Parameter(torch.ones(d_model) * 0.5)
727 self.mix_k = nn.Parameter(torch.ones(d_model) * 0.5)
728 self.mix_v = nn.Parameter(torch.ones(d_model) * 0.5)
729 self.mix_g = nn.Parameter(torch.ones(d_model) * 0.5)
730 self.mix_a = nn.Parameter(torch.ones(d_model) * 0.5)
731
732 # Projections
733 self.W_r = nn.Linear(d_model, d_model, bias=False)
734 self.W_k = nn.Linear(d_model, d_model, bias=False)
735 self.W_v = nn.Linear(d_model, d_model, bias=False)
736 self.W_g = nn.Linear(d_model, d_model, bias=False)
737 self.W_o = nn.Linear(d_model, d_model, bias=False)
738
739 # Non-diagonal state transition: a, b projections (zero-init)
740 self.W_a = nn.Linear(d_model, d_model, bias=False)
741 self.W_b = nn.Linear(d_model, d_model, bias=False)
742 nn.init.zeros_(self.W_a.weight)
743 nn.init.zeros_(self.W_b.weight)
744
745 # Data-dependent decay (official: -softplus(-x) - 0.5)
746 self.base_decay = nn.Parameter(torch.zeros(n_heads, self.d_head) + 0.5)
747 self.W_decay = nn.Linear(d_model, n_heads * self.d_head, bias=False)
748 nn.init.zeros_(self.W_decay.weight)
749
750 # Bonus (position-0 bias)
751 self.bonus = nn.Parameter(torch.zeros(n_heads, self.d_head))
752
753 self.ln_out = nn.GroupNorm(n_heads, d_model)

755 def _token_shift(self, x, last_x):
756 if last_x is None:
757 last_x = torch.zeros_like(x[:, 0])
758 return torch.cat([last_x.unsqueeze(1), x[:, :-1]], dim=1)

760 @staticmethod
761 def _decay(base, delta):
762 """-softplus(-(base + delta)) - 0.5 → always < -0.5."""
763 return -F.softplus(-(base + delta)) - 0.5

765 def _chunk_sequential_wkv(self, r, k, v, w, a, b, initial_state):
766 """Chunk-sequential WKV with non-diagonal transition.
767
768 Processes chunks of size C sequentially, but vectorises
769 the per-token state readout within each chunk so the
770 Python loop runs L/C times instead of L times.
771 """
772 B, L, H, dh = r.shape
773 C = self.chunk_size
774 s = r.new_zeros(B, H, dh, dh) if initial_state is None else initial_state
775
776 # Pad to multiple of C
777 pad = (C - L % C) % C
778 if pad > 0:
779 r = F.pad(r, (0, 0, 0, 0, 0, pad))
780 k = F.pad(k, (0, 0, 0, 0, 0, pad))
781 v = F.pad(v, (0, 0, 0, 0, 0, pad))
782 w = F.pad(w, (0, 0, 0, 0, 0, pad))
783 a = F.pad(a, (0, 0, 0, 0, 0, pad))
784 b = F.pad(b, (0, 0, 0, 0, 0, pad))
785 Lp = L + pad
786 nc = Lp // C
787
788 all_outputs = []
789 for ci in range(nc):
790 sl = slice(ci * C, (ci + 1) * C)
791 r_c, k_c, v_c = r[:, sl], k[:, sl], v[:, sl]
792 w_c, a_c, b_c = w[:, sl], a[:, sl], b[:, sl]
793
794 chunk_out = []
795 for t in range(C):
796 decay = torch.exp(w_c[:, t])
797 bS = torch.einsum('bhd,bhde->bhe', b_c[:, t], s)
798 ab_S = torch.einsum('bhd,bhe->bhde', a_c[:, t], bS)
799 s = s * decay.unsqueeze(-1) + ab_S + \
800 torch.einsum('bhd,bhe->bhde', k_c[:, t], v_c[:, t])
801 chunk_out.append(torch.einsum('bhde,bhd->bhe', s, r_c[:, t]))
802 all_outputs.append(torch.stack(chunk_out, dim=1))
803
804 out = torch.cat(all_outputs, dim=1)
805 if pad > 0:
806 out = out[:, :L]
807 return out, s
808
809 def forward(self, x, last_x=None, state=None):
810 """x: [B, L, D] → ([B, L, D], last_x_new, state_new)."""
811 B, L, D = x.shape
812 H, dh = self.n_heads, self.d_head
813 x_prev = self._token_shift(x, last_x)
814
815 r = self.W_r(x * self.mix_r + x_prev * (1 - self.mix_r)).view(B, L, H, dh)
816 k = self.W_k(x * self.mix_k + x_prev * (1 - self.mix_k)).view(B, L, H, dh)
817 v = self.W_v(x * self.mix_v + x_prev * (1 - self.mix_v)).view(B, L, H, dh)
818 g = torch.sigmoid(self.W_g(x * self.mix_g + x_prev * (1 - self.mix_g)))
819
820 # L2-normalize k per head (prevents state blow-up)
821 k = F.normalize(k, p=2, dim=-1)
822
823 # Non-diagonal state transition vectors
824 x_a = x * self.mix_a + x_prev * (1 - self.mix_a)
825 a = torch.sigmoid(self.W_a(x_a)).view(B, L, H, dh)
826 b = self.W_b(x_a).view(B, L, H, dh)
827
828 w = self._decay(self.base_decay, self.W_decay(x).view(B, L, H, dh))
829
830 # Dispatch: FLA Triton kernels on CUDA, else PyTorch fallback
831 if _HAS_FLA_V7 and x.is_cuda:
832 fn = _fla_fused_recurrent_rwkv7 if L <= 64 else _fla_chunk_rwkv7
833 out, s = fn(
834 r=r, w=w, k=k, v=v,
835 a=-a, b=a * b, # FLA convention: a→ -kappa, b→ kappa*a
836 scale=1.0,
837 initial_state=state,
838 output_final_state=True,
839 )
840 else:
841 if not _HAS_FLA_V7 and x.is_cuda:
842 import warnings
843 warnings.warn(
844 "RWKV-7 on CUDA without FLA is slow. "
845 "Install: pip install flash-linear-attention",
846 stacklevel=2,
847 )
848 out, s = self._chunk_sequential_wkv(r, k, v, w, a, b, state)
849
850 out = out.reshape(B, L, D)
851 out = self.ln_out(out.transpose(1, 2)).transpose(1, 2)
852 out = out * g
853 return self.W_o(out), x[:, -1], s

855 def step(self, x, last_x, state):
856 """Single-token step. x, last_x: [B, D], state: [B, H, dh, dh]."""
857 B = x.shape[0]
858 H, dh = self.n_heads, self.d_head
859
860 r = self.W_r(x * self.mix_r + last_x * (1 - self.mix_r)).view(B, H, dh)
861 k = self.W_k(x * self.mix_k + last_x * (1 - self.mix_k)).view(B, H, dh)
862 v = self.W_v(x * self.mix_v + last_x * (1 - self.mix_v)).view(B, H, dh)
863 g = torch.sigmoid(self.W_g(x * self.mix_g + last_x * (1 - self.mix_g)))
864
865 k = F.normalize(k, p=2, dim=-1)
866
867 x_a = x * self.mix_a + last_x * (1 - self.mix_a)
868 a = torch.sigmoid(self.W_a(x_a)).view(B, H, dh)
869 b = self.W_b(x_a).view(B, H, dh)
870
871 w = self._decay(self.base_decay, self.W_decay(x).view(B, H, dh))
872 decay = torch.exp(w)
873
874 # S = diag(decay) * S + (a b^T) @ S + k v^T
875 bS = torch.einsum('bhd,bhde->bhe', b, state)
876 ab_S = torch.einsum('bhd,bhe->bhde', a, bS)
877 state = state * decay.unsqueeze(-1) + ab_S + \
878 torch.einsum('bhd,bhe->bhde', k, v)
879
880 o = torch.einsum('bhde,bhd->bhe', state, r).reshape(B, self.d_model)
881 o = self.ln_out(o.unsqueeze(-1)).squeeze(-1)
882 o = o * g
883 return self.W_o(o), x, state

921# ── RWKV-8 "Heron" — ROSA (Peng, 2025 — experimental) ──────────
922#
923# Replaces the linear-recurrence attention with ROSA (Rapid Online
924# Suffix Automaton): a neurosymbolic mechanism that finds the
925# longest matching suffix of quantised Q in the history of K,
926# returning the V symbol after the match.
927#
928# ⚠ EXPERIMENTAL — the suffix-matching core is inherently
929# sequential (not parallelisable on GPU). Bit-packing and
930# output scatter are vectorised with PyTorch tensor ops.
931# Streaming (step) recomputes on full context.
932#
933# Reference: github.com/BlinkDL/RWKV-LM/tree/main/RWKV-v8
934# ──────────────────────────────────────────────────────────────────
935
936def _rosa_match(q_syms, k_syms, v_syms, max_ctx=0):
937 """Suffix matching (reference impl from official RWKV-8).
938
939 For each position i, find the longest suffix of q[0:i+1] that
940 appears in k[0:i], and return v[match_end].
941
942 Args:
943 max_ctx: if >0, only look back at most max_ctx positions.
944 """
945 n = len(q_syms)
946 idx = [0] * n
947 ln = [0] * n
948 for i in range(n):
949 found = False
950 max_w = min(i + 1, max_ctx) if max_ctx > 0 else i + 1
951 start_j = max(0, i - max_ctx) if max_ctx > 0 else 0
952 for w in range(max_w, 0, -1):
953 t = q_syms[i + 1 - w: i + 1]
954 for j in range(i - w, start_j - 1, -1):
955 if k_syms[j: j + w] == t:
956 s = j + w
957 if s < n:
958 idx[i] = v_syms[s]
959 ln[i] = w
960 found = True
961 break
962 if found:
963 break
964 return idx, ln
965
966
967def _rosa_match_batch(q_packed, k_packed, v_packed, max_ctx=0):
968 """Run _rosa_match over all groups, returning (idx, ln) tensors."""
969 B, T, G = q_packed.shape
970 idx_out = torch.zeros(B, T, G, dtype=torch.long)
971 ln_out = torch.zeros(B, T, G, dtype=torch.long)
972 for b in range(B):
973 for g in range(G):
974 qs = q_packed[b, :, g].tolist()
975 ks = k_packed[b, :, g].tolist()
976 vs = v_packed[b, :, g].tolist()
977 idx, ln = _rosa_match(qs, ks, vs, max_ctx=max_ctx)
978 idx_out[b, :, g] = torch.tensor(idx)
979 ln_out[b, :, g] = torch.tensor(ln)
980 return idx_out, ln_out

983class _ROSA(nn.Module):
984 """ROSA attention with vectorised bit-packing (RWKV-8).
985
986 Args:
987 max_ctx: maximum lookback window for suffix matching.
988 0 = unlimited (slow for long sequences).
989 """
990
991 def __init__(self, d_model: int, bits: int = 4, max_ctx: int = 512):
992 super().__init__()
993 assert d_model % bits == 0
994 self.bits = bits
995 self.n_groups = d_model // bits
996 self.max_ctx = max_ctx
997 self.emb = nn.Parameter(torch.ones(1, 1, d_model))
998 # Pre-compute bit-shift powers
999 self.register_buffer('_powers', 1 << torch.arange(bits))
1000
1001 def _pack_bits(self, x):
1002 """Quantise & pack: [B, T, D] → [B, T, G] int symbols."""
1003 B, T, _ = x.shape
1004 xb = (x > 0).long() # [B, T, D]
1005 xb = xb.view(B, T, self.n_groups, self.bits) # [B, T, G, bits]
1006 return (xb * self._powers.to(x.device)).sum(-1) # [B, T, G]
1007
1008 def _unpack_to_sign(self, syms, matched):
1009 """Unpack symbols to signed embedding output.
1010
1011 Args:
1012 syms: [B, T, G] int symbols (matched v values)
1013 matched: [B, T, G] bool (whether a match was found)
1014 Returns: [B, T, D] float
1015 """
1016 B, T, G = syms.shape
1017 device = self.emb.device
1018 syms = syms.to(device)
1019 matched = matched.to(device)
1020
1021 # Unpack each symbol into bits: [B, T, G, bits]
1022 bits_expanded = ((syms.unsqueeze(-1) >> self._powers.to(device)) & 1).float()
1023 signs = bits_expanded * 2 - 1 # 0 → -1, 1 → +1
1024 # [B, T, G, bits] → [B, T, D]
1025 signs = signs.view(B, T, -1)
1026 # Apply embedding magnitude
1027 out = signs * self.emb # broadcast [1,1,D]
1028 # Zero out unmatched positions: matched [B,T,G] → [B,T,D]
1029 mask = matched.unsqueeze(-1).expand(B, T, G, self.bits).reshape(B, T, -1)
1030 return out * mask.float()
1031
1032 def forward(self, q, k, v):
1033 """q, k, v: [B, T, D] → [B, T, D]."""
1034 device = q.device
1035 # Vectorised bit-packing (on GPU if available)
1036 q_packed = self._pack_bits(q).cpu()
1037 k_packed = self._pack_bits(k).cpu()
1038 v_packed = self._pack_bits(v).cpu()
1039
1040 # Suffix matching (CPU — inherently sequential)
1041 idx, ln = _rosa_match_batch(q_packed, k_packed, v_packed,
1042 max_ctx=self.max_ctx)
1043
1044 # Vectorised output construction (back on GPU)
1045 return self._unpack_to_sign(idx, ln > 0)

1112# ── Griffin (De et al., 2024 — "Griffin / RecurrentGemma") ──────
1113#
1114# Hybrid architecture: Real-Gated Linear Recurrent Unit (RG-LRU)
1115# for long-range memory + local sliding-window causal attention
1116# for fine-grained byte patterns. Each block has three residual
1117# sub-layers:
1118# 1. RG-LRU (linear recurrence, O(1) per step)
1119# 2. Local causal attention (small window, bounded KV cache)
1120# 3. SwiGLU FFN
1121#
1122# References
1123# De et al., "Griffin: Mixing Gated Linear Recurrences with
1124# Local Attention for Efficient Language Models", 2024.
1125# ──────────────────────────────────────────────────────────────────
1126
1127class _RGLRU(nn.Module):
1128 """Real-Gated Linear Recurrent Unit — diagonal SSM with input-
1129 dependent gating.
1130
1131 Recurrence:
1132 λ_t = exp(−softplus(ν) · σ(a(x_t))) per-dim decay
1133 h_t = λ_t ⊙ h_{t-1} + (1 − λ_t) ⊙ gate(x_t) ⊙ proj(x_t)
1134 """
1135
1136 def __init__(self, d_model: int):
1137 super().__init__()
1138 self.d_model = d_model
1139 self.input_proj = nn.Linear(d_model, d_model)
1140 self.gate_proj = nn.Linear(d_model, d_model)
1141 self.recurrence_gate = nn.Linear(d_model, d_model)
1142 self.log_base_decay = nn.Parameter(torch.ones(d_model) * -3.0)
1143
1144 def _decay(self, x):
1145 a = torch.sigmoid(self.recurrence_gate(x))
1146 return torch.exp(-F.softplus(self.log_base_decay) * a)
1147
1148 def forward(self, x):
1149 """x: [B, L, D] → [B, L, D]."""
1150 B, L, D = x.shape
1151 inp = self.input_proj(x) * torch.sigmoid(self.gate_proj(x))
1152 lam = self._decay(x) # [B, L, D]
1153 h = x.new_zeros(B, D)
1154 outputs = []
1155 for t in range(L):
1156 h = lam[:, t] * h + (1 - lam[:, t]) * inp[:, t]
1157 outputs.append(h)
1158 return torch.stack(outputs, dim=1)
1159
1160 def step(self, x, h):
1161 """x, h: [B, D] → ([B, D], h_new)."""
1162 inp = self.input_proj(x) * torch.sigmoid(self.gate_proj(x))
1163 lam = self._decay(x.unsqueeze(1)).squeeze(1)
1164 h = lam * h + (1 - lam) * inp
1165 return h, h

1168class GriffinBlock(nn.Module):
1169 """Griffin-style hybrid: RG-LRU + local attention + SwiGLU FFN.
1170
1171 Parameters
1172 ----------
1173 d_model : int – hidden dimension.
1174 n_heads : int – number of attention heads for local attention.
1175 local_window : int – sliding-window width for local attention.
1176 """
1177
1178 def __init__(self, d_model: int, n_heads: int = 4, local_window: int = 128):
1179 super().__init__()
1180 assert d_model % n_heads == 0
1181 self.n_heads = n_heads
1182 self.d_head = d_model // n_heads
1183 self.d_model = d_model
1184 self.local_window = local_window
1185
1186 # Sub-layer 1: RG-LRU
1187 self.ln1 = _RMSNorm(d_model)
1188 self.rg_lru = _RGLRU(d_model)
1189
1190 # Sub-layer 2: local causal attention
1191 self.ln2 = _RMSNorm(d_model)
1192 self.q_proj = nn.Linear(d_model, d_model, bias=False)
1193 self.k_proj = nn.Linear(d_model, d_model, bias=False)
1194 self.v_proj = nn.Linear(d_model, d_model, bias=False)
1195 self.out_proj = nn.Linear(d_model, d_model, bias=False)
1196
1197 # Sub-layer 3: FFN
1198 self.ln3 = _RMSNorm(d_model)
1199 self.ff = _SwiGLU(d_model)
1200
1201 def _local_attention(self, x):
1202 B, L, D = x.shape
1203 q = self.q_proj(x).view(B, L, self.n_heads, self.d_head).transpose(1, 2)
1204 k = self.k_proj(x).view(B, L, self.n_heads, self.d_head).transpose(1, 2)
1205 v = self.v_proj(x).view(B, L, self.n_heads, self.d_head).transpose(1, 2)
1206
1207 if L <= self.local_window:
1208 y = F.scaled_dot_product_attention(q, k, v, is_causal=True)
1209 else:
1210 mask = _make_sliding_window_mask(L, self.local_window, 0, x.device)
1211 y = F.scaled_dot_product_attention(q, k, v, attn_mask=mask)
1212 return self.out_proj(y.transpose(1, 2).reshape(B, L, D))
1213
1214 def forward(self, x, inference_params=None):
1215 # Recurrence
1216 y = self.ln1(x)
1217 y = self.rg_lru(y)
1218 x = x + y
1219 # Local attention
1220 y = self.ln2(x)
1221 y = self._local_attention(y)
1222 x = x + y
1223 # FFN
1224 y = self.ln3(x)
1225 y = self.ff(y)
1226 return x + y
1227
1228 def init_cache(self, batch_size: int, device):
1229 h_rec = torch.zeros(batch_size, self.d_model, device=device)
1230 k_cache = torch.zeros(batch_size, self.n_heads, 0, self.d_head,
1231 device=device)
1232 v_cache = torch.zeros(batch_size, self.n_heads, 0, self.d_head,
1233 device=device)
1234 return (h_rec, k_cache, v_cache)
1235
1236 def step(self, x, cache):
1237 h_rec, k_cache, v_cache = cache
1238 B = x.shape[0]
1239
1240 # Recurrence step
1241 y = self.ln1(x)
1242 y, h_rec = self.rg_lru.step(y, h_rec)
1243 x = x + y
1244
1245 # Local attention step (rolling KV cache)
1246 y = self.ln2(x).unsqueeze(1) # [B,1,D]
1247 q = self.q_proj(y).view(B, 1, self.n_heads, self.d_head).transpose(1, 2)
1248 k_new = self.k_proj(y).view(B, 1, self.n_heads, self.d_head).transpose(1, 2)
1249 v_new = self.v_proj(y).view(B, 1, self.n_heads, self.d_head).transpose(1, 2)
1250
1251 k_cache = torch.cat([k_cache, k_new], dim=2)
1252 v_cache = torch.cat([v_cache, v_new], dim=2)
1253 if k_cache.shape[2] > self.local_window:
1254 k_cache = k_cache[:, :, -self.local_window:]
1255 v_cache = v_cache[:, :, -self.local_window:]
1256
1257 scale = self.d_head ** -0.5
1258 attn = (q @ k_cache.transpose(-2, -1)) * scale
1259 attn = F.softmax(attn, dim=-1)
1260 y = (attn @ v_cache).transpose(1, 2).reshape(B, 1, self.d_model)
1261 y = self.out_proj(y).squeeze(1)
1262 x = x + y
1263
1264 # FFN
1265 y = self.ln3(x)
1266 y = self.ff(y)
1267 return x + y, (h_rec, k_cache, v_cache)

1270# ── xLSTM (Beck et al., 2024 — "xLSTM: Extended LSTM") ─────────
1271#
1272# mLSTM variant with:
1273# • Exponential gating (input gate can exceed 1 for amplification)
1274# • Matrix-valued cell state C ∈ R^{d_head × d_head}
1275# • Covariance normaliser for numerical stability
1276# • Log-space stabilisation of gates (max-trick)
1277#
1278# Recurrence (per head, in log-space-stabilised form):
1279# m_t = max(log_f_t + m_{t-1}, log_i_t)
1280# f' = exp(log_f + m_{t-1} − m_t)
1281# i' = exp(log_i − m_t)
1282# C_t = f' · C_{t-1} + i' · (v_t ⊗ k_t)
1283# n_t = f' · n_{t-1} + i' · k_t
1284# h_t = C_t @ q_t / max(|n_t · q_t|, 1)
1285#
1286# References
1287# Beck et al., "xLSTM: Extended Long Short-Term Memory", 2024.
1288# ──────────────────────────────────────────────────────────────────
1289
1290class _mLSTMCell(nn.Module):
1291 """mLSTM cell — matrix memory with exponential gating."""
1292
1293 def __init__(self, d_model: int, n_heads: int = 4):
1294 super().__init__()
1295 assert d_model % n_heads == 0
1296 self.n_heads = n_heads
1297 self.d_head = d_model // n_heads
1298 self.d_model = d_model
1299
1300 self.W_q = nn.Linear(d_model, d_model, bias=False)
1301 self.W_k = nn.Linear(d_model, d_model, bias=False)
1302 self.W_v = nn.Linear(d_model, d_model, bias=False)
1303
1304 # Scalar gates per head
1305 self.W_i = nn.Linear(d_model, n_heads) # input gate (exp)
1306 self.W_f = nn.Linear(d_model, n_heads) # forget gate (sigmoid)
1307
1308 self.out_proj = nn.Linear(d_model, d_model, bias=False)
1309 self.group_norm = nn.GroupNorm(n_heads, d_model)
1310
1311 def forward(self, x, state=None):
1312 """x: [B, L, D] → ([B, L, D], new_state).
1313
1314 state = (C, n, m) with shapes
1315 C: [B, H, dh, dh] matrix cell
1316 n: [B, H, dh] normaliser
1317 m: [B, H] log-space stabiliser
1318 """
1319 B, L, D = x.shape
1320 H, dh = self.n_heads, self.d_head
1321
1322 q = self.W_q(x).view(B, L, H, dh)
1323 k = self.W_k(x).view(B, L, H, dh) * (dh ** -0.5)
1324 v = self.W_v(x).view(B, L, H, dh)
1325
1326 log_f = -F.softplus(-self.W_f(x)) # log(σ(·)), always ≤ 0
1327 log_i = self.W_i(x) # can be > 0
1328
1329 if state is None:
1330 C = x.new_zeros(B, H, dh, dh)
1331 n = x.new_zeros(B, H, dh)
1332 m = x.new_zeros(B, H)
1333 else:
1334 C, n, m = state
1335
1336 outputs = []
1337 for t in range(L):
1338 lf = log_f[:, t] # [B, H]
1339 li = log_i[:, t]
1340 m_new = torch.max(lf + m, li)
1341 f_prime = torch.exp(lf + m - m_new) # [B, H]
1342 i_prime = torch.exp(li - m_new)
1343 m = m_new
1344
1345 k_t, v_t, q_t = k[:, t], v[:, t], q[:, t]
1346
1347 C = f_prime.unsqueeze(-1).unsqueeze(-1) * C + \
1348 i_prime.unsqueeze(-1).unsqueeze(-1) * \
1349 torch.einsum('bhd,bhe->bhde', k_t, v_t)
1350 n = f_prime.unsqueeze(-1) * n + i_prime.unsqueeze(-1) * k_t
1351
1352 h_t = torch.einsum('bhde,bhd->bhe', C, q_t)
1353 denom = torch.einsum('bhd,bhd->bh', n, q_t) \
1354 .unsqueeze(-1).abs().clamp(min=1.0)
1355 outputs.append(h_t / denom)
1356
1357 out = torch.stack(outputs, dim=1).reshape(B, L, D)
1358 out = self.group_norm(out.transpose(1, 2)).transpose(1, 2)
1359 return self.out_proj(out), (C, n, m)
1360
1361 def step(self, x, state):
1362 """Single-token step. x: [B, D], state: (C, n, m)."""
1363 B = x.shape[0]
1364 H, dh = self.n_heads, self.d_head
1365 C, n, m = state
1366
1367 q = self.W_q(x).view(B, H, dh)
1368 k = self.W_k(x).view(B, H, dh) * (dh ** -0.5)
1369 v = self.W_v(x).view(B, H, dh)
1370
1371 lf = -F.softplus(-self.W_f(x)).squeeze(-1) if self.n_heads == 1 \
1372 else -F.softplus(-self.W_f(x)) # [B, H]
1373 li = self.W_i(x) if self.n_heads > 1 else self.W_i(x)
1374
1375 m_new = torch.max(lf + m, li)
1376 f_prime = torch.exp(lf + m - m_new)
1377 i_prime = torch.exp(li - m_new)
1378 m = m_new
1379
1380 C = f_prime.unsqueeze(-1).unsqueeze(-1) * C + \
1381 i_prime.unsqueeze(-1).unsqueeze(-1) * \
1382 torch.einsum('bhd,bhe->bhde', k, v)
1383 n = f_prime.unsqueeze(-1) * n + i_prime.unsqueeze(-1) * k
1384
1385 h = torch.einsum('bhde,bhd->bhe', C, q)
1386 denom = torch.einsum('bhd,bhd->bh', n, q) \
1387 .unsqueeze(-1).abs().clamp(min=1.0)
1388 h = (h / denom).reshape(B, self.d_model)
1389 h = self.group_norm(h.unsqueeze(-1)).squeeze(-1)
1390 return self.out_proj(h), (C, n, m)

57 __device__ void build_cdf_fast(const float* probs_row, int K, uint32_t* cdf) {
58 const uint32_t TOTAL = 1u << PRECISION;
59 if (K <= 0) { cdf[0] = 0; cdf[1] = TOTAL; return; }
60 const uint32_t free_weight = TOTAL - (uint32_t)K;
61 double norm = 0.0;
62 for (int i = 0; i < K; ++i) norm += (double)probs_row[i];
63 if (!(norm > 0.0) || !isfinite(norm)) {
64 cdf[0] = 0;
65 uint32_t acc = 0;
66 for (int i=0;i<K;i++) { cdf[i] = acc; acc += (free_weight / (uint32_t)K) + 1u; }
67 cdf[K] = TOTAL; return;
68 }
69 double scale = (double)free_weight / norm;
70 double cumulative_float = 0.0;
71 uint32_t accumulated_slack = 0;
72 for (int i=0;i<K;i++) {
73 uint32_t left = (uint32_t)(cumulative_float * scale) + accumulated_slack;
74 cdf[i] = left;
75 cumulative_float += (double)probs_row[i];
76 accumulated_slack += 1u;
77 }
78 cdf[K] = TOTAL;
79 }

88 __global__ void encode_kernel(const int32_t* d_symbols, const float* d_probs, int N, int K,
89 uint32_t* d_out_words, EncState* states, int pitch_words,
90 const uint8_t* d_mask) {
91 int idx = blockIdx.x * blockDim.x + threadIdx.x;
92 if (idx >= N) return;
93 if (d_mask && d_mask[idx] == 0) return;
94 EncState &st = states[idx];
95 const int32_t s = d_symbols[idx];
96 const float* probs_row = d_probs + (size_t)idx * (size_t)K;
97 if (s < 0 || s >= K) return;
98 uint32_t cdf[1025];
99 build_cdf_fast(probs_row, K, cdf);
100 uint32_t left = cdf[s];
101 uint32_t prob = cdf[s+1] - cdf[s];
102 unsigned long long scale = st.range >> PRECISION;
103 st.range = scale * (unsigned long long)prob;
104 unsigned long long old_lower = st.lower;
105 st.lower = st.lower + scale * (unsigned long long)left;
106 // Handle transition out of inverted
107 if (st.inverted_num > 0) {
108 unsigned long long sum = st.lower + st.range;
109 if (sum > st.lower) {
110 uint32_t first_word, subsequent;
111 if (st.lower < old_lower) { first_word = st.first_inv_lower_word + 1u; subsequent = 0u; }
112 else { first_word = st.first_inv_lower_word; subsequent = 0xFFFFFFFFu; }
113 int widx = st.write_idx_words;
114 uint32_t* out = d_out_words + (size_t)idx * (size_t)pitch_words;
115 out[(size_t)widx++] = first_word;
116 for (int i=1;i<st.inverted_num;i++) out[(size_t)widx++] = subsequent;
117 st.write_idx_words = widx;
118 st.inverted_num = 0;
119 }
120 }
121 // Renormalize if needed (emit possibly multiple words)
122 while (st.range < (1ull << (64-32))) {
123 uint32_t lower_word = (uint32_t)(st.lower >> (64-32));
124 st.lower <<= 32;
125 st.range <<= 32;
126 if (st.inverted_num > 0) {
127 if (st.inverted_num < 0x7FFFFFFF) st.inverted_num += 1;
128 } else {
129 unsigned long long sum = st.lower + st.range;
130 if (sum > st.lower) {
131 int widx = st.write_idx_words;
132 uint32_t* out = d_out_words + (size_t)idx * (size_t)pitch_words;
133 out[(size_t)widx++] = lower_word;
134 st.write_idx_words = widx;
135 } else {
136 st.inverted_num = 1;
137 st.first_inv_lower_word = lower_word;
138 }
139 }
140 }
141 }

143 __global__ void finalize_kernel(int N, uint32_t* d_out_words, EncState* states, int pitch_words, int* sizes_words) {
144 int idx = blockIdx.x * blockDim.x + threadIdx.x;
145 if (idx >= N) return;
146 EncState &st = states[idx];
147 uint32_t* out = d_out_words + (size_t)idx * (size_t)pitch_words;
148 if (st.range == 0xFFFFFFFFFFFFFFFFull) { sizes_words[idx] = 0; return; }
149 unsigned long long point = st.lower + ((1ull << (64-32)) - 1ull);
150 if (st.inverted_num > 0) {
151 uint32_t first_word, subsequent;
152 if (point < st.lower) { first_word = st.first_inv_lower_word + 1u; subsequent = 0u; }
153 else { first_word = st.first_inv_lower_word; subsequent = 0xFFFFFFFFu; }
154 int widx = st.write_idx_words;
155 out[(size_t)widx++] = first_word;
156 for (int i=1;i<st.inverted_num;i++) out[(size_t)widx++] = subsequent;
157 st.write_idx_words = widx;
158 st.inverted_num = 0;
159 }
160 uint32_t point_word = (uint32_t)(point >> (64-32));
161 int widx = st.write_idx_words;
162 out[(size_t)widx++] = point_word;
163 unsigned long long upper = st.lower + st.range;
164 uint32_t upper_word = (uint32_t)(upper >> (64-32));
165 if (upper_word == point_word) out[(size_t)widx++] = 0u;
166 st.write_idx_words = widx;
167 sizes_words[idx] = widx;
168 }

170 __global__ void init_dec_kernel(int N, const uint32_t* d_in_words, int* sizes_words, DecState* dst, int pitch_words) {
171 int idx = blockIdx.x * blockDim.x + threadIdx.x;
172 if (idx >= N) return;
173 const uint32_t* in = d_in_words + (size_t)idx * (size_t)pitch_words;
174 DecState &st = dst[idx];
175 st.lower = 0ull; st.range = 0xFFFFFFFFFFFFFFFFull; st.point = 0ull; st.read_idx_words = 0;
176 for (int i=0;i<2;i++) {
177 uint32_t w = (st.read_idx_words < sizes_words[idx]) ? in[st.read_idx_words++] : 0u;
178 st.point = (st.point << 32) | (unsigned long long)w;
179 }

182 __global__ void decode_step_kernel(int N, int K, const float* d_probs, const uint32_t* d_in_words, int* sizes_words,
183 DecState* st_arr, int pitch_words, int32_t* out_symbols, const uint8_t* d_mask) {
184 int idx = blockIdx.x * blockDim.x + threadIdx.x;
185 if (idx >= N) return;
186 if (d_mask && d_mask[idx] == 0) return;
187 DecState &st = st_arr[idx];
188 const float* probs_row = d_probs + (size_t)idx * (size_t)K;
189 const uint32_t* in = d_in_words + (size_t)idx * (size_t)pitch_words;
190 if (K <= 0) { out_symbols[idx] = 0; return; }
191 uint32_t cdf[1025];
192 build_cdf_fast(probs_row, K, cdf);
193 unsigned long long scale = st.range >> PRECISION;
194 unsigned long long q = (st.point - st.lower) / scale;
195 if (q >= (1ull<<PRECISION)) q = (1ull<<PRECISION)-1ull;
196 uint32_t target = (uint32_t)q;
197 int next_symbol = 1;
198 while (next_symbol <= K && !(cdf[next_symbol] > target)) ++next_symbol;
199 int s = next_symbol - 1; if (s < 0) s = 0; if (s >= K) s = K-1;
200 out_symbols[idx] = (int32_t)s;
201 uint32_t left = cdf[s]; uint32_t prob = cdf[s+1] - cdf[s];
202 st.lower = st.lower + scale * (unsigned long long)left;
203 st.range = scale * (unsigned long long)prob;
204 while (st.range < (1ull << (64-32))) {
205 st.lower <<= 32;
206 st.range <<= 32;
207 uint32_t w = (st.read_idx_words < sizes_words[idx]) ? in[st.read_idx_words++] : 0u;
208 st.point = (st.point << 32) | (unsigned long long)w;
209 }
210 }

237 void load_compressed_from_host(py::list compressed_list) {
238 if ((int)compressed_list.size() != N) throw std::runtime_error("compressed_list length must equal N");
239 std::vector<int> sizes_host(N, 0);
240 std::vector<uint32_t> buf((size_t)N * (size_t)pitch, 0u);
241 for (int i=0;i<N;i++) {
242 py::array arr = py::array::ensure(compressed_list[i]);
243 if (!arr || arr.ndim()!=1 || arr.itemsize()!=4 || arr.dtype().kind()!='u') throw std::runtime_error("Each compressed item must be uint32[?]");
244 size_t nwords = (size_t)arr.shape(0);
245 if (nwords > (size_t)pitch) throw std::runtime_error("Compressed stream exceeds pitch; increase pitch");
246 const uint32_t* src = static_cast<const uint32_t*>(arr.data());
247 uint32_t* dst = buf.data() + (size_t)i * (size_t)pitch;
248 std::memcpy(dst, src, nwords * sizeof(uint32_t));
249 sizes_host[i] = (int)nwords;
250 }
251 cudaMemcpy(d_words, buf.data(), buf.size()*sizeof(uint32_t), cudaMemcpyHostToDevice);

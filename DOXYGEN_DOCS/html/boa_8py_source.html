36 def _uvarint_encode(x: int) -> bytes:
37 out = bytearray()
38 while True:
39 b = x & 0x7F; x >>= 7
40 out.append(b | (0x80 if x else 0))
41 if not x: break
42 return bytes(out)

44 def _uvarint_decode(buf: memoryview, pos: int):
45 x = 0; s = 0
46 while True:
47 b = buf[pos]; pos += 1
48 x |= (b & 0x7F) << s
49 if not (b & 0x80): return x, pos
50 s += 7

52 def _as_bytes(obj) -> bytes:
53 if isinstance(obj, (bytes, bytearray)): return bytes(obj)
54 if torch.is_tensor(obj):
55 t = obj.detach().contiguous().to("cpu")
56 if t.dtype != torch.uint8: t = t.view(torch.uint8)
57 return t.numpy().tobytes()
58 arr = np.asarray(obj)
59 if arr.dtype != np.uint8: arr = arr.view(np.uint8)

62 def _pad4(b: bytes) -> bytes:
63 r = len(b) & 3
64 return b if r == 0 else (b + b"\x00" * (4 - r))

79 def _split_to_chunks(self, data_bytes: bytes, seq_size: int = 0, chunks_count: int = 0):
80 n = len(data_bytes)
81 if seq_size and chunks_count:
82 chunk_len = int(seq_size); n_chunks = math.ceil(n / chunk_len)
83 elif seq_size:
84 chunk_len = int(seq_size); n_chunks = math.ceil(n / chunk_len)
85 elif chunks_count:
86 n_chunks = max(int(chunks_count), 1); chunk_len = math.ceil(n / n_chunks)
87 else:
88 raise ValueError("Provide either 'seq_size' or 'chunks_count'.")
89 chunks = []
90 for i in range(n_chunks):
91 s = i * chunk_len; e = min(s + chunk_len, n)
92 if s >= e: break
93 arr = np.frombuffer(memoryview(data_bytes)[s:e], dtype=np.uint8).astype(np.int64)
94 chunks.append(arr)
95 last_len = len(chunks[-1]) if chunks else 0
96 self.metadata = {
97 'chunk_len': int(chunk_len),
98 'n_chunks': int(len(chunks)),
99 'uncompressed_len': int(n),
100 'last_chunk_len': int(last_len if last_len else chunk_len),
101 }
102 return chunks, int(chunk_len)

133 def _read_file(self):
134 data = Path(self.filepath).read_bytes()
135 mm = memoryview(data); p = 0
136 if bytes(mm[p:p+4]) != self.MAGIC: raise ValueError("Bad file magic")
137 p += 4
138 version, = struct.unpack_from('<I', mm, p); p += 4
139 if version != self.VERSION: raise ValueError(f"Unsupported version {version}")
140 _flags, = struct.unpack_from('<I', mm, p); p += 4
141 ulen, = struct.unpack_from('<Q', mm, p); p += 8
142 chunk_len, = struct.unpack_from('<I', mm, p); p += 4
143 n, = struct.unpack_from('<I', mm, p); p += 4
144 last_chunk_len, = struct.unpack_from('<I', mm, p); p += 4
145 hlen = mm[p]; p += 1
146 _fp = bytes(mm[p:p+hlen]); p += hlen
147
148 crc = struct.unpack_from('<I', mm, len(mm)-4)[0]
149 # Support both IDX1 and IDX2
150 idx_pos_2 = data.rfind(b'IDX2')
151 idx_pos_1 = data.rfind(b'IDX1')
152 idx_pos = max(idx_pos_2, idx_pos_1)
153 if idx_pos < 0: raise ValueError("Index not found")
154 idx_ver = 2 if idx_pos == idx_pos_2 else 1
155 if (zlib.crc32(data[idx_pos:len(data)-4]) & 0xFFFFFFFF) != crc:
156 raise ValueError("Bad index CRC")
157
158 q = idx_pos + 4 # skip IDXn tag
159 first_bytes = list(mm[q:q+n]); q += n
160
161 if idx_ver == 1:
162 offsets = [0]*n; pos = q; prev = 0
163 for i in range(n):
164 d, pos = _uvarint_decode(mm, pos); prev += d; offsets[i] = prev
165 comp_lens = [0]*n
166 for i in range(n):
167 L, pos = _uvarint_decode(mm, pos); comp_lens[i] = L
168 else:
169 # IDX2: only lengths, reconstruct offsets
170 comp_lens = [0]*n; pos = q
171 for i in range(n):
172 L, pos = _uvarint_decode(mm, pos); comp_lens[i] = L
173 offsets = [0]*n; off = 0
174 for i in range(n):
175 offsets[i] = off; off += comp_lens[i]
176
177 payload = mm[p:idx_pos]
178 compressed_list = [bytes(payload[offsets[i]: offsets[i]+comp_lens[i]]) for i in range(n)]
179 full_lens = [int(chunk_len)]*(n-1) + [int(last_chunk_len)]
180
181 self.compressed_data = compressed_list # bytes
182 self.first_bytes = first_bytes # list[int]
183 self.lengths = full_lens # uncompressed lens
184 self.metadata = {
185 'chunk_len': int(chunk_len),
186 'n_chunks': int(n),
187 'last_chunk_len': int(last_chunk_len),
188 'uncompressed_len': int(ulen),
189 }

306 idx += bytes(first_bytes) # n bytes
307 for L in lengths:
308 idx += _uvarint_encode(L)

1"""
2HydraBOA — Multi-head chunked Mamba model for fast byte-level compression.
3
4Predicts K bytes per backbone step (default K=4), cutting Mamba compute by K×.
5
6Architecture
7────────────
81. **Chunked Backbone**: groups K bytes into a single embedding vector and
9 passes the down-sampled sequence through Mamba. Sequence length ≡ L/K.
102. **Chained Heads**: K ultra-lightweight prediction heads. Head *k*
11 receives the backbone context H_t concatenated with embeddings of the
12 *k* preceding bytes within the current chunk (teacher-forced at train
13 time).
14
15This file is completely self-contained and does NOT modify any existing
16boa-constrictor source files.
17"""

47class _RMSNorm(nn.Module):
48 """Root-Mean-Square LayerNorm (Zhang & Sennrich 2019)."""
49
50 def __init__(self, d_model: int, eps: float = 1e-6):
51 super().__init__()
52 self.weight = nn.Parameter(torch.ones(d_model))
53 self.eps = eps
54
55 def forward(self, x):
56 rms = x.float().pow(2).mean(-1, keepdim=True).add(self.eps).rsqrt()
57 return (x.float() * rms).to(x.dtype) * self.weight

60class _SwiGLU(nn.Module):
61 """SwiGLU feed-forward (Shazeer 2020). ≈ 2.67× d_model hidden."""
62
63 def __init__(self, d_model: int):
64 super().__init__()
65 hidden = ((8 * d_model // 3 + 7) // 8) * 8
66 self.w_gate = nn.Linear(d_model, hidden, bias=False)
67 self.w_up = nn.Linear(d_model, hidden, bias=False)
68 self.w_down = nn.Linear(hidden, d_model, bias=False)
69
70 def forward(self, x):
71 return self.w_down(F.silu(self.w_gate(x)) * self.w_up(x))

76class _HydraMambaBlock(nn.Module):
77 """Original Mamba-v1 block: LN → Mamba → LN → FFN(4×, GELU) + residual.
78
79 Matches the ``MambaBlockV1`` in model.py (backbone="mambav1").
80 """
81
82 def __init__(self, d_model: int):
83 super().__init__()
84 self.ln1 = nn.LayerNorm(d_model)
85 if IS_CUDA:
86 self.mamba = _CudaMamba(d_model=d_model)
87 else:
88 cfg = _CpuMambaConfig(d_model=d_model, n_layers=0, use_cuda=False)
89 self.mamba = _CpuMamba(cfg)
90 self.ln2 = nn.LayerNorm(d_model)
91 self.ff = nn.Sequential(
92 nn.Linear(d_model, 4 * d_model),
93 nn.GELU(),
94 nn.Linear(4 * d_model, d_model),
95 )
96
97 def forward(self, x, inference_params=None):
98 y = self.ln1(x)
99 if IS_CUDA:
100 y = self.mamba(y, inference_params=inference_params)
101 else:
102 y = self.mamba(y)
103 y = self.ln2(y)
104 y = self.ff(y)
105 return x + y
106
107 # CPU-only streaming helpers (mambapy path)
108 if not IS_CUDA and _HAS_CPU_MAMBA:
109 def init_cache(self, batch_size: int, device):
110 d_inner = self.mamba.config.d_inner
111 d_conv = self.mamba.config.d_conv
112 inputs = torch.zeros(batch_size, d_inner, d_conv - 1, device=device)
113 return (None, inputs)
114
115 def step(self, x, cache):
116 y = self.ln1(x)
117 y, cache = self.mamba.step(y, cache)
118 y = self.ln2(y)
119 y = self.ff(y)
120 return x + y, cache

152class HydraBOA(nn.Module):
153 r"""Multi-head chunked Mamba byte predictor.
154
155 Parameters
156 ----------
157 d_model : int
158 Hidden dimension (default 256).
159 num_layers : int
160 Number of Mamba blocks in the backbone (default 1).
161 vocab_size : int
162 Byte alphabet size (default 256).
163 K : int
164 Chunk size — number of bytes predicted per backbone step (default 4).
165
166 Training
167 --------
168 ``forward(x)`` accepts ``x`` of shape ``[B, L]`` with ``L`` divisible
169 by *K* and returns logits ``[B, L, vocab_size]``. The logit at
170 position *i* predicts the byte at position *i* (not *i+1*).
171 Causality is guaranteed by the backbone shift (BOS token) and the
172 chained-head structure.
173
174 Loss::
175
176 loss = F.cross_entropy(logits.view(-1, V), x.view(-1))
177
178 Streaming
179 ---------
180 For compression / decompression::
181
182 cache = model.init_stream(max_chunks=T, batch_size=B, device=dev)
183
184 # For each K-byte chunk:
185 H_t = model.step_backbone(prev_chunk_bytes, cache)
186 for k in range(K):
187 logits_k = model.predict_head(k, H_t, prev_bytes_in_chunk)
188 probs_k = softmax(logits_k)
189 # feed probs_k to the range coder
190 """
191
192 def __init__(self, d_model: int = 256, num_layers: int = 1,
193 vocab_size: int = 256, K: int = 4):
194 super().__init__()
195 self.K = K
196 self.d_model = d_model
197 self.vocab_size = vocab_size
198 self._backbone_name = "hydra_mambav1"
199
200 # ── shared byte embedding ──
201 self.byte_embed = nn.Embedding(vocab_size, d_model)
202
203 # ── chunk embedding: concat K byte-embeddings → d_model ──
204 self.chunk_proj = nn.Linear(K * d_model, d_model)
205
206 # ── learned BOS embedding (bootstraps the first chunk) ──
207 self.bos_embed = nn.Parameter(torch.randn(d_model) * 0.02)
208
209 # ── Mamba backbone ──
210 self.blocks = nn.ModuleList(
211 [_HydraMambaBlock(d_model) for _ in range(num_layers)]
212 )
213 self.backbone_norm = nn.LayerNorm(d_model)
214
215 # ── K chained prediction heads ──
216 # Head k: (1 + k) · d_model → d_model → vocab_size
217 self.heads = nn.ModuleList([
218 nn.Sequential(
219 nn.Linear((1 + k) * d_model, d_model),
220 nn.ReLU(),
221 nn.Linear(d_model, vocab_size),
222 )
223 for k in range(K)
224 ])
225
226 _tag_mamba_layers(self)

230 def _make_chunk_embeds(self, chunks: torch.LongTensor) -> torch.Tensor:
231 """chunks: [B, T, K] → [B, T, d_model]"""
232 emb = self.byte_embed(chunks) # [B, T, K, D]
233 B, T, K, D = emb.shape
234 return self.chunk_proj(emb.reshape(B, T, K * D)) # [B, T, D]
235
236 # ── training forward ─────────────────────────────────────────
237
238 def forward(self, x: torch.LongTensor) -> torch.Tensor:
239 """
240 x : [B, L] long tensor of byte values (L divisible by K)
241 Returns logits [B, L, vocab_size].
242 """
243 B, L = x.shape
244 K, D = self.K, self.d_model
245 assert L % K == 0, f"L={L} must be divisible by K={K}"
246 T = L // K
247
248 chunks = x.reshape(B, T, K) # [B, T, K]
249
250 # ── backbone ──
251 chunk_embs = self._make_chunk_embeds(chunks) # [B, T, D]
252 bos = self.bos_embed.unsqueeze(0).unsqueeze(0).expand(B, 1, D)
253 # Shifted: [BOS, c_0, c_1, …, c_{T-2}] so that output t only
254 # sees chunks < t.
255 backbone_in = torch.cat([bos, chunk_embs[:, :-1, :]], dim=1) # [B, T, D]
256
257 h = backbone_in
258 for blk in self.blocks:
259 h = blk(h) # full parallel scan
260 H = self.backbone_norm(h) # [B, T, D]
261
262 # ── chained heads (teacher-forced) ──
263 byte_embs = self.byte_embed(chunks) # [B, T, K, D]
264
265 logits_list = []
266 for k in range(K):
267 if k == 0:
268 inp = H # [B, T, D]
269 else:
270 prev = byte_embs[:, :, :k, :].reshape(B, T, k * D)
271 inp = torch.cat([H, prev], dim=-1) # [B, T, (1+k)·D]
272 logits_list.append(self.heads[k](inp)) # [B, T, V]
273
274 # Interleave heads → [B, T, K, V] → [B, L, V]
275 logits = torch.stack(logits_list, dim=2).reshape(B, L, self.vocab_size)
276 return logits

291 @torch.inference_mode()
292 def step_backbone(self, prev_chunk_bytes, cache,
293 batch_size: int = 1) -> torch.Tensor:
294 """Feed one chunk embedding through the backbone → H_t.
295
296 Parameters
297 ----------
298 prev_chunk_bytes : [B, K] long tensor (previous chunk)
299 *None* for the very first chunk (BOS used).
300 cache : streaming state from ``init_stream``.
301 batch_size : used only when prev_chunk_bytes is None.
302
303 Returns
304 -------
305 H_t : Tensor [B, d_model]
306 """
307 dev = next(self.parameters()).device
308
309 if prev_chunk_bytes is None:
310 x = self.bos_embed.unsqueeze(0).expand(batch_size, -1) # [B, D]
311 else:
312 emb = self.byte_embed(prev_chunk_bytes) # [B, K, D]
313 B, K, D = emb.shape
314 x = self.chunk_proj(emb.reshape(B, K * D)) # [B, D]
315
316 if IS_CUDA:
317 h = x.unsqueeze(1) # [B, 1, D]
318 for blk in self.blocks:
319 h = blk(h, inference_params=cache)
320 H = self.backbone_norm(h.squeeze(1)) # [B, D]
321 _bump_offset(cache, 1)
322 else:
323 h = x # [B, D]
324 for i, blk in enumerate(self.blocks):
325 h, cache[i] = blk.step(h, cache[i])
326 H = self.backbone_norm(h) # [B, D]
327
328 return H

330 @torch.inference_mode()
331 def predict_head(self, k: int, H_t: torch.Tensor,
332 prev_bytes: torch.LongTensor = None) -> torch.Tensor:
333 """Run head *k* and return logits [B, vocab_size].
334
335 Parameters
336 ----------
337 k : int (0 … K-1)
338 H_t : [B, d_model]
339 prev_bytes : [B, k] long tensor of bytes already decoded in this
340 chunk. *None* (or empty) for k=0.
341 """
342 if k == 0:
343 inp = H_t
344 else:
345 assert prev_bytes is not None and prev_bytes.shape[-1] == k
346 emb = self.byte_embed(prev_bytes) # [B, k, D]
347 B = H_t.shape[0]
348 prev_flat = emb.reshape(B, k * self.d_model)
349 inp = torch.cat([H_t, prev_flat], dim=-1) # [B, (1+k)·D]
350 return self.heads[k](inp) # [B, V]

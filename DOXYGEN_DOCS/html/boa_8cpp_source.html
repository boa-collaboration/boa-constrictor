53std::vector<float> softmax(const float* logits, int size) {
54 std::vector<float> probs(size);
55 float max_val = logits[0];
56 for(int i=1; i<size; ++i) if(logits[i] > max_val) max_val = logits[i];
57
58 double sum_exp = 0.0;
59 for(int i=0; i<size; ++i) {
60 double val = (double)repro_exp(logits[i] - max_val);
61 probs[i] = (float)val;
62 sum_exp += val;
63 }
64 double inv_sum = 1.0 / sum_exp;
65 for(int i=0; i<size; ++i) probs[i] *= (float)inv_sum;
66 return probs;
67}

85static uint32_t crc32_table[256];
86static bool crc32_table_init = false;
87
88static void init_crc32_table() {
89 if (crc32_table_init) return;
90 for (uint32_t i = 0; i < 256; ++i) {
91 uint32_t c = i;
92 for (int k = 0; k < 8; ++k) {
93 c = (c & 1) ? (0xEDB88320u ^ (c >> 1)) : (c >> 1);
94 }
95 crc32_table[i] = c;
96 }
97 crc32_table_init = true;
98}
99
100static uint32_t crc32_compute(const uint8_t* data, size_t len) {
101 init_crc32_table();
102 uint32_t c = 0xFFFFFFFFu;
103 for (size_t i = 0; i < len; ++i) {
104 c = crc32_table[(c ^ data[i]) & 0xFFu] ^ (c >> 8);
105 }
106 return c ^ 0xFFFFFFFFu;
107}

109void auto_size_params(const MambaConfig& config, int& batch_size, size_t& chunk_size, bool is_compression) {
110 size_t free_vram, total_vram;
111 get_gpu_vram_info(free_vram, total_vram);
112
113 // Safety buffer: use 90% of free VRAM
114 size_t target_vram = (size_t)(free_vram * 0.9);
115
116 if (chunk_size == 0) chunk_size = 4096;
117
118 // Initial guess for batch size if not provided
119 if (batch_size <= 0) {
120 if (is_compression) batch_size = 64; // Conservative start
121 else batch_size = 1024; // Decompression is lighter
122 }
123
124 // Adjust batch size based on memory
125 size_t mem = BoaPredictorGPU::estimate_memory_static(config, 256, config.n_layers, batch_size, (int)chunk_size, is_compression);
126
127 if (mem > target_vram) {
128 // Reduce batch size
129 while (batch_size > 1 && mem > target_vram) {
130 batch_size /= 2;
131 mem = BoaPredictorGPU::estimate_memory_static(config, 256, config.n_layers, batch_size, (int)chunk_size, is_compression);
132 }
133 // If even batch=1 doesn't fit, reduce chunk_size
134 while (chunk_size > 256 && mem > target_vram) {
135 chunk_size /= 2;
136 mem = BoaPredictorGPU::estimate_memory_static(config, 256, config.n_layers, batch_size, (int)chunk_size, is_compression);
137 }
138 } else {
139 // Increase batch size to fill VRAM
140 int new_batch = batch_size;
141 while (true) {
142 int test_batch = new_batch * 2;
143 size_t test_mem = BoaPredictorGPU::estimate_memory_static(config, 256, config.n_layers, test_batch, (int)chunk_size, is_compression);
144 if (test_mem < target_vram && test_batch <= 8192) {
145 new_batch = test_batch;
146 } else {
147 break;
148 }
149 }
150 batch_size = new_batch;
151 }
152
153 mem = BoaPredictorGPU::estimate_memory_static(config, 256, config.n_layers, batch_size, (int)chunk_size, is_compression);
154 std::cout << "Auto-sizing: Selected Batch=" << batch_size << ", Chunk=" << chunk_size
155 << " (" << mem / (1024*1024) << " MB VRAM est, " << free_vram / (1024*1024) << " MB free)\n";
156}

158static void uvarint_encode(std::vector<uint8_t>& out, uint64_t x) {
159 while (true) {
160 uint8_t b = (uint8_t)(x & 0x7Fu);
161 x >>= 7;
162 out.push_back((uint8_t)(b | (x ? 0x80u : 0u)));
163 if (!x) break;
164 }
165}
166
167static uint64_t uvarint_decode(const std::vector<uint8_t>& buf, size_t& pos) {
168 uint64_t x = 0; int shift = 0;
169 while (true) {
170 uint8_t b = buf[pos++];
171 x |= (uint64_t)(b & 0x7Fu) << shift;
172 if (!(b & 0x80u)) break;
173 shift += 7;
174 }
175 return x;
176}

190static Boa2Container read_boa2(const std::string& path) {
191 Boa2Container out;
192 std::ifstream fin(path, std::ios::binary);
193 std::vector<uint8_t> data((std::istreambuf_iterator<char>(fin)), std::istreambuf_iterator<char>());
194 fin.close();
195 if (data.size() < 32) throw std::runtime_error("Invalid BOA2 file");
196
197 size_t p = 0;
198 if (std::memcmp(data.data(), "BOA2", 4) != 0) throw std::runtime_error("Bad BOA2 magic");
199 p += 4;
200 uint32_t version = 0; std::memcpy(&version, data.data() + p, 4); p += 4;
201 if (version != 1) throw std::runtime_error("Unsupported BOA2 version");
202 uint32_t flags = 0; std::memcpy(&flags, data.data() + p, 4); p += 4;
203 out.warm_start = (flags & 1u) != 0;
204 std::memcpy(&out.total_size, data.data() + p, 8); p += 8;
205 std::memcpy(&out.chunk_len, data.data() + p, 4); p += 4;
206 uint32_t n_chunks_u32 = 0; std::memcpy(&n_chunks_u32, data.data() + p, 4); p += 4;
207 out.num_chunks = (int)n_chunks_u32;
208 std::memcpy(&out.last_chunk_len, data.data() + p, 4); p += 4;
209 uint8_t fp_len = data[p++];
210 if (out.warm_start && fp_len >= 4) {
211 std::memcpy(&out.interleave_batch, data.data() + p, 4);
212 std::cout << "Warm-start file detected. Interleave batch=" << out.interleave_batch << std::endl;
213 }
214 p += fp_len;
215 size_t payload_start = p;
216
217 // find IDX1 or IDX2
218 size_t idx_pos = std::string::npos;
219 int idx_version = 0;
220 for (size_t i = data.size() - 4; i-- > 0;) {
221 if (data[i] == 'I' && data[i+1] == 'D' && data[i+2] == 'X') {
222 if (data[i+3] == '2') { idx_pos = i; idx_version = 2; break; }
223 if (data[i+3] == '1') { idx_pos = i; idx_version = 1; break; }
224 }
225 }
226 if (idx_pos == std::string::npos) throw std::runtime_error("IDX not found");
227 if (data.size() < idx_pos + 4 + 4) throw std::runtime_error("Bad BOA2 IDX");
228
229 uint32_t crc_file = 0; std::memcpy(&crc_file, data.data() + data.size() - 4, 4);
230 uint32_t crc_calc = crc32_compute(data.data() + idx_pos, data.size() - idx_pos - 4);
231 if (crc_file != crc_calc) throw std::runtime_error("Bad BOA2 CRC");
232
233 size_t q = idx_pos + 4;
234 out.first_bytes.resize(out.num_chunks);
235 for (int i = 0; i < out.num_chunks; ++i) out.first_bytes[i] = data[q++];
236
237 std::vector<uint64_t> offsets(out.num_chunks, 0);
238 std::vector<uint64_t> lengths(out.num_chunks, 0);
239
240 if (idx_version == 1) {
241 // IDX1: delta-encoded offsets + lengths
242 uint64_t prev = 0;
243 for (int i = 0; i < out.num_chunks; ++i) {
244 uint64_t delta = uvarint_decode(data, q);
245 prev += delta;
246 offsets[i] = prev;
247 }
248 for (int i = 0; i < out.num_chunks; ++i) {
249 lengths[i] = uvarint_decode(data, q);
250 }
251 } else {
252 // IDX2: only lengths (offsets reconstructed)
253 for (int i = 0; i < out.num_chunks; ++i) {
254 lengths[i] = uvarint_decode(data, q);
255 }
256 uint64_t off = 0;
257 for (int i = 0; i < out.num_chunks; ++i) {
258 offsets[i] = off;
259 off += lengths[i];
260 }
261 }
262
263 std::vector<uint8_t> payload(data.begin() + payload_start, data.begin() + idx_pos);
264 out.streams.resize(out.num_chunks);
265 for (int i = 0; i < out.num_chunks; ++i) {
266 uint64_t off = offsets[i];
267 uint64_t len = lengths[i];
268 if (off + len > payload.size()) throw std::runtime_error("BOA2 payload out of range");
269 out.streams[i].assign(payload.begin() + off, payload.begin() + off + len);
270 }
271
272 out.lengths.resize(out.num_chunks);
273 for (int i = 0; i < out.num_chunks; ++i) {
274 out.lengths[i] = (i == out.num_chunks - 1) ? (int)out.last_chunk_len : (int)out.chunk_len;
275 }
276 return out;
277}

280void compress_v2(const std::string& model_path, const std::string& input_path, const std::string& output_path, MambaConfig config, int max_chunks, int gpu_batch, size_t chunk_size, bool warm_start, int tile_size = 0) {
281#ifdef ENABLE_ENERGY
282 EnergyTracker* energy_tracker = nullptr;
283 if (g_measure_energy) {
284 energy_tracker = new EnergyTracker();

290 std::ifstream fin(input_path, std::ios::binary);
291 std::vector<uint8_t> data((std::istreambuf_iterator<char>(fin)), std::istreambuf_iterator<char>());
292 fin.close();
293
294 size_t total_size = data.size();
295 int num_chunks = (int)((total_size + chunk_size - 1) / chunk_size);
296 if (max_chunks > 0) {
297 num_chunks = std::min(num_chunks, max_chunks);
298 }
299 size_t processed_size = std::min(total_size, (size_t)num_chunks * chunk_size);
300
301 std::cout << "Compressing " << processed_size << " bytes in " << num_chunks << " chunks..." << std::endl;
302
303 size_t last_chunk_len = (num_chunks > 1)
304 ? (processed_size - (size_t)(num_chunks - 1) * chunk_size)

307 std::vector<std::vector<uint8_t>> all_compressed_chunks(num_chunks);
308 std::vector<uint8_t> all_first_bytes(num_chunks);

310 int BATCH_SIZE = (gpu_batch > 0) ? gpu_batch : 64;
311 if (warm_start && BATCH_SIZE > num_chunks) BATCH_SIZE = num_chunks;
312
313 // Tile size: sub-chunk the forward pass to reduce VRAM, allowing larger batches
314 if (tile_size <= 0 || tile_size > (int)chunk_size) tile_size = (int)chunk_size;
315 int num_tiles = ((int)chunk_size + tile_size - 1) / tile_size;
316
317 // Warm-start interleaving: S = steps per lane, each lane processes contiguous original chunks
318 int S = warm_start ? ((num_chunks + BATCH_SIZE - 1) / BATCH_SIZE) : 1;
319 int num_iters = warm_start ? S : ((num_chunks + BATCH_SIZE - 1) / BATCH_SIZE);
320
321 if (warm_start) {
322 std::cout << "Using GPU Warm-Start Interleaved Compression. Batch=" << BATCH_SIZE
323 << ", Steps/lane=" << S << " (cold-starts: " << BATCH_SIZE << " instead of " << num_chunks << ")" << std::endl;
324 } else {
325 std::cout << "Using GPU Parallel Batching. Batch=" << BATCH_SIZE << std::endl;
326 }
327 if (tile_size < (int)chunk_size) {
328 std::cout << "Tiled forward: tile=" << tile_size << ", " << num_tiles << " tiles/chunk" << std::endl;
329 }
330
331 gpu_init_exp_lut();
332
333 BoaPredictorGPU gpu_model(config, 256, config.n_layers, BATCH_SIZE);
334 gpu_model.load_weights(model_path);
335
336 RCState* d_rc_states;
337 unsigned char* d_out_bufs;
338 int pitch_words = (int)((chunk_size * 10 + 4096 + 3) / 4);
339 int pitch_bytes = pitch_words * 4;
340
341 malloc_device((float**)&d_rc_states, BATCH_SIZE * sizeof(RCState));
342 malloc_device((float**)&d_out_bufs, (size_t)BATCH_SIZE * pitch_bytes);
343
344 int* d_batch_data[2] = {nullptr, nullptr};
345 int* h_batch_data[2] = {nullptr, nullptr};
346 std::vector<int> h_batch_lengths[2];
347 for (int i = 0; i < 2; ++i) {
348 malloc_device((float**)&d_batch_data[i], (size_t)BATCH_SIZE * chunk_size * sizeof(int));
349 checkCudaErrors(cudaHostAlloc((void**)&h_batch_data[i], (size_t)BATCH_SIZE * chunk_size * sizeof(int), cudaHostAllocDefault));
350 h_batch_lengths[i].resize(BATCH_SIZE);
351 }
352
353 int* d_sizes;
354 malloc_device((float**)&d_sizes, BATCH_SIZE * sizeof(int));
355
356 int* d_tokens;
357 int* d_lengths;
358 float* d_logits = nullptr;
359 int* d_sub_tokens = nullptr;
360 malloc_device((float**)&d_tokens, BATCH_SIZE * sizeof(int));
361 malloc_device((float**)&d_lengths, BATCH_SIZE * sizeof(int));
362 malloc_device(&d_logits, (size_t)BATCH_SIZE * tile_size * 256 * sizeof(float));
363 if (tile_size < (int)chunk_size) {
364 malloc_device((float**)&d_sub_tokens, (size_t)BATCH_SIZE * tile_size * sizeof(int));
365 }
366
367 unsigned char* h_out_bufs[2] = {nullptr, nullptr};
368 int* h_sizes[2] = {nullptr, nullptr};
369 for (int i = 0; i < 2; ++i) {
370 checkCudaErrors(cudaHostAlloc((void**)&h_out_bufs[i], (size_t)BATCH_SIZE * pitch_bytes, cudaHostAllocDefault));
371 checkCudaErrors(cudaHostAlloc((void**)&h_sizes[i], (size_t)BATCH_SIZE * sizeof(int), cudaHostAllocDefault));
372 }
373
374 gpu_model.allocate_chunk(tile_size);
375
376 double t_h2d = 0.0;
377 double t_compute = 0.0;
378 double t_d2h = 0.0;
379 double t_fwd = 0.0;
380 double t_encode = 0.0;
381
382 cudaEvent_t ev_start, ev_mid, ev_end;
383 checkCudaErrors(cudaEventCreate(&ev_start));
384 checkCudaErrors(cudaEventCreate(&ev_mid));
385 checkCudaErrors(cudaEventCreate(&ev_end));
386
387 auto start_time = std::chrono::high_resolution_clock::now();
388 int chunks_done = 0;
389
390 for (int iter = 0; iter < num_iters; ++iter) {
391 int buf = iter & 1;
392
393 // Compute current_batch and abs_idx mapping
394 int current_batch;
395 if (warm_start) {
396 // Interleaved: lane j -> original chunk (j * S + iter)
397 // Lanes with j*S+iter >= num_chunks are inactive (always trailing)
398 current_batch = 0;
399 for (int b = 0; b < BATCH_SIZE; ++b) {
400 if ((int64_t)b * S + iter < num_chunks) current_batch = b + 1;
401 else break;
402 }
403 } else {
404 int chunk_idx = iter * BATCH_SIZE;
405 current_batch = std::min(BATCH_SIZE, num_chunks - chunk_idx);
406 }
407
408 auto now = std::chrono::high_resolution_clock::now();
409 double elapsed = std::chrono::duration<double>(now - start_time).count();
410 print_progress(chunks_done, num_chunks, elapsed, processed_size, "Compressing");
411
412 memset(h_batch_data[buf], 0, (size_t)BATCH_SIZE * chunk_size * sizeof(int));
413 for (int b = 0; b < current_batch; ++b) {
414 int abs_idx = warm_start ? (b * S + iter) : (iter * BATCH_SIZE + b);
415 if (abs_idx >= num_chunks) { h_batch_lengths[buf][b] = 0; continue; }
416 size_t start_off = (size_t)abs_idx * chunk_size;
417 size_t len = std::min(chunk_size, total_size - start_off);
418 for(size_t k=0; k<len; ++k) {
419 h_batch_data[buf][(size_t)b * chunk_size + k] = data[start_off + k];
420 }
421 all_first_bytes[abs_idx] = (len > 0) ? data[start_off] : 0;
422 h_batch_lengths[buf][b] = (int)len;
423 }
424
425 auto t0 = std::chrono::high_resolution_clock::now();
426 checkCudaErrors(cudaMemcpy(d_batch_data[buf], h_batch_data[buf], (size_t)BATCH_SIZE * chunk_size * sizeof(int), cudaMemcpyHostToDevice));
427 auto t1 = std::chrono::high_resolution_clock::now();
428 t_h2d += std::chrono::duration<double>(t1 - t0).count();
429
430 t0 = std::chrono::high_resolution_clock::now();
431 gpu_rc_init(d_rc_states, current_batch);
432 // Warm-start: only reset model cache on first iteration
433 if (!warm_start || iter == 0) {
434 gpu_model.reset_cache();
435 }
436
437 checkCudaErrors(cudaMemcpy(d_lengths, h_batch_lengths[buf].data(), current_batch * sizeof(int), cudaMemcpyHostToDevice));
438
439 checkCudaErrors(cudaEventRecord(ev_start));
440
441 // Tiled forward + encode: process chunk in tiles of tile_size
442 for (int tile_idx = 0; tile_idx < num_tiles; ++tile_idx) {
443 int sub_start = tile_idx * tile_size;
444 int sub_end = std::min(sub_start + tile_size, (int)chunk_size);
445 int sub_len = sub_end - sub_start;
446
447 // Gather sub-chunk tokens from strided [B, chunk_size] to compact [B, sub_len]
448 int* fwd_tokens;
449 if (d_sub_tokens) {
450 checkCudaErrors(cudaMemcpy2D(d_sub_tokens, sub_len * sizeof(int),
451 d_batch_data[buf] + sub_start, chunk_size * sizeof(int),
452 sub_len * sizeof(int), current_batch,
453 cudaMemcpyDeviceToDevice));
454 fwd_tokens = d_sub_tokens;
455 } else {
456 fwd_tokens = d_batch_data[buf]; // no tiling, use directly
457 }
458
459 // Forward model (state carries forward between tiles)
460 gpu_model.forward_chunk(fwd_tokens, d_logits, sub_len);
461
462 // Encode this tile's token range
463 // For non-final tiles: max_len = sub_end + 1 so the loop encodes up to sub_end
464 // For final tile: max_len = sub_end, loop stops at len-1 (natural chunk end)
465 bool is_last_tile = (sub_end >= (int)chunk_size);
466 int enc_max_len = is_last_tile ? sub_end : (sub_end + 1);
467 gpu_rc_encode_chunk_warp(d_logits, d_batch_data[buf], d_lengths, d_rc_states, (unsigned int*)d_out_bufs,
468 pitch_words, 256, current_batch, (int)chunk_size, enc_max_len, sub_start, sub_len * 256);
469 }
470
471 checkCudaErrors(cudaEventRecord(ev_end));
472 checkCudaErrors(cudaEventSynchronize(ev_end));
473 float ms_fwd = 0.0f;
474 checkCudaErrors(cudaEventElapsedTime(&ms_fwd, ev_start, ev_end));
475 t_fwd += ms_fwd / 1000.0;
476
477 gpu_rc_finish_batch(d_rc_states, (unsigned int*)d_out_bufs, pitch_words, d_sizes, current_batch);
478 checkCudaErrors(cudaDeviceSynchronize());
479 t1 = std::chrono::high_resolution_clock::now();
480 t_compute += std::chrono::duration<double>(t1 - t0).count();
481
482 t0 = std::chrono::high_resolution_clock::now();
483 checkCudaErrors(cudaMemcpy(h_out_bufs[buf], d_out_bufs, (size_t)current_batch * pitch_bytes, cudaMemcpyDeviceToHost));
484 checkCudaErrors(cudaMemcpy(h_sizes[buf], d_sizes, (size_t)current_batch * sizeof(int), cudaMemcpyDeviceToHost));
485 t1 = std::chrono::high_resolution_clock::now();
486 t_d2h += std::chrono::duration<double>(t1 - t0).count();
487 for (int b = 0; b < current_batch; ++b) {
488 int abs_idx = warm_start ? (b * S + iter) : (iter * BATCH_SIZE + b);
489 if (abs_idx >= num_chunks) continue;
490 int size_words = h_sizes[buf][b];
491 all_compressed_chunks[abs_idx].resize(size_words * 4);
492 memcpy(all_compressed_chunks[abs_idx].data(), h_out_bufs[buf] + (size_t)b * pitch_bytes, size_words * 4);
493 }
494 chunks_done += current_batch;
495 }
496
497 auto end_time = std::chrono::high_resolution_clock::now();
498 double duration = std::chrono::duration<double>(end_time - start_time).count();
499 print_progress(num_chunks, num_chunks, duration, processed_size, "Compressing");
500
501 double mb_s = (double)processed_size / (1024.0 * 1024.0) / duration;
502 std::cout << "\nDone. Time: " << std::fixed << std::setprecision(3) << duration << "s, Avg Throughput: " << mb_s << " MB/s\n";
503
504 if (g_show_timings) {
505 std::cout << "Timing (s): H2D=" << t_h2d << " Compute=" << t_compute << " D2H=" << t_d2h
506 << " Forward=" << t_fwd << " Encode=" << t_encode << "\n";
507 }
508
509#ifdef ENABLE_ENERGY
510 if (energy_tracker) {
511 energy_tracker->stop();
512 energy_tracker->calculate_energy();

524 checkCudaErrors(cudaEventDestroy(ev_end));

538 // Write BOA2 container
539 std::ofstream fout(output_path, std::ios::binary);
540 const char magic[4] = {'B','O','A','2'};
541 const uint32_t version = 1;
542 const uint32_t flags_val = warm_start ? 1u : 0u;
543 const uint32_t chunk_len_u32 = (uint32_t)chunk_size;
544 const uint32_t n_chunks_u32 = (uint32_t)num_chunks;
545 const uint32_t last_chunk_u32 = (uint32_t)last_chunk_len;
546 const uint8_t fp_len = warm_start ? 4 : 0; // 4 bytes for interleave_batch when warm-start

550 fout.write(reinterpret_cast<const char*>(&flags_val), 4);
551 fout.write(reinterpret_cast<const char*>(&processed_size), 8);
552 fout.write(reinterpret_cast<const char*>(&chunk_len_u32), 4);
553 fout.write(reinterpret_cast<const char*>(&n_chunks_u32), 4);
554 fout.write(reinterpret_cast<const char*>(&last_chunk_u32), 4);
555 fout.write(reinterpret_cast<const char*>(&fp_len), 1);
556 if (warm_start) {
557 uint32_t interleave_batch = (uint32_t)BATCH_SIZE;
558 fout.write(reinterpret_cast<const char*>(&interleave_batch), 4);
559 }
560
561 std::vector<uint64_t> offsets(num_chunks, 0);
562 std::vector<uint64_t> lengths(num_chunks, 0);
563 uint64_t off = 0;
564 for (int i = 0; i < num_chunks; ++i) {
565 offsets[i] = off;
566 lengths[i] = all_compressed_chunks[i].size();
567 fout.write(reinterpret_cast<const char*>(all_compressed_chunks[i].data()), all_compressed_chunks[i].size());
568 off += lengths[i];
569 }
570
571 std::vector<uint8_t> idx;
572 idx.insert(idx.end(), {'I','D','X','2'});
573 idx.insert(idx.end(), all_first_bytes.begin(), all_first_bytes.end());
574 // IDX2: only lengths (offsets are reconstructed as cumulative sum)
575 for (int i = 0; i < num_chunks; ++i) {
576 uvarint_encode(idx, lengths[i]);
577 }
578 uint32_t crc = crc32_compute(idx.data(), idx.size());
579 fout.write(reinterpret_cast<const char*>(idx.data()), idx.size());
580 fout.write(reinterpret_cast<const char*>(&crc), 4);
581 fout.close();

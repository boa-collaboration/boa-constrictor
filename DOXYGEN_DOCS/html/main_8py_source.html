22def resolve_config_path(config_arg: str, experiments_root: Path = Path('experiments')) -> Path:
23 """Resolve a --config argument which may be a path or an experiment name.
24
25 Order:
26 1. If the argument is an existing file path, return it.
27 2. If it's a simple experiment name (no existing path), look for
28 experiments/<name>/<name>.yaml and return if exists.
29 3. Fallback to configs/<name>.yaml if present.
30 4. Raise FileNotFoundError.
31 """
32 if config_arg is None:
33 return None
34 p = Path(config_arg)
35 # Direct file path provided
36 if p.exists():
37 return p
38
39 # Try experiments/<name>/<name>.yaml
40 name = p.stem
41 exp_cfg = experiments_root / name / f"{name}.yaml"
42 if exp_cfg.exists():
43 return exp_cfg
44
45 # Try configs/<name>.yaml
46 cfg_cfg = Path('configs') / f"{name}.yaml"
47 if cfg_cfg.exists():
48 return cfg_cfg
49
50 raise FileNotFoundError(f"Could not resolve config argument '{config_arg}' to a config file")

344 def _infer_backbone_from_keys(keys):
345 """Infer backbone type from state_dict key patterns.
346
347 Each backbone produces distinctive key names inside
348 ``blocks.<i>.`` — we check for those fingerprints.
349 """
350 ks = set(keys)
351 has = lambda pat: any(pat in k for k in ks) # noqa: E731
352
353 # Unique sub-module names per backbone
354 # RWKV-8 ROSA has .rosa_mix.rosa.
355 if has('.rosa_mix.'): return 'rwkv8'
356 # RWKV-7 has .time_mix.W_a (non-diagonal transition); check before rwkv6
357 if has('.time_mix.') and has('.time_mix.W_a.'): return 'rwkv7'
358 if has('.time_mix.'): return 'rwkv6'
359 if has('.rg_lru.'): return 'griffin'
360 if has('.mlstm.'): return 'xlstm'
361 if has('.mamba2.'): return 'mamba2'
362 if has('.mingru.'): return 'mingru'
363 if has('.lstm.'): return 'lstm'
364 if has('.gru.'): return 'gru'
365 # Transformer has q_proj but no recurrence sub-modules
366 if has('.q_proj.') or has('.kv_proj.'):
367 return 'transformer'
368 # Distinguish mamba (improved) vs mambav1 by FFN structure
369 if has('.mamba.'):
370 # Improved mamba uses SwiGLU (ff.w_gate) + final_norm
371 if has('.ff.w_gate.') and has('final_norm.'):
372 return 'mamba'
373 return 'mambav1'
374 return None

376 def _load_model_from_path(model, path: Path, expected_backbone: str = None):
377 if not path.exists():
378 raise FileNotFoundError(f"Model file not found: {path}")
379 obj = torch.load(path, map_location='cpu')
380
381 saved_backbone = None
382 state_dict = None
383
384 if isinstance(obj, dict):
385 # New backbone-aware format: {'backbone': str, 'state_dict': dict}
386 if 'backbone' in obj and 'state_dict' in obj:
387 saved_backbone = obj['backbone']
388 state_dict = obj['state_dict']
389 # Legacy format with state_dict under a key
390 elif 'state_dict' in obj:
391 state_dict = obj['state_dict']
392 # Pure state_dict (all string keys, no 'backbone' key)
393 elif all(isinstance(k, str) for k in obj.keys()):
394 state_dict = obj
395
396 if state_dict is not None:
397 # For legacy checkpoints, infer backbone from key patterns
398 if saved_backbone is None:
399 inferred = _infer_backbone_from_keys(state_dict.keys())
400 if inferred:
401 saved_backbone = inferred
402 print(f" Inferred backbone from checkpoint keys: {inferred}")
403
404 # Backbone compatibility check
405 if saved_backbone and expected_backbone and saved_backbone != expected_backbone:
406 raise ValueError(
407 f"Backbone mismatch: checkpoint was trained with "
408 f"'{saved_backbone}' but current config specifies "
409 f"'{expected_backbone}'. Change backbone in config to "
410 f"'{saved_backbone}' or retrain."
411 )
412 if saved_backbone:
413 print(f" Checkpoint backbone: {saved_backbone}")
414 elif expected_backbone:
415 print(f" Legacy checkpoint (could not infer backbone). "
416 f"Assuming compatible with '{expected_backbone}'.")
417
418 # Load with strict=True first to catch mismatches cleanly
419 missing, unexpected = model.load_state_dict(state_dict, strict=False)
420 if missing or unexpected:
421 print(f" [WARN] state_dict load: {len(missing)} missing, "
422 f"{len(unexpected)} unexpected keys")
423 if len(missing) > len(state_dict) * 0.5:
424 raise ValueError(
425 f"Too many missing keys ({len(missing)}/{len(state_dict)}). "
426 f"This checkpoint is likely incompatible with backbone "
427 f"'{expected_backbone}'."
428 )
429 return model
430
431 # If whole model was saved
432 if hasattr(obj, 'state_dict') and hasattr(obj, 'parameters'):
433 if expected_backbone and hasattr(obj, '_backbone_name'):
434 if obj._backbone_name != expected_backbone:
435 raise ValueError(
436 f"Backbone mismatch: saved model uses "
437 f"'{obj._backbone_name}' but config specifies "
438 f"'{expected_backbone}'."
439 )
440 return obj
441
442 raise ValueError(f"Unrecognized checkpoint format at {path}")

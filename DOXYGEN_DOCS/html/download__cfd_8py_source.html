31def generate_turbulent_field_3d(nx, ny, nz, spectrum_func, seed=None):
32 """
33 Generate a 3D turbulent field using spectral synthesis
34
35 Args:
36 nx, ny, nz: Grid dimensions
37 spectrum_func: Function that takes wavenumber and returns energy
38 seed: Random seed for reproducibility
39
40 Returns:
41 3D numpy array with turbulent field
42 """
43 if seed is not None:
44 np.random.seed(seed)
45
46 # Create wavenumber grid
47 kx = np.fft.fftfreq(nx, 1.0/nx)
48 ky = np.fft.fftfreq(ny, 1.0/ny)
49 kz = np.fft.fftfreq(nz, 1.0/nz)
50
51 KX, KY, KZ = np.meshgrid(kx, ky, kz, indexing='ij')
52 K = np.sqrt(KX**2 + KY**2 + KZ**2)
53 K[K == 0] = 1e-10 # Avoid division by zero
54
55 # Generate random phases
56 phases = np.random.uniform(0, 2*np.pi, (nx, ny, nz))
57
58 # Apply energy spectrum
59 amplitudes = np.sqrt(spectrum_func(K))
60
61 # Create Fourier coefficients
62 fourier_field = amplitudes * np.exp(1j * phases)
63
64 # Inverse FFT to get real-space field
65 real_field = np.fft.ifftn(fourier_field).real
66
67 return real_field.astype(np.float32)

69def kolmogorov_spectrum(k, k_peak=4, epsilon=1.0):
70 """
71 Kolmogorov energy spectrum for isotropic turbulence
72 E(k) ∝ k^(-5/3) in inertial range
73 """
74 # Add exponential cutoff at high wavenumbers
75 # Peak at k_peak, then -5/3 decay
76 E = np.zeros_like(k)
77 mask = k > 0
78 E[mask] = (k[mask]**4) * np.exp(-2*(k[mask]/k_peak)**2) * (k[mask]**(-5/3))
79 return E

81def generate_isotropic_turbulence_data():
82 """
83 Generate synthetic isotropic turbulence data
84 Returns data as a numpy array similar to what JHTDB would provide
85 """
86 print(f"Generating synthetic turbulence data: {GRID_SIZE_X}x{GRID_SIZE_Y}x{GRID_SIZE_Z}")

93 lambda k: kolmogorov_spectrum(k, ENERGY_SPECTRUM_PEAK),
94 seed=42

97 print("Generating velocity field (v component)...")
98 v = generate_turbulent_field_3d(
99 GRID_SIZE_X, GRID_SIZE_Y, GRID_SIZE_Z,
100 lambda k: kolmogorov_spectrum(k, ENERGY_SPECTRUM_PEAK),
101 seed=43
102 )

104 print("Generating velocity field (w component)...")
105 w = generate_turbulent_field_3d(
106 GRID_SIZE_X, GRID_SIZE_Y, GRID_SIZE_Z,
107 lambda k: kolmogorov_spectrum(k, ENERGY_SPECTRUM_PEAK),
108 seed=44
109 )
110
111 print("Generating pressure field...")
112 # Pressure fluctuations typically have similar spectral characteristics
113 p = generate_turbulent_field_3d(
114 GRID_SIZE_X, GRID_SIZE_Y, GRID_SIZE_Z,
115 lambda k: kolmogorov_spectrum(k, ENERGY_SPECTRUM_PEAK) * 0.5,
116 seed=45
117 )
118
119 # Normalize fields to reasonable physical values
120 # Typical velocity fluctuations: ~1 m/s
121 u = (u - u.mean()) / u.std() * 0.5
122 v = (v - v.mean()) / v.std() * 0.5
123 w = (w - w.mean()) / w.std() * 0.5
124
125 # Pressure fluctuations: ~1 Pa
126 p = (p - p.mean()) / p.std() * 1.0
127
128 # Stack into single array: (X, Y, Z, 4)
129 print("\nCombining fields...")
130 combined_data = np.stack([u, v, w, p], axis=-1)
131
132 print(f"\nData shape: {combined_data.shape}")
133 print(f"Data dtype: {combined_data.dtype}")
134 print(f"Velocity magnitude range: [{np.sqrt(u**2 + v**2 + w**2).min():.3f}, {np.sqrt(u**2 + v**2 + w**2).max():.3f}]")
135 print(f"Pressure range: [{p.min():.3f}, {p.max():.3f}]")
136
137 return combined_data

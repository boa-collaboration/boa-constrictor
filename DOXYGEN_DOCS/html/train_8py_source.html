16# --- eval loop (reports mean bpp across the loader) ---
17@torch.inference_mode()
18def evaluate_bpp(model, loader, criterion, device="cuda", vocab_size=256):
19 model.eval().to(device)
20 total_loss = 0.0
21 total_tokens = 0
22 for batch in loader:
23 x = batch[:, :-1].to(device)
24 y = batch[:, 1:].to(device)
25 logits = model(x) # [B, L-1, vocab_size]
26 loss = criterion(logits.reshape(-1, vocab_size), y.reshape(-1))
27 total_loss += loss.item() * y.numel()
28 total_tokens += y.numel()
29 mean_nll = total_loss / max(1, total_tokens)
30 bpp = mean_nll / np.log(2) # bits per input byte
31 return bpp

34
35 IS_CUDA = torch.cuda.is_available() and device == "cuda"
36
37 # Pre-process position weights (if provided) for fast per-token weighting
38 _pw = None
39 if position_weights is not None:
40 # position_weights: [seq_len] float tensor — weight for each target position
41 _pw = torch.tensor(position_weights, dtype=torch.float32, device=device)
42 print(f"[INFO] Position-weighted loss: coarse={position_weights[0]:.2f}, fine={position_weights[-1]:.2f}")

52 def get_autocast_dtype(precision):
53 if precision == "bf16":
54 return torch.bfloat16
55 elif precision == "fp16":
56 return torch.float16
57 elif precision == "fp8":
58 try:
59 return torch.float8_e5m2 # Hopper architecture only (H100 / RTX 5090)
60 except AttributeError:
61 print("[WARN] FP8 not supported on this PyTorch build, falling back to FP16")
62 return torch.float16
63 else:
64 return torch.float32
65
66 AUTODTYPE = get_autocast_dtype(PRECISION)
67 amp_enabled = PRECISION in ["bf16", "fp16", "fp8"] and IS_CUDA
68 save_half = PRECISION in ["fp16", "fp8"] # save weights as fp16 when training in reduced precision (bf16 saves as fp32)
69 # GradScaler is not needed/used for bf16 (no loss scaling required)
70 use_scaler = PRECISION in ["fp16", "fp8"] and IS_CUDA
71 scaler = torch.amp.GradScaler("cuda", enabled=use_scaler)
72
73 best_val_bpp = float("inf")
74 best_epoch = 0
75 epochs_without_improvement = 0
76 best_state_dict = None
77
78 # Wandb logging helper
79 _wb = wandb_run # None when wandb is disabled
80
81 # Helper to strip torch.compile's _orig_mod. prefix from state_dict keys
82 def _clean_state_dict(sd):
83 return {k.replace('_orig_mod.', ''): v for k, v in sd.items()}
84
85 def _get_backbone_name(m):
86 # torch.compile wraps model in OptimizedModule; unwrap to find _backbone_name
87 inner = getattr(m, '_orig_mod', m)
88 return getattr(inner, '_backbone_name', 'unknown')
89
90 model.train().to(device)
91 train_steps_per_epoch = len(train_loader)
92 total_train_steps = max(1, train_steps_per_epoch)
93 global_step = (start_epoch - 1) * total_train_steps
94 for epoch in range(start_epoch, NUM_EPOCHS + 1):
95 epoch_start = time.perf_counter()
96 epoch_tokens = 0
97 loader = tqdm(train_loader, total=total_train_steps, desc=f"Epoch {epoch} [{PRECISION}]", disable=not progress)
98 for batch in loader:
99 x = batch[:, :-1].to(device, non_blocking=True)
100 y = batch[:, 1:].to(device, non_blocking=True)
101 epoch_tokens += y.numel()
102
103 optimizer.zero_grad(set_to_none=True)
104
105 # --- Automatic mixed precision block ---
106 with torch.autocast(device_type=device, dtype=AUTODTYPE, enabled=amp_enabled):
107 logits = model(x)
108 if _pw is not None:
109 # Per-position weighted loss: tile weights across the sequence
110 # x has shape [B, seq_len-1]; positions repeat with period event_bytes
111 B, L = y.shape
112 pw_len = len(_pw)
113 # Tile position weights to cover L positions
114 reps = (L + pw_len - 1) // pw_len
115 weights = _pw.repeat(reps)[:L] # [L]
116 weights = weights.unsqueeze(0).expand(B, -1) # [B, L]
117 import torch.nn.functional as F_local
118 loss_tok = F_local.cross_entropy(
119 logits.reshape(-1, vocab_size), y.reshape(-1), reduction="none"
120 ).reshape(B, L)
121 loss = (loss_tok * weights).sum() / weights.sum()
122 else:
123 loss = criterion(logits.reshape(-1, vocab_size), y.reshape(-1))
124
125 # --- Scaled backward for FP16/FP8 ---
126 scaler.scale(loss).backward()
127 if grad_clip > 0:
128 scaler.unscale_(optimizer)
129 torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=grad_clip)
130 scaler.step(optimizer)
131 scaler.update()
132
133 # --- Progress ---
134 bits_per_byte = loss.item() / np.log(2)
135 global_step += 1
136 if progress:
137 loader.set_postfix(
138 loss=f"{loss.item():.4f}",
139 bits=f"{bits_per_byte:.3f}",
140 ratio=f"{(8 / bits_per_byte):.2f}x"
141 )
142
143 # --- Wandb step-level logging ---
144 if _wb is not None:
145 _wb.log({
146 "train/loss": loss.item(),
147 "train/bpp": bits_per_byte,
148 "train/compression_ratio": 8.0 / max(bits_per_byte, 1e-8),
149 }, step=global_step)
150
151 epoch_elapsed = time.perf_counter() - epoch_start
152 tok_per_sec = epoch_tokens / max(epoch_elapsed, 1e-6)
153 mb_per_sec = epoch_tokens / (1024 * 1024) / max(epoch_elapsed, 1e-6)
154 print(f" Epoch {epoch} throughput: {tok_per_sec:,.0f} tok/s ({mb_per_sec:.2f} MB/s), {epoch_elapsed:.1f}s")
155
156 _sd = _clean_state_dict(model.state_dict())
157 if save_half:
158 _sd = {k: v.half() if v.is_floating_point() else v for k, v in _sd.items()}
159 _ckpt = {
160 'state_dict': _sd,
161 'backbone': _get_backbone_name(model),
162 }
163 torch.save(_ckpt, f"{name}_{datetime.now().strftime('%dth%b')}_Checkpoint_epoch_{epoch}_{PRECISION}.pt")
164 val_bpp = evaluate_bpp(model, val_loader, criterion, device=device, vocab_size=vocab_size)
165 current_lr = optimizer.param_groups[0]['lr']
166 print(f"[Epoch {epoch}] val bpp={val_bpp:.4f} (ratio ~ {8/val_bpp:.2f}x) lr={current_lr:.2e}")
167
168 # --- Wandb epoch-level logging ---
169 if _wb is not None:
170 _wb.log({
171 "epoch": epoch,
172 "val/bpp": val_bpp,
173 "val/compression_ratio": 8.0 / max(val_bpp, 1e-8),
174 "train/throughput_tok_s": tok_per_sec,
175 "train/throughput_MB_s": mb_per_sec,
176 "train/epoch_time_s": epoch_elapsed,
177 "train/lr": current_lr,
178 "best/val_bpp": best_val_bpp,
179 "best/epoch": best_epoch,
180 }, step=global_step)
181
182 if scheduler is not None:
183 scheduler.step()
184
185 # --- Early stopping logic ---
186 if val_bpp < best_val_bpp:
187 best_val_bpp = val_bpp
188 best_epoch = epoch
189 epochs_without_improvement = 0
190 import copy
191 best_state_dict = copy.deepcopy(model.state_dict())
192 else:
193 epochs_without_improvement += 1
194 if patience > 0:
195 print(f" [early stop] no improvement for {epochs_without_improvement}/{patience} epochs (best={best_val_bpp:.4f} @ epoch {best_epoch})")
196 if patience > 0 and epochs_without_improvement >= patience:
197 print(f" [early stop] Stopping at epoch {epoch}. Restoring best model from epoch {best_epoch}.")
198 break
199
200 # Restore best model if early stopping was used
201 if best_state_dict is not None and patience > 0:
202 model.load_state_dict(best_state_dict)
203 print(f" [early stop] Restored best model (val bpp={best_val_bpp:.4f}, epoch {best_epoch})")
204
205 _final_sd = _clean_state_dict(model.state_dict())
206 if save_half:
207 _final_sd = {k: v.half() if v.is_floating_point() else v for k, v in _final_sd.items()}
208 _final_ckpt = {
209 'state_dict': _final_sd,
210 'backbone': _get_backbone_name(model),
211 }
212 torch.save(_final_ckpt, f"{name}_final_model_{PRECISION}.pt")
213 test_bpp = evaluate_bpp(model, test_loader, criterion, device=device, vocab_size=vocab_size)
214 print(f"[TEST] bpp={test_bpp:.4f} ratio ~ {8/test_bpp:.2f}x")
215
216 # --- Wandb final summary ---
217 if _wb is not None:

219 "test/bpp": test_bpp,
220 "test/compression_ratio": 8.0 / max(test_bpp, 1e-8),
221 }, step=global_step)
222 _wb.summary["test_bpp"] = test_bpp
223 _wb.summary["test_compression_ratio"] = 8.0 / max(test_bpp, 1e-8)
224 _wb.summary["best_val_bpp"] = best_val_bpp
225 _wb.summary["best_epoch"] = best_epoch

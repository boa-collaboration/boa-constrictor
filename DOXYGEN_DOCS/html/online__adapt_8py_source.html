1"""
2Online head adaptation for BOA compression.
3
4During compression/decompression, the prediction head is continuously
5fine-tuned on the bytes being processed. Both encoder and decoder
6see the same bytes and perform identical gradient updates, keeping
7their models perfectly synchronized.
8
9This is the neural equivalent of cmix's online model adaptation —
10the single most impactful technique in compression.
11
12Design:
13 - Backbone (Mamba/LSTM/etc.) stays FROZEN
14 - Only the head (Linear → SiLU/ReLU → Linear, ~50K params for d=128)
15 is adapted
16 - Adaptation happens BETWEEN batches (not inside the streaming loop)
17 so we don't fight @torch.inference_mode or Mamba's InferenceParams
18 - After each batch of N streams finishes, we take the raw bytes that
19 were just compressed/decompressed, run ONE forward pass through the
20 full model, and do K gradient steps on the head only
21 - Both encoder and decoder process the same bytes → identical updates
22
23Usage in boa.py (between batch loops):
24 adapter = HeadAdapter(model, lr=2e-4)
25 for batch_start in range(0, n_chunks, gpu_streams):
26 # ... compress batch ...
27 adapter.adapt_on_batch(batch_bytes, device=device)
28"""
29

38class HeadAdapter:
39 """Manages online adaptation of the prediction head during compression.
40
41 Between batches, takes the compressed/decompressed bytes and runs
42 K gradient steps on the head using the full model forward pass.
43 The backbone stays frozen — only head parameters receive gradients.
44 """
45
46 def __init__(self, model, lr: float = 2e-4, adapt_steps: int = 1,
47 max_grad_norm: float = 1.0, max_seq_len: int = 4096):
48 self.model = model
49 self.lr = lr
50 self.adapt_steps = adapt_steps
51 self.max_grad_norm = max_grad_norm
52 self.max_seq_len = max_seq_len # cap sequence length for adaptation to save memory
53
54 # Identify head parameters
55 self.head = model.head
56 self.head_params = list(self.head.parameters())
57
58 # Save original head state so we can restore after compress/decompress
59 # (ensures encoder and decoder both start from the same checkpoint)
60 self._original_state = {k: v.clone() for k, v in self.head.state_dict().items()}
61
62 # Freeze everything except head
63 for param in model.parameters():
64 param.requires_grad_(False)
65 for param in self.head_params:
66 param.requires_grad_(True)
67
68 # Deterministic SGD — no momentum so encoder/decoder produce
69 # bitwise identical updates given the same data
70 self.optimizer = torch.optim.SGD(self.head_params, lr=lr)
71
72 def restore_head(self):
73 """Restore head to original (pre-adaptation) state.
74 Call after compress() finishes so that decompress() starts identically.
75 """
76 self.head.load_state_dict(self._original_state)
77 self._de_inference_head()
78 # Reset optimizer state
79 self.optimizer = torch.optim.SGD(self.head_params, self.lr)
80
81 def _de_inference_head(self):
82 """Replace any inference-mode-tainted parameters with fresh copies.
83
84 compress_GPU / decompress_GPU use @torch.inference_mode(), which
85 permanently marks parameters used in that context as inference
86 tensors. Such tensors cannot participate in autograd.
87 We fix this by replacing them with non-inference clones.
88 """
89 for name, module in self.head.named_modules():
90 if isinstance(module, torch.nn.Linear):
91 if module.weight.is_inference():
92 module.weight = torch.nn.Parameter(
93 module.weight.data.clone(), requires_grad=True
94 )
95 if module.bias is not None and module.bias.is_inference():
96 module.bias = torch.nn.Parameter(
97 module.bias.data.clone(), requires_grad=True
98 )
99 # Refresh param list and optimizer after parameter replacement
100 self.head_params = list(self.head.parameters())
101 for p in self.head_params:
102 p.requires_grad_(True)
103 self.optimizer = torch.optim.SGD(self.head_params, self.lr)
104
105 def _get_backbone_features(self, x: torch.Tensor) -> torch.Tensor:
106 """Run backbone only (frozen, no grad) to get hidden states.
107
108 x: [B, L] long tensor of byte sequences
109 Returns: [B, L, D] plain tensor (no grad, no inference mode)
110 """
111 # Use no_grad + explicit inference_mode(False) to ensure we get
112 # plain tensors even if Mamba blocks have cached inference tensors
113 with torch.no_grad(), torch.inference_mode(False):
114 h = self.model.embedding(x) # [B, L, D]
115
116 # Check if model has inference_params-style forward
117 if hasattr(self.model, '_backbone_name') and \
118 self.model._backbone_name in ('mamba', 'mambav1', 'mamba2'):
119 # For Mamba models, use the full forward pass through blocks
120 # (NOT streaming mode — full sequence parallel)
121 for blk in self.model.blocks:
122 h = blk(h) # No inference_params → runs in training/parallel mode
123 else:
124 for blk in self.model.blocks:
125 h = blk(h)
126
127 # Apply final_norm if present
128 if hasattr(self.model, 'final_norm'):
129 h = self.model.final_norm(h)
130
131 return h.detach().clone() # clone ensures plain tensor, no inference mode residue

133 def adapt_on_batch(self, byte_sequences: list, device: str = "cuda"):
134 """Adapt the head on a batch of byte sequences.
135
136 byte_sequences: list of 1D byte tensors or numpy arrays (the raw
137 bytes that were just compressed/decompressed).
138 device: torch device string.
139
140 Must be called OUTSIDE @torch.inference_mode() — we explicitly
141 disable inference mode and fix tainted parameters here.
142 """
143 if not byte_sequences or self.adapt_steps <= 0:
144 return
145
146 # Fix parameters tainted by @torch.inference_mode() in codec
147 self._de_inference_head()
148
149 # Explicitly exit inference mode for the adaptation step
150 with torch.inference_mode(False):
151 self._adapt_on_batch_impl(byte_sequences, device)
152
153 def _adapt_on_batch_impl(self, byte_sequences: list, device: str):
154 """Inner implementation of adapt_on_batch (runs outside inference mode)."""
155 # Convert to tensor and truncate to max_seq_len for memory efficiency
156 def _seq_len(s):
157 if isinstance(s, torch.Tensor):
158 return s.numel()
159 return len(s) if hasattr(s, '__len__') else int(s.shape[-1])
160 max_len = min(max(_seq_len(s) for s in byte_sequences), self.max_seq_len)
161 batch_size = len(byte_sequences)
162
163 # Build padded tensor [B, L]
164 x = torch.zeros(batch_size, max_len, dtype=torch.long, device=device)
165 for i, seq in enumerate(byte_sequences):
166 if isinstance(seq, torch.Tensor):
167 s = seq.to(device).flatten()[:max_len]
168 elif isinstance(seq, (bytes, bytearray)):
169 s = torch.tensor(list(seq[:max_len]), dtype=torch.long, device=device)
170 else:
171 import numpy as np
172 s = torch.from_numpy(np.frombuffer(seq, dtype=np.uint8)[:max_len].copy()).to(device)
173 x[i, :len(s)] = s.to(torch.long)
174
175 # Input: x[:, :-1], Target: x[:, 1:]
176 x_in = x[:, :-1]
177 y = x[:, 1:]
178
179 if x_in.shape[1] < 1:
180 return
181
182 # Get backbone features (frozen, no grad)
183 h = self._get_backbone_features(x_in) # [B, L-1, D]
184
185 # Adaptation steps: gradient only flows through the head
186 self.head.train()
187 for _ in range(self.adapt_steps):
188 logits = self.head(h) # [B, L-1, V]
189 V = logits.shape[-1]
190 loss = F.cross_entropy(logits.reshape(-1, V), y.reshape(-1))
191
192 self.optimizer.zero_grad()
193 loss.backward()
194 if self.max_grad_norm > 0:
195 torch.nn.utils.clip_grad_norm_(self.head_params, self.max_grad_norm)
196 self.optimizer.step()
197
198 # Back to eval for inference
199 self.head.eval()

201 def adapt_on_raw_bytes(self, raw_bytes: bytes, chunk_len: int,
202 device: str = "cuda", max_chunks: int = 64):
203 """Convenience: adapt on raw bytes split into chunks.
204
205 Takes a contiguous byte buffer and splits it into chunks for adaptation.
206 Useful when called from boa.py.
207 """
208 import numpy as np
209 total = len(raw_bytes)
210 n_chunks = min((total + chunk_len - 1) // chunk_len, max_chunks)
211
212 seqs = []
213 for i in range(n_chunks):
214 s = i * chunk_len
215 e = min(s + chunk_len, total)
216 seqs.append(torch.tensor(list(raw_bytes[s:e]), dtype=torch.long))
217
218 self.adapt_on_batch(seqs, device=device)

6__inline__ __device__ float warpReduceMax(float val) {
7 for (int offset = 16; offset > 0; offset /= 2)
8 val = fmaxf(val, __shfl_down_sync(0xFFFFFFFF, val, offset));
9 return val;
10}

12__inline__ __device__ float warpReduceSum(float val) {
13 for (int offset = 16; offset > 0; offset /= 2)
14 val += __shfl_down_sync(0xFFFFFFFF, val, offset);
15 return val;
16}

18// Inclusive Scan
19__inline__ __device__ float warpScanSum(float val) {
20 float temp = 0.0f;
21 #pragma unroll
22 for (int offset = 1; offset <= 16; offset *= 2) {
23 temp = __shfl_up_sync(0xFFFFFFFF, val, offset);
24 if (threadIdx.x % 32 >= offset) val += temp;
25 }
26 return val;
27}

29__global__ void ker_rc_encode_chunk_parallel_warp(const float* logits_base, const int* tokens_base, const int* lengths,
30 RCState* states, unsigned int* out_bufs,
31 int pitch_words, int vocab_size, int batch_size,
32 int chunk_len, int max_len, int start_t, int logit_stride) {
33 int b = blockIdx.x; // One block (warp) per batch sequence
34 if (b >= batch_size) return;
35
36 int lane = threadIdx.x; // 0..31
37
38 // Load State into registers
39 RCState st;
40 if (lane == 0) st = states[b];
41 // Sync to ensure lane 0 loaded it? No, other lanes don't need it yet.
42
43 int len = lengths[b];
44 int valid_len = (len > max_len) ? max_len : len;
45
46 // Pointers for this batch
47 const float* my_logits = logits_base + b * logit_stride; // logit_stride is usually chunk_len * vocab
48 const int* my_tokens = tokens_base + b * chunk_len; // Tokens are [Batch, Length] usually, or flat?
49 // boa.cpp passes d_batch_data[buf] which is [Batch * Chunk]. So stride is Chunk.
50
51 // Output pointer
52 unsigned int* my_out = out_bufs + (size_t)b * pitch_words;
53
54 // Shared Memory for CDF
55 // We need 257 elements for CDF.
56 // Probs can reuse.
57 __shared__ volatile float s_cdf[260]; // padded
58
59 // Loop over time steps
60 for (int t = start_t; t < valid_len; ++t) {
61 int token = my_tokens[t];
62 const float* l_ptr = my_logits + t * vocab_size; // 256 floats
63
64 // 1. Load & Max Reduction
65 // Each thread loads 8 elements (256 / 32 = 8).
66 float my_vals[8];
67 float my_max = -1e30f;
68
69 #pragma unroll
70 for (int i = 0; i < 8; ++i) {
71 int idx = lane * 8 + i;
72 float val = l_ptr[idx];
73 my_vals[i] = val; // Store for next step
74 my_max = fmaxf(my_max, val);
75 }
76 float batch_max = warpReduceMax(my_max);
77 batch_max = __shfl_sync(0xFFFFFFFF, batch_max, 0); // Broadcast max
78
79 // 2. Exp & Sum
80 float my_sum = 0.0f;
81 #pragma unroll
82 for (int i = 0; i < 8; ++i) {
83 float val = gpu_exp_lut(my_vals[i] - batch_max);
84 my_vals[i] = val; // Store exp'd val
85 my_sum += val;
86 }
87 float batch_sum = warpReduceSum(my_sum);
88 batch_sum = __shfl_sync(0xFFFFFFFF, batch_sum, 0); // Broadcast sum
89
90 float inv_sum = 1.0f / batch_sum;
91
92 // 3. Normalize & Local CDF
93 float local_cdf = 0.0f;
94
95 // This is tricky. We need global Scan.
96 // Each thread has 8 elements.
97 // Let's first sum our 8 elements to get "thread total".
98 float thread_sum = my_sum * inv_sum;
99
100 // Prefix scan of thread totals across warp
101 float warp_prefix = warpScanSum(thread_sum);
102 float warp_start = warp_prefix - thread_sum; // Exclusive scan result for this thread's block
103
104 // Now compute CDF for 8 elements and store to shared
105 float running = warp_start;
106 #pragma unroll
107 for (int i = 0; i < 8; ++i) {
108 int idx = lane * 8 + i;
109 // s_cdf[idx] SHOULD be exclusive sum (starts at 0).
110 // But we can store Probs or CDF?
111 // Standard RC needs: cdf[target] (low) and cdf[target+1] (high).
112 // So s_cdf[k] = sum(p_0...p_{k-1}).
113
114 // Convert to int scale on fly? Or keep float?
115 // "build_cdf_fast" logic converts to int scale.
116 // Let's follow "build_cdf_fast" logic:
117 // scale = (double)free_weight / norm. float is fine.
118 // cdf[i] = (cumulative * scale).
119
120 // Let's store Normalized Float CDF in shared.
121 // s_cdf[0] = 0.
122 // s_cdf[idx] = running.
123 // BUT idx=0 must be 0.
124
125 s_cdf[idx] = running;
126 running += my_vals[i] * inv_sum; // Add prob
127 }
128 // Last element total
129 if (lane == 31) s_cdf[256] = 1.0f; // Close to 1.0
130
131 // 4. Encode (Lane 0 only)
132 if (lane == 0) {
133 // Replicate rc_encode_symbol_probs logic but with shared memory CDF
134 // We need to Convert Float CDF to Int CDF 16-bit-ish scale?
135 // ker_rc_encode uses "build_cdf_fast" which maps float to uint.
136 /*
137 double scale = (double)free_weight / norm;
138 unsigned int left = (unsigned int)(cumulative_float * scale) + accumulated_slack;
139 */
140 // We can do this map here for just the target symbol!
141 // We don't need full Int CDF. Just Target Low/High.
142
143 float prob_low = s_cdf[token];
144 float prob_high = s_cdf[token+1];
145 // Fix edge case for token=255 -> s_cdf[256] needed.
146 if (token == 255) prob_high = 1.0f;
147
148 // Convert to Int Scale (2^16-ish?)
149 // Actually existing logic uses ~2^24 or something.
150 // build_cdf_fast: free_weight = TOTAL - vocab(256) - slack...
151 // TOTAL = 1<<16?
152 // In exact_math.hpp? #define TOTAL (1<<PRECISION)? No.
153 // Let's check build_cdf_fast.
154
155 const unsigned int TOTAL = (1<<16); // Check this constant!
156 // Assume 16 bit total for now as standard RC.
157
158 unsigned int low_int = (unsigned int)(prob_low * (TOTAL - 256)) + token;
159 unsigned int high_int = (unsigned int)(prob_high * (TOTAL - 256)) + (token + 1);
160
161 // Verify:
162 // For i=0: low=0.
163 // For i=256: high = (1.0 * (T-256)) + 256 = T. Correct.
164
165 unsigned int left = low_int;
166 unsigned int prob = high_int - low_int;
167
168 // Standard RC Encode Step
169 unsigned long long scale_rc = st.range >> 16; // PRECISION?
170 unsigned long long old_lower = st.lower;
171
172 st.range = scale_rc * (unsigned long long)prob;
173 st.lower = st.lower + scale_rc * (unsigned long long)left;
174
175 // Renormalization Loop
176 if (st.inverted_num > 0) {
177 unsigned long long sum = st.lower + st.range;
178 if (sum > st.lower) {
179 unsigned int first_word = (st.lower < old_lower) ? st.first_inv_lower_word + 1u : st.first_inv_lower_word;
180 unsigned int subsequent = (st.lower < old_lower) ? 0u : 0xFFFFFFFFu;
181 out_bufs[(size_t)b * pitch_words + st.write_idx_words++] = first_word;
182 for (int k=1; k<st.inverted_num; k++) out_bufs[(size_t)b * pitch_words + st.write_idx_words++] = subsequent;
183 st.inverted_num = 0;
184 }
185 }
186 while (st.range < (1ull << (64-32))) {
187 unsigned int lower_word = (unsigned int)(st.lower >> (64-32));
188 st.lower <<= 32; st.range <<= 32;
189 if (st.inverted_num > 0) {
190 st.inverted_num++;
191 } else {
192 unsigned long long sum = st.lower + st.range;
193 if (sum > st.lower) {
194 out_bufs[(size_t)b * pitch_words + st.write_idx_words++] = lower_word;
195 } else {
196 st.inverted_num = 1;
197 st.first_inv_lower_word = lower_word;
198 }
199 }
200 }
201 }
202 // Need sync? Lane 0 is independent now until next iteration Load.
203 // But next iteration overwrites s_cdf? Yes.
204 // Warp threads are implicitly synced?
205 // Safer to sync before overwriting shared.
206 // Or if we use __shfl based logic for next iter, we are fine.
207 // But s_cdf is written by all.
208 // Lane 0 reads s_cdf[token].
209 // If token corresponds to lane X, lane X might overwrite it next iter.
210 // So we need sync.
211 // __syncwarp();
212 }
213
214 // Save state
215 if (lane == 0) states[b] = st;
216}

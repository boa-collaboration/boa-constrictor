15// Helper: read fp16 values from file and convert to fp32
16inline void read_fp16_to_fp32(std::ifstream& f, std::vector<float>& dst, int count) {
17 std::vector<uint16_t> tmp16(count);
18 f.read(reinterpret_cast<char*>(tmp16.data()), count * sizeof(uint16_t));
19 for (int i = 0; i < count; ++i) {
20 // IEEE 754 half-precision to single-precision conversion
21 uint16_t h = tmp16[i];
22 uint32_t sign = (uint32_t)(h >> 15) << 31;
23 uint32_t exp = (h >> 10) & 0x1F;
24 uint32_t mant = h & 0x3FF;
25 uint32_t f32;
26 if (exp == 0) {
27 if (mant == 0) {
28 f32 = sign; // +/- zero
29 } else {
30 // Denormalized: convert to normalized fp32
31 exp = 1;
32 while (!(mant & 0x400)) { mant <<= 1; exp--; }
33 mant &= 0x3FF;
34 f32 = sign | ((127 - 15 + exp) << 23) | (mant << 13);
35 }
36 } else if (exp == 31) {
37 f32 = sign | 0x7F800000 | (mant << 13); // Inf/NaN
38 } else {
39 f32 = sign | ((exp - 15 + 127) << 23) | (mant << 13);
40 }
41 std::memcpy(&dst[i], &f32, 4);
42 }
43}

45// Helper for transposing weights (Out, In) -> (In, Out) for RowMajor GEMM
46std::vector<float> transpose_weights(const std::vector<float>& w, int rows, int cols) {
47 std::vector<float> t(w.size());
48 for(int i=0; i<rows; ++i) {
49 for(int j=0; j<cols; ++j) {
50 t[j*rows + i] = w[i*cols + j];
51 }
52 }
53 return t;
54}

227 void step_batch(float* x_in, float* x_out) {
228 // 1. In Proj (fused GEMM+bias)
229 gemm_gpu_batch_bias(x_in, in_proj_w, in_proj_b, buf_xz, batch_size, 2*config.d_inner, config.d_model);
230
231 // 2. Optimized Conv1D (Direct from buf_xz)
232 gpu_mamba_conv1d_batch(buf_xz, conv_state, conv1d_w, conv1d_b, buf_conv, config.d_inner, config.d_conv, 2*config.d_inner, batch_size);
233
234 // 3. x_proj
235 int stride_delta = config.dt_rank + 2*config.d_state;
236 gemm_gpu_batch(buf_conv, x_proj_w, buf_delta, batch_size, stride_delta, config.d_inner);
237
238 // 4. dt_proj (Strided access from buf_delta, skip copy)
239 gemm_gpu_batch_strided(buf_delta, stride_delta, dt_proj_w, config.d_inner, buf_dt, config.d_inner, batch_size, config.d_inner, config.dt_rank);
240
241 // 5. Fused SSM Tail: Softplus(logits) + SSM(x) + Gating(z)
242 // Passes buf_conv for x, buf_xz directly for z extraction.
243 gpu_mamba_ssm_fused_batch(buf_conv, buf_xz, buf_dt, dt_proj_b, A_log, D, buf_delta, ssm_state, buf_y, config.d_inner, config.d_state, config.dt_rank, batch_size);
244
245 // 6. Out Proj (fused GEMM+bias)
246 gemm_gpu_batch_bias(buf_y, out_proj_w, out_proj_b, x_out, batch_size, config.d_model, config.d_inner);
247 }

298 void forward_chunk(float* x_in, float* x_out, int length) {
299 int batch_eff = batch_size * length;
300
301 // 1. In Proj (fused GEMM+bias)
302 gemm_gpu_batch_bias(x_in, in_proj_w, in_proj_b, chunk_buf_xz, batch_eff, 2*config.d_inner, config.d_model);
303
304 // 2. Conv1d (Strided x input)
305 gpu_mamba_conv1d_chunk(chunk_buf_xz, conv_state, conv1d_w, conv1d_b, chunk_buf_conv, batch_size, length, config.d_inner, config.d_conv, 2*config.d_inner);
306
307 // 3. Delta Proj
308 int stride_delta = config.dt_rank + 2*config.d_state;
309 gemm_gpu_batch(chunk_buf_conv, x_proj_w, chunk_buf_delta, batch_eff, stride_delta, config.d_inner);
310
311 // Split Delta
312 gpu_copy_strided(chunk_buf_delta, chunk_buf_dt_in, stride_delta, config.dt_rank, batch_eff);
313 gpu_copy_strided(chunk_buf_delta + config.dt_rank, chunk_buf_B, stride_delta, config.d_state, batch_eff);
314 gpu_copy_strided(chunk_buf_delta + config.dt_rank + config.d_state, chunk_buf_C, stride_delta, config.d_state, batch_eff);
315
316 // 4. dt Proj
317 gemm_gpu_batch(chunk_buf_dt_in, dt_proj_w, chunk_buf_dt, batch_eff, config.d_inner, config.dt_rank);
318 gpu_add_bias_softplus_batch(chunk_buf_dt, dt_proj_b, config.d_inner, batch_eff);
319
320 // 5. SSM
321 gpu_mamba_ssm_chunk(chunk_buf_conv, chunk_buf_dt, A_log, D,
322 chunk_buf_B, chunk_buf_C, ssm_state, chunk_buf_y,
323 batch_size, length, config.d_inner, config.d_state);
324
325 // 6. Gate (Strided z)
326 gpu_gate_strided(chunk_buf_y, chunk_buf_xz + config.d_inner, config.d_inner, 2*config.d_inner, batch_eff * config.d_inner);
327
328 // 7. Out Proj (fused GEMM+bias)
329 gemm_gpu_batch_bias(chunk_buf_y, out_proj_w, out_proj_b, x_out, batch_eff, config.d_model, config.d_inner);
330 }

456 void forward_chunk(float* x_in, float* x_out, int length) {
457 int batch_eff = batch_size * length;
458 int d4 = 4 * d_model;
459
460 // 1. Pre-compute ih for all timesteps: [B*L, 4d] (fused GEMM+bias)
461 gemm_gpu_batch_bias(x_in, w_ih, b_ih, chunk_buf_ih, batch_eff, d4, d_model);
462
463 // 2. Sequential loop over timesteps
464 for (int t = 0; t < length; ++t) {
465 // hh = h @ W_hh^T + b_hh for current timestep (fused GEMM+bias)
466 gemm_gpu_batch_bias(h_state, w_hh, b_hh, buf_hh, batch_size, d4, d_model);
467
468 // Extract ih for timestep t: pointer into pre-computed buffer
469 // Layout: [b0_t0, b0_t1, ..., b0_tL-1, b1_t0, ...]
470 // We need ih[b*L+t] for each b. This is strided access.
471 // Use a gather kernel or loop with per-batch offsets.
472 // Simpler: compute ih per-step too (less memory but more compute)
473 // Actually, with layout [B*L, 4d] where B*L is contiguous batches,
474 // timestep t for batch b is at index (b*length + t).
475 // We need a contiguous [B, 4d] slice. Use scatter_timestep in reverse.
476 // For simplicity, extract to buf_ih using a strided copy.
477
478 // Gather ih[b*L+t, :] for all b into buf_ih[b, :]
479 for (int b_idx = 0; b_idx < batch_size; ++b_idx) {
480 size_t src_off = ((size_t)b_idx * length + t) * d4;
481 size_t dst_off = (size_t)b_idx * d4;
482 checkCudaErrors(cudaMemcpyAsync(
483 (char*)buf_ih + dst_off * sizeof(float),
484 (char*)chunk_buf_ih + src_off * sizeof(float),
485 d4 * sizeof(float), cudaMemcpyDeviceToDevice, g_stream));
486 }
487
488 // Fused gates
489 gpu_lstm_fused_batch(buf_ih, buf_hh, h_state, c_state, d_model, batch_size);
490
491 // Scatter h_state to output: out[b*L+t, :] = h[b, :]
492 gpu_scatter_timestep(h_state, x_out, batch_size, length, t, d_model);
493 }
494 }

596 gpu_gru_fused_batch(buf_ih, buf_hh, h_state, d_model, batch_size);

699struct FeedForwardGPU {

711 malloc_device(&buf, (size_t)batch * d_hidden * sizeof(float));
712 }
713 void free() {
714 free_device(w1); free_device(b1); free_device(w2); free_device(b2); free_device(buf);
715 }
716 void load_weights(std::ifstream& f) {
717 load_vec(f, w1, d_hidden * d_model, true, d_hidden, d_model);
718 load_vec(f, b1, d_hidden);
719 load_vec(f, w2, d_model * d_hidden, true, d_model, d_hidden);
720 load_vec(f, b2, d_model);
721 }
722
723 // Chunk Logic
724 float* chunk_buf;
725 int max_alloc = 0;
726
727 void allocate_chunk(int length) {
728 if (max_alloc >= length) return;
729 if (max_alloc > 0) free_device(chunk_buf);
730 max_alloc = length;
731 size_t total = (size_t)batch_size * length;
732 malloc_device(&chunk_buf, total * d_hidden * sizeof(float));
733 }
734 void free_chunk() {
735 if (max_alloc > 0) free_device(chunk_buf);
736 max_alloc = 0;
737 }
738
739 void forward_chunk(float* x, float* out, int length, const float* residual = nullptr) {
740 int batch_eff = batch_size * length;
741 // Fused GEMM + bias + GELU (single kernel)
742 gemm_gpu_batch_bias_act(x, w1, b1, chunk_buf, batch_eff, d_hidden, d_model, GEMM_ACT_GELU);
743
744 if (residual) {
745 // Fused GEMM + bias + residual add (single kernel)
746 gemm_gpu_batch_bias_res(chunk_buf, w2, b2, residual, out, batch_eff, d_model, d_hidden);
747 } else {
748 gemm_gpu_batch_bias(chunk_buf, w2, b2, out, batch_eff, d_model, d_hidden);
749 }
750 }
751
752 void step_batch(float* x, float* out, const float* residual = nullptr) {
753 // x: [batch, d_model], w1: [d_model, d_hidden], buf: [batch, d_hidden]
754 // Fused GEMM + bias + GELU (single kernel)
755 gemm_gpu_batch_bias_act(x, w1, b1, buf, batch_size, d_hidden, d_model, GEMM_ACT_GELU);
756
757 // buf: [batch, d_hidden], w2: [d_hidden, d_model], out: [batch, d_model]
758 if (residual) {
759 // Fused GEMM + bias + residual add (single kernel)
760 gemm_gpu_batch_bias_res(buf, w2, b2, residual, out, batch_size, d_model, d_hidden);
761 } else {
762 gemm_gpu_batch_bias(buf, w2, b2, out, batch_size, d_model, d_hidden);
763 }
764 }

834 void step_batch(float* x_in, float* x_out, float* buf_res) {
835 if (backbone_type == BACKBONE_MAMBA) {
836 // Mamba V1: single residual — x + ff(ln2(mamba(ln1(x))))
837 // Fused: save residual + layernorm in one kernel (eliminates D2D memcpy)

840 mamba.step_batch(x_in, x_in);
841 gpu_layernorm_batch(x_in, ln2.weight, ln2.bias, d_model, batch_size);
842 // Fused: FF + residual add (residual folded into w2 GEMM store)
843 ff.step_batch(x_in, x_in, buf_res);
844 } else {
845 // RNN backbones: dual residual — x = x + backbone(ln1(x)); x = x + ff(ln2(x))
846 // Fused: save residual + layernorm in one kernel
847 gpu_layernorm_save_batch(x_in, buf_res, ln1.weight, ln1.bias, d_model, batch_size);

849 switch(backbone_type) {

851 case BACKBONE_GRU: gru.step_batch(x_in, x_in); break;
852 case BACKBONE_MINGRU: mingru.step_batch(x_in, x_in); break;
853 default: break;
854 }
855
856 // First residual: x = backbone_out + original_x
857 gpu_add_batch(x_in, buf_res, d_model, batch_size);
858
859 // Second residual: x = x + ff(ln2(x))
860 // Fused: save residual + layernorm in one kernel
861 gpu_layernorm_save_batch(x_in, buf_res, ln2.weight, ln2.bias, d_model, batch_size);
862 // Fused: FF + residual add (residual folded into w2 GEMM store)
863 ff.step_batch(x_in, x_in, buf_res);
864 }

866 if (x_in != x_out) {
867 checkCudaErrors(cudaMemcpyAsync(x_out, x_in, (size_t)batch_size * d_model * sizeof(float), cudaMemcpyDeviceToDevice, g_stream));

1006 }

1010 if (g_model_fp16) {
1011 read_fp16_to_fp32(f, h_emb, (int)embedding_size);

1036 gpu_embedding_lookup_batch(d_tokens, embedding, buf_x, config.d_model, batch_size);
1037
1038 // 2. Blocks
1039 for(int i=0; i<n_layers; ++i) {
1040 blocks[i].step_batch(buf_x, buf_x, buf_res);
1041 }
1042
1043 // 3. Final norm (RMSNorm for non-mamba backbones)
1044 if (has_final_norm) {
1045 gpu_rmsnorm_batch(buf_x, final_norm_w, config.d_model, batch_size);
1046 }
1047
1048 // 4. Head: Linear -> Activation -> Linear (fused GEMM+bias+activation)
1049 int head_act = has_final_norm ? GEMM_ACT_SILU : GEMM_ACT_RELU;

1103 static size_t estimate_memory_static(MambaConfig config, int vocab_size, int n_layers, int batch_size, int chunk_len, bool is_compression) {
1104 size_t dmod = config.d_model;
1105 size_t w_ff = (4*dmod*dmod + 4*dmod + dmod*4*dmod + dmod) * 4;
1106 size_t w_ln = 2 * (2 * dmod) * 4; // ln1 + ln2
1107 size_t w_head = (dmod*dmod + dmod + vocab_size*dmod + vocab_size) * 4;
1108 size_t w_emb = (size_t)vocab_size * dmod * 4;
1109 size_t b_ff = (4*dmod) * 4;
1110 size_t b_pred = (dmod + dmod + dmod) * 4;
1111
1112 size_t w_backbone = 0, b_backbone = 0, c_backbone = 0;
1113
1114 switch(config.backbone) {
1115 case BACKBONE_MAMBA: {
1116 size_t din = config.d_inner;
1117 size_t dstate = config.d_state;
1118 size_t dconv = config.d_conv;
1119 size_t dtrank = config.dt_rank;
1120 size_t stride_delta = dtrank + 2 * dstate;
1121 w_backbone = (2*din*dmod + 2*din + din*dconv + din + stride_delta*din + din*dtrank + din + din*dstate + din + dmod*din + dmod) * 4;
1122 b_backbone = (din*dconv + din*dstate + 2*din + din + din + stride_delta + dtrank + din + din) * 4;
1123 if (is_compression) c_backbone = (2*din + din + stride_delta + dtrank + din + din + 2*dstate) * 4;
1124 break;
1125 }
1126 case BACKBONE_LSTM:
1127 w_backbone = (4*dmod*dmod + 4*dmod + 4*dmod*dmod + 4*dmod) * 4; // w_ih, b_ih, w_hh, b_hh
1128 b_backbone = (dmod + dmod + 4*dmod + 4*dmod) * 4; // h, c, buf_ih, buf_hh
1129 if (is_compression) c_backbone = (4*dmod + dmod) * 4; // chunk_buf_ih, chunk_buf_out
1130 break;
1131 case BACKBONE_GRU:
1132 w_backbone = (3*dmod*dmod + 3*dmod + 3*dmod*dmod + 3*dmod) * 4;
1133 b_backbone = (dmod + 3*dmod + 3*dmod) * 4; // h, buf_ih, buf_hh
1134 if (is_compression) c_backbone = (3*dmod + dmod) * 4;
1135 break;
1136 case BACKBONE_MINGRU:
1137 w_backbone = (dmod*dmod + dmod + dmod*dmod + dmod) * 4; // w_z, b_z, w_h, b_h
1138 b_backbone = (dmod + dmod + dmod) * 4; // h, buf_z, buf_h
1139 if (is_compression) c_backbone = (dmod + dmod) * 4; // chunk_buf_z, chunk_buf_h
1140 break;
1141 }
1142
1143 size_t fixed = (w_backbone + w_ff + w_ln) * n_layers + w_head + w_emb;
1144 size_t per_batch = (size_t)batch_size * ((b_backbone + b_ff) * n_layers + b_pred);
1145
1146 size_t per_chunk = 0;
1147 if (is_compression) {
1148 size_t c_ff = (4*dmod) * 4;
1149 size_t c_pred = (dmod + dmod + dmod) * 4;
1150 size_t c_logits = (size_t)vocab_size * 4;
1151 per_chunk = (size_t)batch_size * (size_t)chunk_len * ((c_backbone + c_ff) * n_layers + c_pred + c_logits);
1152 } else {
1153 per_chunk = (size_t)batch_size * (size_t)chunk_len * sizeof(int);
1154 }
1155
1156 return fixed + per_batch + per_chunk;
1157 }

1"""
2ATLAS jets round-trip helper
3
4This script:
5 1) Downloads the ATLAS HDF5 file from EOS (tries xrdcp, falls back to HTTPS)
6 2) Extracts the 'jets' dataset to:
7 - atlas.npz (compressed npz)
8 - atlas.bin (raw bytes) + atlas.meta.json (shape/dtype metadata)
9 3) Reconstructs an HDF5 file from a decompressed atlas.bin (+ meta)
10 4) Compares the original and reconstructed HDF5 for equality of 'jets'
11
12Example end-to-end:
13 python download.py --all-steps \
14 --src root://eospublic.cern.ch//eos/opendata/atlas/datascience/ATLAS-FTAG-2023-05/mc-flavtag-ttbar-large.h5
15
16If you already have a decompressed 'atlas.bin' from an external compressor, you can run:
17 python download.py --reconstruct --compare --bin atlas.bin --meta atlas.meta.json
18"""

54def download_atlas_h5(src: str, dst: str) -> None:
55 if os.path.exists(dst):
56 print(f"[download] Already present: {dst}")
57 return
58
59 os.makedirs(os.path.dirname(dst) or ".", exist_ok=True)
60
61 xrdcp = shutil.which("xrdcp")
62 if src.startswith("root://") and xrdcp is not None:
63 # Use xrdcp when available
64 import subprocess
65
66 print(f"[download] Using xrdcp to fetch {src} -> {dst}")
67 subprocess.check_call([xrdcp, "-f", src, dst])
68 return
69
70 # Fallback to HTTPS
71 https_url = root_to_https(src)
72 print(f"[download] Fetching via HTTPS: {https_url} -> {dst}")
73 with urllib.request.urlopen(https_url) as r, open(dst, "wb") as f:
74 # Stream with a basic progress indicator
75 total = int(r.headers.get("Content-Length", 0))
76 chunk = 1 << 20
77 downloaded = 0
78 while True:
79 buf = r.read(chunk)
80 if not buf:
81 break
82 f.write(buf)
83 downloaded += len(buf)
84 if total:
85 pct = 100.0 * downloaded / total
86 print(f"\r[download] {downloaded}/{total} bytes ({pct:.1f}%)", end="")
87 print()

96def save_bin(out_bin: str) -> None:
97 print(f"[bin] Writing {out_bin}")
98 meta_path = out_bin.replace(".bin", ".meta.json")
99 # Stream 'jets' dataset from atlas.h5 to a raw binary file (row-major)
100 with h5py.File("atlas.h5", "r") as h5:
101 if "jets" not in h5:
102 raise KeyError("Dataset 'jets' not found in atlas.h5")
103 dset = h5["jets"]
104 shape = tuple(dset.shape)
105 dtype = dset.dtype
106 dtype_descr = dtype.descr # preserves structured field layout
107 # Write raw bytes in chunks to avoid excessive memory usage
108 with open(out_bin, "wb") as f:
109 chunk_rows = max(1, 1 << 12) # ~4096 rows per chunk
110 n_rows = dset.shape[0] if dset.ndim >= 1 else 1
111 for sl in iter_slices(n_rows, chunk_rows):
112 chunk = np.ascontiguousarray(dset[sl])
113 f.write(chunk.tobytes(order="C"))
114 # Save sidecar metadata to reconstruct shape/dtype
115 with open(meta_path, "w", encoding="utf-8") as m:
116 json.dump({"shape": shape, "dtype": dtype.str, "dtype_descr": dtype_descr, "order": "C"}, m)

120def reconstruct_h5_from_bin(bin_path: str, out_h5: str) -> None:
121 # Load metadata for dtype/shape
122 meta_path = bin_path.replace(".bin", ".meta.json")
123 with open(meta_path, "r", encoding="utf-8") as m:
124 meta = json.load(m)
125 # Prefer structured dtype description when available; coerce to list-of-tuples
126 def _coerce_dtype_descr(descr_obj):
127 import ast as _ast
128 if isinstance(descr_obj, str):
129 # Sometimes stored as a string; parse safely
130 descr_obj = _ast.literal_eval(descr_obj)
131 if isinstance(descr_obj, (list, tuple)):
132 coerced = []
133 for elem in descr_obj:
134 if isinstance(elem, (list, tuple)):
135 coerced.append(tuple(elem))
136 elif isinstance(elem, str):
137 coerced.append(tuple(_ast.literal_eval(elem)))
138 else:
139 raise TypeError(f"Unsupported dtype_descr element type: {type(elem)}")
140 return coerced
141 raise TypeError(f"Unsupported dtype_descr type: {type(descr_obj)}")
142
143 if "dtype_descr" in meta:
144 dtype = np.dtype(_coerce_dtype_descr(meta["dtype_descr"]))
145 else:
146 dtype = np.dtype(meta["dtype"])
147 shape = tuple(meta["shape"])
148 count = int(np.prod(shape))
149
150 # Reconstruct array from raw bytes
151 arr = np.fromfile(bin_path, dtype=dtype, count=count).reshape(shape)
152
153 # Write to HDF5 as dataset 'jets'
154 print(f"[reconstruct] Writing {out_h5}")
155 os.makedirs(os.path.dirname(out_h5) or ".", exist_ok=True)
156 with h5py.File(out_h5, "w") as f:
157 f.create_dataset("jets", data=arr, chunks=True, compression="gzip", compression_opts=4)

160def iter_slices(n_rows: int, chunk_rows: int) -> Iterable[slice]:
161 for start in range(0, n_rows, chunk_rows):
162 end = min(n_rows, start + chunk_rows)
163 yield slice(start, end)

166def compare_h5_jets(a_path: str, b_path: str, atol: float = 0.0, rtol: float = 0.0) -> bool:
167 """Compare dataset 'jets' between two HDF5 files. Returns True if equal."""
168 with h5py.File(a_path, "r") as fa, h5py.File(b_path, "r") as fb:
169 if "jets" not in fa or "jets" not in fb:
170 raise KeyError("Both files must contain a 'jets' dataset")
171 da, db = fa["jets"], fb["jets"]
172 if da.shape != db.shape or da.dtype != db.dtype:
173 print("[compare] Shape or dtype mismatch:", da.shape, da.dtype, "vs", db.shape, db.dtype)
174 return False
175
176 # Chunked comparison along first axis to avoid high memory
177 n_rows = da.shape[0] if da.ndim >= 1 else 1
178 chunk_rows = max(1, 1 << 12) # ~4096 rows per chunk
179 total_chunks = math.ceil(n_rows / chunk_rows)
180 for i, sl in enumerate(iter_slices(n_rows, chunk_rows), 1):
181 a = da[sl]
182 b = db[sl]
183 # Handle structured vs plain numeric arrays
184 if a.dtype.fields: # structured dtype
185 for name, (fld_dtype, _) in a.dtype.fields.items():
186 av = a[name]
187 bv = b[name]
188 kind = np.dtype(fld_dtype).kind
189 if kind in ("f",):
190 eq = np.allclose(av, bv, atol=atol, rtol=rtol, equal_nan=True)
191 else:
192 eq = np.array_equal(av, bv)
193 if not eq:
194 print(
195 f"[compare] Field '{name}' mismatch in rows {sl.start}:{sl.stop} (chunk {i}/{total_chunks})"
196 )
197 return False
198 else:
199 if not np.allclose(a, b, atol=atol, rtol=rtol, equal_nan=True):
200 print(f"[compare] Mismatch found in rows {sl.start}:{sl.stop} (chunk {i}/{total_chunks})")
201 return False
202 if i % 50 == 0 or i == total_chunks:
203 print(f"[compare] {i}/{total_chunks} chunks OK")
204 print("[compare] All chunks equal")
205 return True

207def save_200m(out_bin: str) -> None:
208 limit_bytes = 200 * 1024 * 1024
209 base, ext = os.path.splitext("atlas.bin")
210 trunc_path = f"{base}_200m{ext}"
211 bufsize = 1024 * 1024
212 remaining = limit_bytes
213 with open("atlas.bin", "rb") as src, open(out_bin, "wb") as dst:
214 while remaining > 0:
215 chunk = src.read(min(bufsize, remaining))
216 if not chunk:
217 break
218 dst.write(chunk)
219 remaining -= len(chunk)

222def extract_colmajor(h5_path: str, meta_path: str, out_bin: str, out_200m: str) -> None:
223 """Extract column-major binary from ATLAS HDF5 jets dataset.
224
225 Each field's values are written contiguously across all jets, giving a
226 column-oriented layout. Also creates a 200 MB training chunk.
227 """
228 with open(meta_path) as f:
229 meta = json.load(f)
230 fields = meta["dtype_descr"]
231 n_jets = meta["shape"][0]
232
233 print(f"[col-major] ATLAS jets: {n_jets:,} jets, {len(fields)} fields")
234
235 with h5py.File(h5_path, "r") as hf:
236 ds = hf["jets"]
237 total_bytes = 0
238 with open(out_bin, "wb") as f_out:
239 for i, (name, dt) in enumerate(fields):
240 col = np.array(ds[name])
241 raw = col.tobytes()
242 f_out.write(raw)
243 total_bytes += len(raw)
244 print(f" [{i:2d}] {name:45s} {dt} {col.dtype.itemsize}B × {len(col):,} = {len(raw):,} bytes")
245
246 print(f"[col-major] Wrote {out_bin}: {total_bytes:,} bytes ({total_bytes/1e6:.1f} MB)")
247
248 # Create shuffled 200 MB training sample
249 rng = np.random.default_rng(42)
250 bytes_per_jet = sum(np.dtype(dt).itemsize for _, dt in fields)
251 target_jets = 200_000_000 // bytes_per_jet
252 indices = rng.permutation(n_jets)[:target_jets]
253
254 with h5py.File(h5_path, "r") as hf:
255 ds = hf["jets"]
256 with open(out_200m, "wb") as f_out:
257 for name, dt in fields:
258 col = np.array(ds[name])[indices]
259 f_out.write(col.tobytes())
260
261 print(f"[col-major] Training chunk: {out_200m} ({os.path.getsize(out_200m)/1048576:.1f} MB)")

264def main(argv: list[str] | None = None) -> int:

272 p.add_argument("--download", action="store_true", help="Download the HDF5 if missing")

280 if args.all_steps:

42def download_file(url, local_filename, expected_size=None):
43 """
44 Download a file from URL with progress bar
45
46 Args:
47 url: URL to download from
48 local_filename: Local path to save file
49 expected_size: Expected file size in bytes (for progress bar)
50 """
51 print(f"Downloading from: {url}")
52 print(f"Saving to: {local_filename}")
53
54 try:
55 response = requests.get(url, stream=True, timeout=30)
56 response.raise_for_status()
57
58 total_size = int(response.headers.get('content-length', 0))
59 if total_size == 0 and expected_size:
60 total_size = expected_size
61
62 block_size = 8192
63
64 with open(local_filename, 'wb') as f:
65 with tqdm(total=total_size, unit='B', unit_scale=True, desc=os.path.basename(local_filename)) as pbar:
66 for chunk in response.iter_content(chunk_size=block_size):
67 if chunk:
68 f.write(chunk)
69 pbar.update(len(chunk))
70
71 print(f"✓ Downloaded successfully: {os.path.getsize(local_filename) / (1024**2):.2f} MB")
72 return True
73
74 except requests.exceptions.RequestException as e:
75 print(f"✗ Download failed: {e}")
76 if os.path.exists(local_filename):
77 os.remove(local_filename)
78 return False

80def extract_camel_data_from_hdf5(hdf5_file, output_bin, target_size_mb=1024):
81 """
82 Extract particle data from CAMEL HDF5 snapshot and save as binary
83
84 Args:
85 hdf5_file: Path to HDF5 snapshot file
86 output_bin: Output binary file path
87 target_size_mb: Target size in MB
88 """
89 print(f"\nExtracting data from {hdf5_file}...")
90
91 with h5py.File(hdf5_file, 'r') as f:
92 print("\nAvailable particle types:")
93 for key in f.keys():
94 print(f" - {key}")
95
96 # CAMEL snapshots contain different particle types:
97 # PartType0: Gas particles
98 # PartType1: Dark matter
99 # PartType4: Stars
100 # PartType5: Black holes
101
102 # Focus on gas particles (PartType0) for astrophysical data
103 if 'PartType0' not in f:
104 print("Error: No gas particles (PartType0) found in snapshot")
105 return None
106
107 gas = f['PartType0']
108 print(f"\nGas particle datasets:")
109 for key in gas.keys():
110 if isinstance(gas[key], h5py.Dataset):
111 print(f" - {key}: shape={gas[key].shape}, dtype={gas[key].dtype}")
112
113 # Key fields to extract:
114 # - Coordinates: 3D positions
115 # - Velocities: 3D velocities
116 # - Density: gas density
117 # - InternalEnergy: thermal energy
118 # - Masses: particle masses
119 # - ElectronAbundance: ionization state
120 # - Metallicity: metal content
121
122 fields_to_extract = []
123 field_names = []
124
125 # Coordinates (3 components)
126 if 'Coordinates' in gas:
127 coords = gas['Coordinates'][:]
128 fields_to_extract.append(coords)
129 field_names.extend(['x', 'y', 'z'])
130 print(f"✓ Loaded Coordinates: {coords.shape}")
131
132 # Velocities (3 components)
133 if 'Velocities' in gas:
134 vels = gas['Velocities'][:]
135 fields_to_extract.append(vels)
136 field_names.extend(['vx', 'vy', 'vz'])
137 print(f"✓ Loaded Velocities: {vels.shape}")
138
139 # Density (scalar)
140 if 'Density' in gas:
141 density = gas['Density'][:].reshape(-1, 1)
142 fields_to_extract.append(density)
143 field_names.append('density')
144 print(f"✓ Loaded Density: {density.shape}")
145
146 # Masses (scalar)
147 if 'Masses' in gas:
148 masses = gas['Masses'][:].reshape(-1, 1)
149 fields_to_extract.append(masses)
150 field_names.append('mass')
151 print(f"✓ Loaded Masses: {masses.shape}")
152
153 # Internal Energy / Temperature (scalar)
154 if 'InternalEnergy' in gas:
155 energy = gas['InternalEnergy'][:].reshape(-1, 1)
156 fields_to_extract.append(energy)
157 field_names.append('internal_energy')
158 print(f"✓ Loaded InternalEnergy: {energy.shape}")
159
160 # ElectronAbundance (scalar)
161 if 'ElectronAbundance' in gas:
162 electron = gas['ElectronAbundance'][:].reshape(-1, 1)
163 fields_to_extract.append(electron)
164 field_names.append('electron_abundance')
165 print(f"✓ Loaded ElectronAbundance: {electron.shape}")
166
167 # Metallicity (can be scalar or array)
168 if 'Metallicity' in gas:
169 metallicity = gas['Metallicity'][:]
170 if len(metallicity.shape) == 1:
171 metallicity = metallicity.reshape(-1, 1)
172 # If multi-element, take total metallicity (first column or sum)
173 if metallicity.shape[1] > 1:
174 metallicity = metallicity[:, 0:1] # Take first element
175 fields_to_extract.append(metallicity)
176 field_names.append('metallicity')
177 print(f"✓ Loaded Metallicity: {metallicity.shape}")
178
179 # Combine all fields
180 if not fields_to_extract:
181 print("Error: No valid fields found to extract")
182 return None
183
184 combined_data = np.concatenate(fields_to_extract, axis=1).astype(np.float32)
185
186 print(f"\nCombined data shape: {combined_data.shape}")
187 print(f"Fields: {field_names}")
188 print(f"Total size: {combined_data.nbytes / (1024**2):.2f} MB")
189
190 # If data is larger than target, subsample
191 target_bytes = target_size_mb * 1024 * 1024
192 if combined_data.nbytes > target_bytes:
193 n_particles_target = target_bytes // (combined_data.shape[1] * 4)
194 print(f"\nSubsampling from {combined_data.shape[0]} to {n_particles_target} particles...")
195 indices = np.random.choice(combined_data.shape[0], n_particles_target, replace=False)
196 combined_data = combined_data[indices]
197 print(f"New size: {combined_data.nbytes / (1024**2):.2f} MB")
198
199 return combined_data, field_names

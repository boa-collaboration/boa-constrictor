55class EnergyTracker:
56 """Context-manager wrapper around CPPJoules EnergyTracker.
57
58 Usage::
59
60 tracker = EnergyTracker()
61 tracker.start()
62 # ... do work ...
63 tracker.stop()
64 results = tracker.results() # dict[str, float]
65
66 Or as a context manager::
67
68 with EnergyTracker() as t:
69 # ... do work ...
70 print(t.results())
71 """
72
73 def __init__(self):
74 lib = _load_lib()
75 self._lib = lib
76 self._handle = lib.energy_tracker_new()
77 self._results = None
78
79 def start(self):
80 """Record starting energy counters."""
81 self._lib.energy_tracker_start(self._handle)
82
83 def stop(self):
84 """Record ending energy counters and compute deltas."""
85 self._lib.energy_tracker_stop(self._handle)
86 self._lib.energy_tracker_calculate(self._handle)
87 self._results = None # invalidate cache

89 def results(self) -> dict:
90 """Return energy results as ``{domain: value}`` dict.
91
92 Keys include ``'Time'`` (seconds) and energy domains like
93 ``'package-0'``, ``'core-0'``, ``'dram-0'``, ``'nvidia_gpu_0'``
94 (values in micro-joules for RAPL, milli-joules for NVML).
95 """
96 if self._results is not None:
97 return dict(self._results)
98
99 raw_ptr = self._lib.energy_tracker_get_csv(self._handle)
100 if raw_ptr is None:
101 return {}
102 text = ctypes.string_at(raw_ptr).decode("utf-8", errors="replace")
103 self._lib.energy_tracker_free_str(raw_ptr)
104
105 reader = csv.reader(io.StringIO(text))
106 rows = list(reader)
107 if len(rows) < 2:
108 return {}
109 headers = [h.strip() for h in rows[0] if h.strip()]
110 values = [v.strip() for v in rows[1] if v.strip()]
111 res = {}
112 for h, v in zip(headers, values):
113 try:
114 res[h] = float(v)
115 except ValueError:
116 res[h] = v
117 self._results = res
118 return dict(res)
119
120 def save_csv(self, path: str):
121 """Save energy results to a CSV file."""
122 self._lib.energy_tracker_save_csv(self._handle, path.encode("utf-8"))
123
124 def __enter__(self):
125 self.start()
126 return self

136 def summary_str(self) -> str:
137 """One-line summary of energy measurement results."""
138 r = self.results()
139 if not r:
140 return "No energy data"
141 time_s = r.get("Time", 0)
142 parts = [f"Time={time_s:.3f}s"]
143 for k, v in r.items():
144 if k == "Time":
145 continue
146 if isinstance(v, float):
147 # Convert micro-joules to joules for readability
148 parts.append(f"{k}={v / 1e6:.4f}J")
149 else:
150 parts.append(f"{k}={v}")
151 return " ".join(parts)

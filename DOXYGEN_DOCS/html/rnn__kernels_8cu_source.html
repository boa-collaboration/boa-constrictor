4// ==========================================
5// LSTM Fused Gate Kernel
6// ==========================================
7// After GEMM: ih = x @ W_ih^T + b_ih, hh = h @ W_hh^T + b_hh
8// Gate layout (PyTorch convention): [i, f, g, o] each of size d_model
9// i = sigmoid(ih_i + hh_i) — input gate
10// f = sigmoid(ih_f + hh_f) — forget gate
11// g = tanh(ih_g + hh_g) — cell candidate
12// o = sigmoid(ih_o + hh_o) — output gate
13// c_new = f * c_old + i * g
14// h_new = o * tanh(c_new)
15
16__global__ void ker_lstm_fused_batch(const float* ih, const float* hh,
17 float* h_state, float* c_state,
18 int d_model, size_t total) {
19 // Process 4 elements per thread when d_model is aligned
20 size_t base = ((size_t)blockIdx.x * blockDim.x + threadIdx.x) * 4;
21 if (base + 3 < total && (d_model % 4 == 0)) {
22 size_t b = base / d_model;
23 int d = base % d_model;
24 int d4 = 4 * d_model;
25 size_t off = b * d4;
26
27 // Load all 4 gates for 4 elements via float4
28 float4 ih_i = reinterpret_cast<const float4*>(&ih[off + d])[0];
29 float4 hh_i = reinterpret_cast<const float4*>(&hh[off + d])[0];
30 float4 ih_f = reinterpret_cast<const float4*>(&ih[off + d_model + d])[0];
31 float4 hh_f = reinterpret_cast<const float4*>(&hh[off + d_model + d])[0];
32 float4 ih_g = reinterpret_cast<const float4*>(&ih[off + 2*d_model + d])[0];
33 float4 hh_g = reinterpret_cast<const float4*>(&hh[off + 2*d_model + d])[0];
34 float4 ih_o = reinterpret_cast<const float4*>(&ih[off + 3*d_model + d])[0];
35 float4 hh_o = reinterpret_cast<const float4*>(&hh[off + 3*d_model + d])[0];
36 float4 c_old = reinterpret_cast<float4*>(&c_state[base])[0];
37
38 float vals_i[4] = {ih_i.x+hh_i.x, ih_i.y+hh_i.y, ih_i.z+hh_i.z, ih_i.w+hh_i.w};
39 float vals_f[4] = {ih_f.x+hh_f.x, ih_f.y+hh_f.y, ih_f.z+hh_f.z, ih_f.w+hh_f.w};
40 float vals_g[4] = {ih_g.x+hh_g.x, ih_g.y+hh_g.y, ih_g.z+hh_g.z, ih_g.w+hh_g.w};
41 float vals_o[4] = {ih_o.x+hh_o.x, ih_o.y+hh_o.y, ih_o.z+hh_o.z, ih_o.w+hh_o.w};
42 float c_arr[4] = {c_old.x, c_old.y, c_old.z, c_old.w};
43
44 float h_arr[4], c_new[4];
45 #pragma unroll
46 for (int n = 0; n < 4; ++n) {
47 #if GPU_REPRO_MATH
48 float iv = repro_sigmoid(vals_i[n]);
49 float fv = repro_sigmoid(vals_f[n]);
50 float gv = repro_tanh(vals_g[n]);
51 float ov = repro_sigmoid(vals_o[n]);
52 #else
53 float iv = 1.0f / (1.0f + expf(-vals_i[n]));
54 float fv = 1.0f / (1.0f + expf(-vals_f[n]));
55 float gv = tanhf(vals_g[n]);
56 float ov = 1.0f / (1.0f + expf(-vals_o[n]));
57 #endif
58 float c = __fadd_rn(__fmul_rn(fv, c_arr[n]), __fmul_rn(iv, gv));
59 #if GPU_REPRO_MATH
60 h_arr[n] = __fmul_rn(ov, repro_tanh(c));
61 #else
62 h_arr[n] = __fmul_rn(ov, tanhf(c));
63 #endif
64 c_new[n] = c;
65 }
66
67 reinterpret_cast<float4*>(&c_state[base])[0] = make_float4(c_new[0], c_new[1], c_new[2], c_new[3]);
68 reinterpret_cast<float4*>(&h_state[base])[0] = make_float4(h_arr[0], h_arr[1], h_arr[2], h_arr[3]);
69 } else {
70 // Scalar fallback
71 for (size_t idx = base; idx < total && idx < base + 4; ++idx) {
72 size_t b = idx / d_model;
73 int d = idx % d_model;
74 int d4 = 4 * d_model;
75
76 float i_val = ih[b * d4 + d] + hh[b * d4 + d];
77 float f_val = ih[b * d4 + d_model + d] + hh[b * d4 + d_model + d];
78 float g_val = ih[b * d4 + 2*d_model + d] + hh[b * d4 + 2*d_model + d];
79 float o_val = ih[b * d4 + 3*d_model + d] + hh[b * d4 + 3*d_model + d];
80
81 #if GPU_REPRO_MATH
82 i_val = repro_sigmoid(i_val);
83 f_val = repro_sigmoid(f_val);
84 g_val = repro_tanh(g_val);
85 o_val = repro_sigmoid(o_val);
86 #else
87 i_val = 1.0f / (1.0f + expf(-i_val));
88 f_val = 1.0f / (1.0f + expf(-f_val));
89 g_val = tanhf(g_val);
90 o_val = 1.0f / (1.0f + expf(-o_val));
91 #endif
92
93 float c = __fadd_rn(__fmul_rn(f_val, c_state[idx]), __fmul_rn(i_val, g_val));
94 #if GPU_REPRO_MATH
95 float h = __fmul_rn(o_val, repro_tanh(c));
96 #else
97 float h = __fmul_rn(o_val, tanhf(c));
98 #endif
99
100 c_state[idx] = c;
101 h_state[idx] = h;
102 }
103 }
104}

115// ==========================================
116// GRU Fused Gate Kernel
117// ==========================================
118// Gate layout (PyTorch convention): [r, z, n] each of size d_model
119// r = sigmoid(ih_r + hh_r) — reset gate
120// z = sigmoid(ih_z + hh_z) — update gate
121// n = tanh(ih_n + r * hh_n) — new gate
122// h_new = (1 - z) * n + z * h_old
123
124__global__ void ker_gru_fused_batch(const float* ih, const float* hh,
125 float* h_state,
126 int d_model, size_t total) {
127 size_t idx = (size_t)blockIdx.x * blockDim.x + threadIdx.x;
128 if (idx >= total) return;
129
130 size_t b = idx / d_model;
131 int d = idx % d_model;
132 int d3 = 3 * d_model;
133
134 float r_i = ih[b * d3 + d];
135 float z_i = ih[b * d3 + d_model + d];
136 float n_i = ih[b * d3 + 2*d_model + d];
137
138 float r_h = hh[b * d3 + d];
139 float z_h = hh[b * d3 + d_model + d];
140 float n_h = hh[b * d3 + 2*d_model + d];
141
142 #if GPU_REPRO_MATH
143 float r = repro_sigmoid(__fadd_rn(r_i, r_h));
144 float z = repro_sigmoid(__fadd_rn(z_i, z_h));
145 float n = repro_tanh(__fadd_rn(n_i, __fmul_rn(r, n_h)));
146 #else
147 float r = 1.0f / (1.0f + expf(-(r_i + r_h)));
148 float z = 1.0f / (1.0f + expf(-(z_i + z_h)));
149 float n = tanhf(n_i + r * n_h);
150 #endif
151
152 h_state[idx] = __fadd_rn(__fmul_rn(1.0f - z, n), __fmul_rn(z, h_state[idx]));
153}

163// ==========================================
164// minGRU activation g(x) = x >= 0 ? x + 0.5 : sigmoid(x)
165// (Appendix B.3, Feng et al. 2024)
166// ==========================================
167__device__ __forceinline__ float mingru_g(float x) {
168 return x >= 0.0f ? (x + 0.5f) : (1.0f / (1.0f + expf(-x)));
169}

171// ==========================================
172// minGRU Fused Step Kernel
173// ==========================================
174// z = sigmoid(z_logits)
175// h_tilde = g(h_raw) — positive activation (log-space compatible)
176// h_new = (1 - z) * h_prev + z * h_tilde
177
178__global__ void ker_mingru_fused_batch(const float* z_logits, const float* h_raw,
179 float* h_state, size_t total) {
180 size_t base = ((size_t)blockIdx.x * blockDim.x + threadIdx.x) * 4;
181 if (base + 3 < total) {
182 float4 zv = reinterpret_cast<const float4*>(&z_logits[base])[0];
183 float4 hv = reinterpret_cast<const float4*>(&h_raw[base])[0];
184 float4 hs = reinterpret_cast<float4*>(&h_state[base])[0];
185
186 float z_arr[4] = {zv.x, zv.y, zv.z, zv.w};
187 float h_arr[4] = {hv.x, hv.y, hv.z, hv.w};
188 float s_arr[4] = {hs.x, hs.y, hs.z, hs.w};
189
190 #pragma unroll
191 for (int n = 0; n < 4; ++n) {
192 #if GPU_REPRO_MATH
193 float z = repro_sigmoid(z_arr[n]);
194 #else
195 float z = 1.0f / (1.0f + expf(-z_arr[n]));
196 #endif
197 float ht = mingru_g(h_arr[n]);
198 s_arr[n] = __fadd_rn(__fmul_rn(1.0f - z, s_arr[n]), __fmul_rn(z, ht));
199 }
200
201 reinterpret_cast<float4*>(&h_state[base])[0] = make_float4(s_arr[0], s_arr[1], s_arr[2], s_arr[3]);
202 } else {
203 for (size_t idx = base; idx < total && idx < base + 4; ++idx) {
204 #if GPU_REPRO_MATH
205 float z = repro_sigmoid(z_logits[idx]);
206 #else
207 float z = 1.0f / (1.0f + expf(-z_logits[idx]));
208 #endif
209 float ht = mingru_g(h_raw[idx]);
210 h_state[idx] = __fadd_rn(__fmul_rn(1.0f - z, h_state[idx]), __fmul_rn(z, ht));
211 }
212 }
213}

223// ==========================================
224// minGRU Chunk Kernel (sequential over time)
225// ==========================================
226// Pre-computed: z_all [B, L, d], h_all [B, L, d]
227// Each thread handles one (batch, dim) pair and loops over L timesteps.
228// Layout: [b*L + t] * d_model + d (row-major [B, L, D])
229
230__global__ void ker_mingru_chunk(const float* z_all, const float* h_all,
231 float* h_state, float* out,
232 int batch, int length, int d_model,
233 size_t total_channels) {
234 size_t idx = (size_t)blockIdx.x * blockDim.x + threadIdx.x;
235 if (idx >= total_channels) return;
236
237 int b = idx / d_model;
238 int d = idx % d_model;
239
240 float h = h_state[idx];
241
242 for (int t = 0; t < length; ++t) {
243 size_t flat = (size_t)(b * length + t) * d_model + d;
244 #if GPU_REPRO_MATH
245 float z = repro_sigmoid(z_all[flat]);
246 #else
247 float z = 1.0f / (1.0f + expf(-z_all[flat]));
248 #endif
249 float ht = mingru_g(h_all[flat]);
250 h = __fadd_rn(__fmul_rn(1.0f - z, h), __fmul_rn(z, ht));
251 out[flat] = h;
252 }
253
254 h_state[idx] = h;
255}

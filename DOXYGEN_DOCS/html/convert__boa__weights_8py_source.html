8# Binary format header:
9# Bytes 0-3: "BOA\x00" (fp32) or "BOA\x01" (fp16)
10# Bytes 4+: weight data
11MAGIC_FP32 = b"BOA\x00"
12MAGIC_FP16 = b"BOA\x01"

16def write_tensor(f, t):
17 if g_use_fp16:
18 f.write(t.half().cpu().numpy().tobytes())
19 else:
20 f.write(t.float().cpu().numpy().tobytes())

23def detect_backbone(sd):
24 """Auto-detect backbone type from state dict keys."""
25 for key in sd:
26 if ".mamba." in key:
27 return "mamba"
28 if ".lstm." in key:
29 return "lstm"
30 if ".gru." in key and ".mingru." not in key:
31 return "gru"
32 if ".mingru." in key:
33 return "mingru"
34 return "mamba" # default

37def export_mamba_block(f, sd, prefix):
38 """Export Mamba backbone weights for one block."""
39 m_pre = f"{prefix}mamba."
40 # in_proj
41 print(f" Writing {m_pre}in_proj.weight {sd[f'{m_pre}in_proj.weight'].shape}")
42 write_tensor(f, sd[f"{m_pre}in_proj.weight"])
43
44 if f"{m_pre}in_proj.bias" in sd:
45 print(f" Writing {m_pre}in_proj.bias {sd[f'{m_pre}in_proj.bias'].shape}")
46 write_tensor(f, sd[f"{m_pre}in_proj.bias"])
47 else:
48 print(f" Writing {m_pre}in_proj.bias (ZEROS - not found)")
49 w_shape = sd[f"{m_pre}in_proj.weight"].shape
50 bias_sim = torch.zeros(w_shape[0], dtype=torch.float32)
51 write_tensor(f, bias_sim)
52
53 # conv1d
54 print(f" Writing {m_pre}conv1d.weight")
55 write_tensor(f, sd[f"{m_pre}conv1d.weight"])
56 if f"{m_pre}conv1d.bias" in sd:
57 print(f" Writing {m_pre}conv1d.bias")
58 write_tensor(f, sd[f"{m_pre}conv1d.bias"])
59
60 # x_proj
61 print(f" Writing {m_pre}x_proj.weight {sd[f'{m_pre}x_proj.weight'].shape}")
62 write_tensor(f, sd[f"{m_pre}x_proj.weight"])

73 # D

90def export_lstm_block(f, sd, prefix):
91 """Export LSTM backbone weights for one block.
92
93 PyTorch nn.LSTM keys:
94 lstm.weight_ih_l0 [4*d, d]
95 lstm.weight_hh_l0 [4*d, d]
96 lstm.bias_ih_l0 [4*d]
97 lstm.bias_hh_l0 [4*d]
98
99 C++ load order: w_ih, b_ih, w_hh, b_hh
100 """
101 m_pre = f"{prefix}lstm."
102 for name in ["weight_ih_l0", "bias_ih_l0", "weight_hh_l0", "bias_hh_l0"]:
103 key = f"{m_pre}{name}"
104 print(f" Writing {key} {sd[key].shape}")
105 write_tensor(f, sd[key])

153def convert(model_path, output_path, backbone=None):
154 global g_use_fp16
155 print(f"Loading {model_path}...")
156 sd = torch.load(model_path, map_location='cpu', weights_only=False)
157 if "model_state_dict" in sd:
158 sd = sd["model_state_dict"]
159 elif "state_dict" in sd:
160 sd = sd["state_dict"]
161
162 # Auto-detect backbone if not specified
163 if backbone is None:
164 backbone = detect_backbone(sd)
165 print(f"Backbone: {backbone}")
166
167 if backbone not in BACKBONE_EXPORTERS:
168 print(f"ERROR: Unknown backbone '{backbone}'. Supported: {list(BACKBONE_EXPORTERS.keys())}")
169 return
170
171 export_block_fn = BACKBONE_EXPORTERS[backbone]
172
173 # Auto-detect fp16 weights from the checkpoint
174 sample_key = next(k for k in sd if 'weight' in k)
175 if sd[sample_key].dtype == torch.float16:
176 print("Detected fp16 weights in checkpoint — saving as fp16 binary")
177 g_use_fp16 = True
178
179 precision_str = "fp16" if g_use_fp16 else "fp32"
180 print(f"Output precision: {precision_str}")
181
182 with open(output_path, 'wb') as f:
183 # Write header
184 f.write(MAGIC_FP16 if g_use_fp16 else MAGIC_FP32)
185
186 # 1. Embedding
187 print("Exporting Embedding...")
188 if "embedding.weight" in sd:
189 write_tensor(f, sd["embedding.weight"])
190 else:
191 print("ERROR: embedding.weight not found")
192 return
193
194 # 2. Blocks
195 i = 0
196 while True:
197 prefix = f"blocks.{i}."
198 if f"{prefix}ln1.weight" not in sd:
199 break
200
201 print(f"Exporting Block {i} ({backbone})...")
202
203 # LN1
204 write_tensor(f, sd[f"{prefix}ln1.weight"])
205 write_tensor(f, sd[f"{prefix}ln1.bias"])
206
207 # Backbone-specific weights
208 export_block_fn(f, sd, prefix)
209
210 # LN2
211 write_tensor(f, sd[f"{prefix}ln2.weight"])
212 write_tensor(f, sd[f"{prefix}ln2.bias"])
213
214 # FF — handle both Sequential (ff.0, ff.2) and named (ff.net.0, ff.net.2)
215 ff_w1_key = f"{prefix}ff.0.weight"
216 if ff_w1_key not in sd:
217 ff_w1_key = f"{prefix}ff.net.0.weight"
218 ff_prefix = ff_w1_key.rsplit(".weight", 1)[0].rsplit(".0", 1)[0]
219
220 write_tensor(f, sd[f"{ff_prefix}.0.weight"])
221 write_tensor(f, sd[f"{ff_prefix}.0.bias"])
222 write_tensor(f, sd[f"{ff_prefix}.2.weight"])
223 write_tensor(f, sd[f"{ff_prefix}.2.bias"])
224
225 i += 1
226
227 print(f"Exported {i} blocks.")
228
229 # 2b. Final RMSNorm (non-mamba backbones)
230 if "final_norm.weight" in sd:
231 print("Exporting final_norm (RMSNorm)...")
232 write_tensor(f, sd["final_norm.weight"])
233
234 # 3. Head
235 print("Exporting Head...")
236 write_tensor(f, sd["head.0.weight"])
237 write_tensor(f, sd["head.0.bias"])
238 write_tensor(f, sd["head.2.weight"])
239 write_tensor(f, sd["head.2.bias"])
240
241 print(f"Done. Saved to {output_path}")

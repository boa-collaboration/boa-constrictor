Detailed Description

RWKV-6 time-mixing with optimised WKV computation.

Uses FLA Triton kernels (``pip install flash-linear-attention``) when
available on CUDA, otherwise falls back to a pure-PyTorch chunk-wise
parallel implementation (O(C²) matmuls per chunk, O(L/C) sequential
chunk iterations).  Both paths are numerically equivalent.

544 def _chunk_wkv(self, r, k, v, w, initial_state):
545 """Chunk-wise parallel WKV (pure PyTorch, works on CPU & GPU).
546
547 Args:
548 r, k, v: [B, L, H, dh]
549 w: [B, L, H, dh] (log-decay, negative)
550 initial_state: [B, H, dh, dh] or None
551 Returns:
552 out: [B, L, H, dh], state: [B, H, dh, dh]
553 """
554 B, L, H, dh = r.shape
555 C = self.chunk_size
556
557 # Pad to multiple of C
558 pad = (C - L % C) % C
559 if pad > 0:
560 r = F.pad(r, (0, 0, 0, 0, 0, pad))
561 k = F.pad(k, (0, 0, 0, 0, 0, pad))
562 v = F.pad(v, (0, 0, 0, 0, 0, pad))
563 w = F.pad(w, (0, 0, 0, 0, 0, pad))
564 Lp = L + pad
565 nc = Lp // C
566
567 r = r.view(B, nc, C, H, dh)
568 k = k.view(B, nc, C, H, dh)
569 v = v.view(B, nc, C, H, dh)
570 w = w.view(B, nc, C, H, dh)
571
572 w_cumsum = w.cumsum(dim=2)
573
574 # Intra-chunk: causal attention with decay
575 rel_log = w_cumsum.unsqueeze(3) - w_cumsum.unsqueeze(2)
576 causal = torch.tril(torch.ones(C, C, device=r.device, dtype=torch.bool))
577 causal = causal.view(1, 1, C, C, 1, 1)
578
579 attn = (r.unsqueeze(3) * k.unsqueeze(2) * torch.exp(rel_log))
580 attn = attn.masked_fill(~causal, 0.0).sum(dim=-1) # [B, nc, Ct, Cs, H]
581 o_intra = torch.einsum('bncsh,bnshd->bnchd',
582 attn.permute(0, 1, 2, 3, 4), v)
583
584 # Inter-chunk: propagate recurrent state
585 state = initial_state if initial_state is not None else r.new_zeros(B, H, dh, dh)
586 inter_list = []
587 for c_idx in range(nc):
588 r_c = r[:, c_idx]
589 decay_c = torch.exp(w_cumsum[:, c_idx])
590 o_inter_c = torch.einsum('bhde,bche->bchd', state, r_c * decay_c)
591 inter_list.append(o_inter_c)
592
593 total_decay = torch.exp(w_cumsum[:, c_idx, -1])
594 state = state * total_decay.unsqueeze(-1)
595 kv_decay = torch.exp(w_cumsum[:, c_idx, -1:] - w_cumsum[:, c_idx])
596 state = state + torch.einsum('bchd,bche->bhde',
597 k[:, c_idx] * kv_decay, v[:, c_idx])
598
599 o_inter = torch.stack(inter_list, dim=1)
600 out = (o_intra + o_inter).reshape(B, Lp, H, dh)
601 if pad > 0:
602 out = out[:, :L]
603 return out, state

643 """Single-token step. x, last_x: [B, D], state: [B, H, dh, dh]."""
644 B = x.shape[0]
645 H, dh = self.n_heads, self.d_head
646
647 r = self.W_r(x * self.mix_r + last_x * (1 - self.mix_r)).view(B, H, dh)
648 k = self.W_k(x * self.mix_k + last_x * (1 - self.mix_k)).view(B, H, dh)
649 v = self.W_v(x * self.mix_v + last_x * (1 - self.mix_v)).view(B, H, dh)
650 g = torch.sigmoid(self.W_g(x * self.mix_g + last_x * (1 - self.mix_g)))
651
652 k = F.normalize(k, p=2, dim=-1)
653
654 w = self._decay(self.base_decay, self.W_decay(x).view(B, H, dh))
655 decay = torch.exp(w)
